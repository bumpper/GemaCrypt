1:1	בְּמַרְבִּ֤ית חֲלָקִים֙ וּבְמַרְבִּ֣ית דְּרָכִ֔ים דִּבֶּ֨ר הָֽאֱלֹהִ֧ים לְפָנִ֛ים אֶל־הָֽאָב֖וֹת בַּנְּבִיאִֽים.
1:2	וּבַיָּמִ֨ים הָאַֽחֲרוֹנִ֜ים הָאֵ֗לֶּה דִּבֶּ֤ר אֵלֵ֨ינוּ֨ בַּבֵּ֔ן אֲשֶׁ֛ר שָׁ֥ת אֹת֖וֹ לְיוֹרֵ֣שׁ הַכֹּ֑ל אֲשֶׁ֣ר עַל־יָד֔וֹ עָשָׂ֖ה גַּ֥ם אֶת־הָעֽוֹלָמִֽים.
1:3	אֲ֠שֶׁר בִּהְיוֹת֞וֹ זֹ֤הַר כְּבוֹדוֹ֨ וְצֶ֣לֶם הֲוָֽיָת֔וֹ וְנֹשֵׂ֥א אֶת־הַכֹּ֖ל בִּדְבַ֣ר גְּבֽוּרָת֑וֹ כַּֽעֲשׂוֹת֤וֹ בְעַצְמוֹ֨ אֶת־נִקְי֣וֹן חַטּאוֹתֵ֔ינוּ יָשַׁ֛ב לִימִ֥ין הַגְּדֻלָּ֖ה בַּמְּרוֹמִֽים.
1:4	וְכֹ֥ה נִהְיָ֖ה ט֣וֹב מִן־הַמַּלְאָכִ֑ים כְּפִ֣י מָ֗ה אֲשֶׁ֥ר יָרַ֛שׁ שֵׁ֥ם נַֽעֲלֶ֖ה עֲלֵיהֶֽם.
1:5	כִּ֞י לְמִ֤י מִן־הַמַּלְאָכִים֨ אָמַ֣ר מֵֽעוֹלָ֔ם בְּנִ֣י אַתָּ֔ה אֲנִ֖י הַיּ֣וֹם יְלִדְתִּ֑יךָ וְע֕וֹד אֲנִי֨ אֶהְיֶה־לּ֣וֹ לְאָ֔ב וְה֖וּא יִהְיֶה־לִּ֥י לְבֵֽן.
1:6	וְע֗וֹד כַּֽ֠אֲשֶׁר ה֣וּא מֵבִ֧יא אֶת־הַבְּכ֛וֹר אֶל־הַתֵּבֵ֖ל ה֣וּא אֹמֵ֑ר וְיִשְׁתַּֽחֲו֥וּ ל֖וֹ כָּל־מַלְאֲכֵ֥י הָֽאֱלֹהִֽים.
1:7	וְאָמְנָ֕ם עַל־הַמַּלְאָכִ֖ים ה֣וּא אֹמֵ֑ר עֹשֶׂ֤ה מַלְאָכָיו֨ רוּח֔וֹת וּמְשָֽׁרְתָ֖יו אֵ֥שׁ לֹהֵֽט.
1:8	וְאֶל־הַבֵּ֕ן כִּסְאֲךָ֥ אֱלֹהִ֖ים עוֹלָ֣ם וָעֶ֑ד שֵׁ֥בֶט מִישׁ֖וֹר שֵׁ֥בֶט מַלְכוּתֶֽךָ.
1:9	אָהַ֥בְתָּ צֶ֖דֶק וַתִּשְׂנָ֣א רֶ֑שַׁע עַל־כֵּ֗ן מְשָֽׁחֲךָ֤ ׀ אֱלֹהִים֨ אֱלֹהֶ֔יךָ שֶׁ֥מֶן שָׂשׂ֖וֹן מֵֽחֲבֵרֶֽיךָ.
1:10	וְע֕וֹד אַתָּ֣ה אֲדֹנָ֔י לְפָנִ֖ים הָאָ֣רֶץ יָסָ֑דְתָּ וּמַֽעֲשֵׂ֥י יָדֶ֖יךָ הַשָּׁמָֽיִם.
1:11	הֵ֥מָּה יֹאבֵ֖דוּ וְאַתָּ֣ה תַֽעֲמ֑וֹד וְכֻלָּ֖ם כַּבֶּ֥גֶד יִבְלֽוּ.
1:12	וְכַלְּב֥וּשׁ תַּֽחֲלִיפֵ֖ם וְיַֽחֲלֹ֑פוּ וְאַתָּ֣ה ה֔וּא וּשְׁנוֹתֶ֖יךָ לֹ֥א יִתָּֽמּוּ.
1:13	וְאֶל־מִ֥י מִן־הַמַּלְאָכִ֖ים אָמַ֣ר מֵֽעוֹלָ֑ם שֵׁ֚ב לִֽימִינִ֔י עַד־אָשִׁ֥ית אֹֽיְבֶ֖יךָ הַד֥וֹם לְרַגְלֶֽיךָ.
1:14	הֲלֹ֣א כֻלָּ֔ם רוּח֥וֹת מְשָֽׁרְתִ֖ים הֵ֑מָּה נִשְׁלָחִ֣ים לְשֵׁר֔וּת בַּֽעֲב֛וּר הָֽעֲתִידִ֥ים לָרֶ֖שֶׁת אֶת־הַתְּשׁוּעָֽה.
2:1	בַּֽעֲב֣וּר זֹ֗את צָרִ֥יךְ לָ֨נוּ֨ עַל־יֶ֔תֶר לְהַקְשִׁ֖יב אֶל־הַדְּבָרִ֣ים הַנִּשְׁמָעִ֑ים פֶּ֖ן נִנָּגֵֽר.
2:2	כִּ֗י אִם־הַדָּבָ֧ר הַנֶּֽאֱמָ֛ר עַל־פִּ֥י מַלְאָכִ֖ים נִהְיָ֣ה נִצָּ֑ב וְכָל־עֲבֵרָ֣ה וּמְרִ֔י לָקַ֖ח גְּמ֥וּל צֶֽדֶק.
2:3	אֵ֚יךְ נִבְרַ֣ח אֲנַ֔חְנוּ בְּהַשְׁמִיטֵ֛נוּ תְּשׁוּעָ֥ה גְדוֹלָ֖ה כָּזֹ֑את אֲשֶׁ֞ר הֵחֵ֣לָּה בְרֵאשִׁ֗ית לְהֵֽאָמֵר֨ עַל־פִּ֣י הָֽאָד֔וֹן וְקֻיְּמָ֥ה לָ֖נוּ מֵאֵ֥ת הַשֹּֽׁמְעִֽים.
2:4	בְּהָעֵ֞ד גַּ֣ם הָֽאֱלֹהִ֗ים בְּאוֹת֨וֹת וּבְמֽוֹפְתִ֜ים וּבִגְבוּר֣וֹת שׁוֹנ֗וֹת וּבְחִלּוּקֵ֛י ר֥וּחַ הַקֹּ֖דֶשׁ כְּפִ֥י רְצוֹנֽוֹ.
2:5	כִּי֩ לֹ֨א לְמַלְאָכִ֜ים שָׁ֤ת תַּחְתֵּיהֶם֨ אֶת־הַתֵּבֵ֣ל הָֽעֲתִידָ֔ה אֲשֶׁ֛ר אֲנַ֥חְנוּ מְדַבְּרִ֖ים עַל־אֹֽדוֹתֶֽיהָ.
2:6	אַ֣ךְ הֵעִ֥יד אִ֛ישׁ בְּמָק֥וֹם אֶחָ֖ד לֵאמֹ֑ר מָ֤ה אֱנוֹשׁ֨ כִּ֣י תִזְכְּרֶ֔נּוּ וּבֶן־אָדָ֖ם כִּ֥י תִפְקְדֶֽנּוּ.
2:7	וַתְּחַסְּרֵ֨הוּ֨ מְעַ֣ט מֵֽאֱלֹהִ֔ים וְכָב֥וֹד וְהָדָ֖ר תְּעַטְּרֵ֑הוּ וַתַּמְשִׁילֵ֖הוּ בְּמַֽעֲשֵׂ֥י יָדֶֽיךָ.
2:8	הַכֹּ֗ל שַׁתָּה֘ תַּ֣חַת רַגְלָיו֒ כִּ֗י בְּהָשִׁ֤ית אֶת־הַכֹּל֨ תַּחְתָּ֔יו מְא֨וּמָה֨ לֹ֣א הִשְׁאִ֔יר בִּלְתִּ֥י מוּשָׁ֖ת תַּחְתָּ֑יו אַ֣ךְ עַתָּ֗ה ע֤וֹד אֵינֶ֣נּוּ רֹאִ֔ים אֶת־הַכֹּ֖ל מוּשָׁ֥ת תַּחְתָּֽיו.
2:9	אַ֣ךְ אֶת־יֵשׁ֗וּעַ הַמְּחֻסָּ֨ר מְעַ֤ט מִמַּלְאָכִים֨ בִּגְלַ֣ל עִנּ֣וּי הַמָּ֔וֶת רֹאִ֣ים אֲנַ֔חְנוּ מְעֻטָּ֖ר בְּכָב֣וֹד וּבְהָדָ֑ר לְמַ֨עַן֨ בְּחֶ֣סֶד הָֽאֱלֹהִ֔ים יִטְעַ֥ם מָ֖וֶת בְּעַ֥ד כָּל־אִֽישׁ.
2:10	כִּי֩ יָֽאֲתָ֨ה ל֜וֹ אֲשֶׁ֨ר הָכֹּ֤ל בַּֽעֲבוּרוֹ֨ וְהַכֹּ֣ל עַל־יָדָ֔יו הַמֵּבִ֛יא בָּנִ֥ים רַבִּ֖ים לְכָב֑וֹד לְהַשְׁלִ֛ים עַל־יְדֵ֥י עִנּוּיִ֖ים אֶת־נְשִׂ֥יא יְשֽׁוּעָתָֽם.
2:11	כִּ֣י גַ֧ם הַמְּקַדֵּ֛שׁ גַּ֥ם הַמְּקֻדָּשִׁ֖ים מֵֽאֶחָ֣ד ׀ כֻּלָּ֑ם אֲשֶׁר֨ בְּסִבָּ֣ה זֹ֔את אֵינֶ֣נּוּ ב֔וֹשׁ לִקְרֹ֥א לָהֶ֖ם אַחִֽים.
2:12	לְאמֹ֑ר אֲסַפְּרָ֤ה שִׁמְךָ֨ לְאֶחָ֔י בְּת֥וֹךְ קָהָ֖ל אֲהַלְלֶֽךָּ.
2:13	וְע֕וֹד אֲנִ֖י אֶבְטַ֣ח בּ֑וֹ וְע֕וֹד הִנֵּ֣ה אָֽנֹכִ֔י וְהַיְלָדִ֗ים אֲשֶׁ֛ר נָ֥תַן לִ֖י הָֽאֱלֹהִֽים.
2:14	לָ֠כֵן בַּֽאֲשֶׁ֞ר הַיְלָדִ֤ים חָֽבְרוּ֨ בְּבָשָׂ֣ר וָדָ֔ם גַּ֣ם ה֧וּא ׀ כְּמוֹ־כֵ֛ן לָקַ֥ח חֵ֖לֶק בָּאֵ֑לֶּה לְמַ֣עַן ׀ עַל־יְדֵ֣י הַמָּ֗וֶת יְאַבֵּ֞ד אֵ֣ת אֲשֶׁ֤ר יֵֽשׁ־לוֹ֨ גְּבוּרַ֣ת הַמָּ֔וֶת זֶ֥ה ה֖וּא הַשָּׂטָֽן.
2:15	וִֽיחַלֵּ֣ץ אֶת־כָּל־אֵ֔לֶּה אֲשֶׁר֨ בְּאֵ֣ימַת מָ֔וֶת כָּל־יְמֵ֣י חַיֵּיהֶ֔ם הָי֥וּ אֲשֵׁמִ֖ים לְעַבְדֽוּת.
2:16	כִּ֥י אָמְנָ֛ם לֹ֥א בַמַּלְאָכִ֖ים ה֣וּא מַֽחֲזִ֑יק כִּ֛י אִם־בְּזֶ֥רַע אַבְרָהָ֖ם ה֥וּא מַֽחֲזִֽיק.
2:17	לָכֵ֗ן הָיָ֤ה רָאוּי֨ ל֔וֹ לְהִדָּמ֥וֹת בַּכֹּ֖ל אֶל־הָֽאַחִ֑ים לְמַ֨עַן יִהְיֶ֜ה רַח֗וּם וְכֹהֵ֨ן גָּד֤וֹל נֶֽאֱמָן֨ בַּדְּבָרִ֣ים אֲשֶׁ֣ר לָֽאֱלֹהִ֔ים לְמַ֥עַן כַּפֵּ֖ר אֶת־חַטּ֥אוֹת הָעָֽם.
2:18	כִּ֗י בַּֽאֲשֶׁ֛ר ה֥וּא עֻנָּ֖ה בִּהְיוֹת֣וֹ מְנֻסֶּ֑ה ה֛וּא יָכֹ֥ל לַֽעֲז֖וֹר אֶת־הַמְּנֻסִּֽים.
3:1	לָכֵן֨ אַחִ֣ים קְדוֹשִׁ֔ים מִתְחַבְּרֵ֖י הָקְּרִיאָ֣ה הַשָּֽׁמַיְמִ֑ית הִתְבּ֨וֹנְנ֜וּ אֶת־הַשָּׁלִ֨יחַ וְהַכֹּהֵ֤ן הַגָּדוֹל֨ אֲשֶׁ֣ר לְהוֹדָֽאָתֵ֔נוּ יֵשׁ֖וּעַ הַמָּשִֽׁיחַ.
3:2	אֲשֶׁ֛ר הָיָ֥ה נֶֽאֱמָ֖ן לְעֹשֵׂ֑הוּ כְּמ֛וֹ גַּ֥ם מֹשֶׁ֖ה בְּכָל־בֵּיתֽוֹ.
3:3	כִּ֚י זֶ֣ה נֶחְשַׁ֔ב שֹׁוֶ֛ה לְכָב֥וֹד רַ֖ב מִמֹּשֶׁ֑ה בַּֽאֲשֶׁ֛ר רַ֥ב מִן־הַבַּ֖יִת כְּב֥וֹד מְכֽוֹנְנֵֽהוּ.
3:4	כִּ֚י כָּל־בַּ֔יִת מְכוֹנָ֖ן מֵֽאֵי־זֶ֣ה אִ֑ישׁ אַ֛ךְ הַמְּכוֹנֵ֥ן אֶת־הַכֹּ֖ל הָֽאֱלֹהִֽים.
3:5	וּמֹשֶׁ֣ה אָמְנָ֔ם נֶֽאֱמָ֥ן בְּכָל־בֵּית֖וֹ כִּמְשָׁרֵ֑ת לְעֵד֥וּת הַדְּבָרִ֖ים הָֽעֲתִידִ֥ים לְהֵֽאָמֵֽר.
3:6	אַ֚ךְ הַמָּשִׁ֔יחַ כְּמ֥וֹ בֵ֖ן עַל־בֵּית֑וֹ אֲשֶׁר֨ בֵּית֣וֹ אֲנַ֔חְנוּ אִם֩ אַ֨ךְ אֶת־הַבִּטְחָ֜ה וְאֵ֣ת ׀ תְּהִלַּ֣ת הַתִּקְוָ֗ה אֲנַ֧חְנוּ אֹֽחֲזִ֛ים מְקֻיֶּ֖מֶת עַד־הַקֵּֽץ.
3:7	לָכֵ֕ן כַּֽאֲשֶׁ֥ר אֹמֵ֖ר ר֣וּחַ הַקֹּ֑דֶשׁ הַיּ֖וֹם אִם־בְּקֹל֥וֹ תִשְׁמָֽעוּ.
3:8	אַל־תַּקְשׁ֥וּ לִבּֽוֹתֵיכֶ֖ם כִּמְרִיבָ֑ה כְּי֥וֹם מַסָּ֖ה בַּמִּדְבָּֽר.
3:9	כַּֽאֲשֶׁ֥ר נִסּ֖וּנִי אֲבֽוֹתֵיכֶ֑ם בְּחָנ֕וּנִי גַּם־רָא֥וּ פָֽעֳלַ֖י אַרְבָּעִ֥ים שָׁנָֽה.
3:10	לָכֵ֕ן אָק֖וּט בַּדּ֣וֹר הַה֑וּא וָֽאֹמַ֗ר תָּמִ֞יד תֹּעֵ֤י לֵבָב֨ הֵ֔ם וְהֵ֖ם לֹא־יָֽדְע֥וּ דְרָכָֽי.
3:11	אֲשֶׁ֥ר נִשְׁבַּ֖עְתִּי בְּאַפִּ֑י אִם־יְבֹא֖וּן אֶל־מְנֽוּחָתִֽי.
3:12	הִשָּֽׁמְר֨וּ לָכֶ֜ם אַחִ֗ים פֶּן־יִהְיֶ֞ה בְּאֵי־זֶ֥ה אִישׁ֨ מִכֶּ֔ם לֵ֥ב רָ֖ע בְּלִֽי־אֵמ֑וּן בְּהִסּ֖וֹג מֵֽאֱלֹהִ֥ים חַיִּֽים.
3:13	כִּ֣י אִם־הַזְהִ֜ירוּ אִ֤ישׁ אֶת־רֵעֵ֨הוּ֨ י֣וֹם י֔וֹם כָּל־ע֖וֹד הַיּ֣וֹם ׀ נִקְרָ֑א לְמַ֗עַן לֹ֥א יָקְשֶׁ֛ה אִ֥ישׁ מִכֶּ֖ם בְּהַדָּחַ֥ת הַחֵֽטְא.
3:14	כִּ֛י מִתְחַבְּרֵ֥י הַמָּשִׁ֖יחַ נִהְיִ֑ינוּ אִם־אֵפ֞וֹא אֶת־רֵאשִׁ֣ית הַתּֽוּשִׁיָּ֗ה אֲנַ֧חְנוּ אֹֽחֲזִ֛ים מְקֻיֶּ֖מֶת עַד־הַקֵּֽץ.
3:15	בְּהֵ֣אָמֵ֔ר הַיּ֖וֹם אִם־בְּקֹל֣וֹ תִשְׁמָ֑עוּ אַל־תַּקְשׁ֥וּ לִבּֽוֹתֵיכֶ֖ם כִּמְרִיבָֽה.
3:16	כִּ֛י אֲחָדִ֥ים כְּשָׁמְעָ֖ם הֵרִ֑יבוּ אַ֗ךְ לֹ֛א כָּל־הַיֹּֽצְאִ֥ים מִמִּצְרַ֖יִם בְּיַד־מֹשֶֽׁה.
3:17	אַ֕ךְ בְּמִ֥י קָ֖ט אַרְבָּעִ֣ים שָׁנָ֑ה הֲלֹ֣א בַחֹֽטְאִ֔ים אֲשֶׁ֥ר פִּגְרֵיהֶ֖ם נָֽפְל֥וּ בַמִּדְבָּֽר.
3:18	וְאֶל־מִ֣י נִשְׁבַּ֔ע לְבִלְתִּ֥י ב֖וֹא אֶל־מְנֽוּחָת֑וֹ אִם־לֹ֕א אֶל־אֲשֶׁ֖ר לֹ֥א הֶֽאֱמִֽינוּ.
3:19	וַֽאֲנַ֣חְנוּ רֹאִ֔ים כִּ֣י לֹ֤א יָֽכְלוּ֨ לָב֔וֹא בַּֽעֲב֖וּר לֹֽא־אֵמֽוּן.
4:1	לָכֵ֣ן נִירָ֔א פֶּ֣ן בְּהִשָּׁאֵ֤ר הַבְטָחַת֨ בּ֣וֹא אֶל־מְנֽוּחָת֔וֹ יֵֽרָאֶ֛ה אִ֥ישׁ מִכֶּ֖ם לְהֵֽעָדֵֽר.
4:2	כִּ֣י גַ֤ם אֲנַ֨חְנוּ֨ הָיִ֣ינוּ מְבֻשָּׂרִ֔ים כְּמ֖וֹ גַּ֣ם הֵ֑ם אַךְ֩ לֹ֨א הוֹעִ֜יל דְּבַ֤ר הַשְּׁמִיעָה֨ לָהֶ֔ם מִבְּלִ֨י הֱיוֹת֧וֹ מְמֻזָּ֛ג עִ֥ם הָֽאֱמוּנָ֖ה לַשֹּֽׁמְעִֽים.
4:3	כִּ֞י אֲנַ֨חְנוּ אֲשֶׁ֤ר הֶֽאֱמַ֨נּוּ֨ בָּאִ֣ים אֶל־הַמְּנוּחָ֔ה כַּֽאֲשֶׁ֣ר אָמַ֗ר אֲשֶׁ֤ר נִשְׁבַּ֨עְתִּי֨ בְּאַפִּ֔י אִם־יְבֹא֖וּן אֶל־מְנֽוּחָתִ֑י וְאָכֵן֨ הַמַּֽעֲשִׂ֣ים כֻּלּ֔וּ לְמִ֖ן הִוָּסֵ֥ד הַתֵּבֵֽל.
4:4	כִּ֣י אָמַ֗ר בְּמָק֥וֹם אֶחָ֛ד עַל־הַיּ֥וֹם הַשְּׁבִיעִ֖י כָּ֑כָה וַיִּשְׁבֹּ֧ת הָֽאֱלֹהִ֛ים בַּיּ֥וֹם הַשְּׁבִיעִ֖י מִכָּל־מַֽעֲשָֽׂיו.
4:5	וּבָזֶ֖ה ע֑וֹד אִם־יְבֹא֖וּן אֶל־מְנֽוּחָתִֽי.
4:6	לָכֵן֨ בַּֽאֲשֶׁ֣ר נִשְׁאָ֔ר לָב֥וֹא אֲחָדִ֖ים אֵלֶ֑יהָ וְהַמְּבֻשָּׂרִ֤ים בָּרִֽאשׁוֹנָה֨ לֹא־בָ֣אוּ בָ֔הּ בַּֽעֲב֖וּר לֹֽא־אֵמֽוּן.
4:7	עוֹד֩ ה֨וּא מַגְבִּ֜יל י֣וֹם אֶחָד֘ הַיּוֹם֒ בְּאָמְר֣וֹ בְדָוִ֔ד אַֽחֲרֵ֛י זְמָ֥ן הַרְבֵּ֖ה כָּזֶ֑ה כַּֽאֲשֶׁ֣ר ׀ נֶֽאֱמַ֣ר לְפָנִ֗ים הַיּוֹם֨ אִם־בְּקֹל֣וֹ תִשְׁמָ֔עוּ אַל־תַּקְשׁ֖וּ לִבּֽוֹתֵיכֶֽם.
4:8	כִּ֛י אִם־הֵנִ֥יחַ יְהוֹשֻׁ֖עַ אֹתָ֑ם לֹ֥א דִבֵּ֛ר אַֽחֲרֵי־כֵ֖ן עַל־י֥וֹם אַחֵֽר.
4:9	לָכֵ֕ן נִשְׁאָ֥ר שַׁבָּת֖וֹן לְעַ֥ם הָֽאֱלֹהִֽים.
4:10	כִּ֗י הַבָּא֨ אֶל־מְנֽוּחָת֔וֹ גַּ֥ם ה֖וּא שָׁבַ֣ת מִמַּֽעֲשָׂ֑יו כַּֽאֲשֶׁר֨ הָֽאֱלֹהִ֔ים מֵאֵ֖לֶּה אֲשֶׁ֥ר לֽוֹ.
4:11	לָכֵן֨ נֶֽחֱרַ֣ץ נָ֔א לָב֖וֹא אֶל־הַמְּנוּחָ֣ה הַהִ֑יא לְמַ֣עַן ׀ לֹֽא־יִפּ֣וֹל אִ֗ישׁ בְּעֶ֛צֶם דִּמְי֥וֹן בְּלִֽי־אֵמ֖וּן הַזֶּֽה.
4:12	כִּי֩ חַ֨י ה֜וּא דְּבַ֤ר הָֽאֱלֹהִים֨ וְרַב־פֹּ֔עַל וְחַ֖ד מִכָּל־חֶ֣רֶב פִּֽיפִיּ֑וֹת וְדֹקֵ֞ר עַ֤ד לְהַבְדָּלַת֨ נֶ֣פֶשׁ וָר֔וּחַ גַּ֤ם פְּרָקִים֨ גַּ֣ם מֹ֔חַ וּמְבַקֵּ֛ר מַחְשְׁב֥וֹת הַלֵּ֖ב וְעֶשְׁתֹּֽנוֹתָֽיו.
4:13	וְאֵ֥ין יְצ֖וּר מְכֻסֶּ֣ה מִפָּנָ֑יו כִּ֣י אִם־כָּל־דְּבָרִ֞ים עֲרוּמִ֤ים וַֽחֲשׂוּפִים֨ לְעֵינָ֔יו אֲשֶׁ֥ר אֵלָ֖יו לָ֥נוּ הַדָּבָֽר.
4:14	לָכֵ֡ן בִּהְי֣וֹת לָנוּ֩ כֹהֵ֨ן רֹ֜אשׁ גָּד֗וֹל אֲשֶׁ֤ר עָבַר֨ אֶת־הַשָּׁמַ֔יִם יֵשׁ֖וּעַ בֶּן־הָֽאֱלֹהִ֑ים נֹֽאחֲזָ֖ה אֶת־הַהֽוֹדָאָֽה.
4:15	כִּ֣י אֵ֥ין לָ֨נוּ֨ כֹּהֵ֣ן גָּד֔וֹל אֲשֶׁ֛ר אֵינֶ֥נּוּ יָכֹ֖ל לְהִתְעַנּ֣וֹת עִם־רִפְיוֹנֵ֑ינוּ כִּ֣י אִם־הַמְּנֻסֶּ֤ה בְכָל־דָּבָר֨ בְּשִׁוּ֔וּי מִבַּלְעֲדֵ֖י חֵֽטְא.
4:16	עַל־כֵּ֕ן נִקְרְבָ֥ה בְּבִטְחָ֖ה אֶל־כִּסֵּ֣א הַחָ֑סֶד לְמַ֨עַן֨ נִקַּ֣ח רַֽחֲמִ֔ים וְנִמְצָ֥א חֶ֛סֶד לְעֶזְרָ֖ה בְּעִתָּֽהּ.
5:1	כִּ֣י ׀ כָּל־כֹּהֵ֣ן גָּד֗וֹל הַלָּק֨וּחַ מִבְּנֵ֤י אָדָם֨ בַּֽעֲב֣וּר בְּנֵֽי־אָדָ֔ם ה֣וּא מוּשָׂ֔ם עַל־הַדְּבָרִ֖ים אֲשֶׁ֣ר לָֽאֱלֹהִ֑ים לְמַ֧עַן יַקְרִ֛יב מְנָח֥וֹת וּזְבָחִ֖ים עַל־חֲטָאִֽים.
5:2	אֲשֶׁ֨ר יָכֹ֤ל לְהִתְעַנּוֹת֨ בְּצֶ֔דֶק עִ֛ם בִּלְתִּ֥י יֽוֹדְעִ֖ים וְתֹעִ֑ים בַּֽאֲשֶׁ֥ר גַּם־ה֖וּא אָפ֥וּף בְּרִפְיֽוֹן.
5:3	וּבַֽעֲב֣וּר זֹ֔את חַיָּ֥ב ה֖וּא לְהַקְרִ֣יב עַל־חֲטָאִ֑ים כַּֽאֲשֶׁר֨ בְּעַ֣ד הָעָ֔ם כֵּ֥ן גַּ֖ם בְּעַ֥ד עַצְמֽוֹ.
5:4	וְאֵ֥ין אִ֛ישׁ לֹקֵ֥חַ לְעַצְמ֖וֹ אֶת־הַכָּב֑וֹד זֽוּלָתִ֗י הַמְּקֹרָא֨ מֵאֵ֣ת הָֽאֱלֹהִ֔ים כַּֽאֲשֶׁ֖ר גַּ֥ם אַֽהֲרֹֽן.
5:5	כֵּ֣ן ׀ גַּ֣ם הַמָּשִׁ֗יחַ לֹ֤א כִבֵּד֨ אֶת־עַצְמ֔וֹ לִהְי֖וֹת לְכֹהֵ֣ן גָּד֑וֹל כִּ֚י אִם־הָֽאֹמֵ֣ר אֵלָ֔יו בְּנִ֣י אַתָּ֔ה אֲנִ֖י הַיּ֥וֹם יְלִדְתִּֽיךָ.
5:6	כַּֽאֲשֶׁ֥ר גַּ֛ם בְּמָק֥וֹם אַחֵ֖ר ה֣וּא אֹמֵ֑ר אַתָּ֤ה כֹהֵן֨ לְעוֹלָ֔ם עַל־דִּבְרָתִ֖י מַלְכִּי־צֶֽדֶק.
5:7	אֲשֶׁ֞ר בִּימֵ֣י בְשָׂר֗וֹ הִקְרִיב֩ תְּפִלּ֨וֹת וְתַֽחֲנוּנִ֜ים בִּצְעָקָ֤ה חֲזָקָה֨ וּבִדְמָע֔וֹת אֶל־הַיָּכֹ֛ל לְהֽוֹשִׁ֥יעַ אֹת֖וֹ מִמָּ֑וֶת וְנִשְׁמַ֖ע מִן־הַיִּרְאָֽה.
5:8	וְאַף֨ כִּי־הָ֣יָה בֵ֔ן לָמַ֗ד מִן־הַדְּבָרִ֛ים אֲשֶׁ֥ר סָבַ֖ל אֶת־הַמִּשְׁמָֽעַת.
5:9	ווּבְהִשָּׁ֣לְמ֔וֹ נִהְיָה֨ לְכָל־הַשֹּֽׁמְעִ֣ים אֵלָ֔יו מְסַבֵּ֖ב תְּשׁוּעַ֥ת עֽוֹלָמִֽים.
5:10	בְּהִקָּרֵ֥א לוֹ֨ מֵאֵ֣ת הָֽאֱלֹהִ֔ים כֹּהֵ֣ן גָּד֔וֹל עַל־דִּבְרָתִ֖י מַלְכִּי־צֶֽדֶק.
5:11	אֲשֶׁ֣ר עַל־אֹֽדוֹתָ֗יו לָ֜נוּ הַרְבֵּ֧ה דְבָרִ֛ים וּקְשֵׁ֥י הַפִּתְר֖וֹן לְדַבֵּ֑ר בַּֽאֲשֶׁ֥ר נִהְיִיתֶ֖ם כְּבֵדֵ֥י הַשְּׁמִיעָֽה.
5:12	כִּ֣י אַ֠תֶּם הַֽחַיָּבִ֞ים לִהְי֣וֹת מְלַמְּדִים֘ מִפְּנֵ֣י הָעֵת֒ ע֗וֹד צֹ֣רֶךְ יֵ֤שׁ לָכֶם֨ לְלַמֵּ֣ד אֶתְכֶ֔ם אֵי־זֶ֣ה הֵ֔ם עִקְּרֵ֥י רֵאשִׁ֖ית מַֽאֲמָרֵ֣י הָֽאֱלֹהִ֑ים וְנִהְיִיתֶם֨ נִצְרָכֵ֣י חָלָ֔ב וְלֹ֖א מַֽאֲכָ֥ל חָזָֽק.
5:13	כִּ֗י כֹּ֣ל אֲשֶׁ֤ר חֶלְקוֹ֨ חָלָ֔ב ה֛וּא בִּלְתִּֽי־מְנֻסֵּ֖ה דְּבַ֣ר הַצְּדָקָ֑ה כִּ֖י עוֹלֵ֥ל הֽוּא.
5:14	אַ֚ךְ לַשְּׁלֵמִ֔ים הַמַּֽאֲכָ֖ל הֶֽחָזָ֑ק אֲשֶׁ֣ר ׀ בִּגְלַ֣ל הַשִּׁמּ֗וּשׁ יֵ֣שׁ לָהֶ֤ם הַֽחוּשִׁים֨ מָרְגָּלִ֔ים לְהַבְחָנַ֖ת ט֥וֹב וָרָֽע.
6:1	בַּֽעֲב֣וּר זֹ֔את בַּֽעֲז֗וֹב אֶת־דְּבַר֨ רֵאשִׁ֣ית הַמָּשִׁ֔יחַ אֶל־הַשְּׁלֵמ֖וּת נַֽעֲבֹ֑רָה בִּבְלִי֩ שׂ֨וּם ע֜וֹד יְס֤וֹד תְּשׁוּבָה֨ מִמַּֽעֲשִׂ֣ים מֵתִ֔ים וֶֽאֱמוּנָ֖ה בֵּֽאלֹהִֽים.
6:2	לֶ֥קַח טְבִיל֖וֹת וּסְמִיכַ֣ת יָדָ֑יִם וּתְקוּמַ֥ת הַמֵּתִ֖ים וּמִשְׁפַּ֥ט עוֹלָֽם.
6:3	וְזֹ֣את נַֽעֲשֶׂ֔ה אִם־אֵפ֖וֹא יַרְשֶׁ֥ה הָֽאֱלֹהִֽים.
6:4	כִּ֣י ׀ נִמְנָ֣ע ה֗וּא אֶל־הַנְּאוֹרִים֨ פַּ֣עַם אַחַ֔ת וְהַטֹּ֣עֲמִ֔ים אֵ֖ת הַמַּתָּנָ֣ה הַשָּֽׁמַיְמִ֑ית וְהַנִּהְיִ֕ים מִתְחַבְּרֵ֖י ר֥וּחַ הַקֹּֽדֶשׁ.
6:5	וְהַטֹּ֣עֲמִ֔ים אֵ֛ת דְּבַ֥ר הָֽאֱלֹהִ֖ים הַטּ֑וֹב וְאֶת־גְּבוּר֖וֹת הָֽעוֹלָ֥ם הַבָּֽא.
6:6	וְהַנְּסוֹגִ֣ים אָח֔וֹר לְחַדֵּ֥שׁ אֹתָ֛ם ע֖וֹד לִתְשׁוּבָ֑ה בִּהְיוֹתָם֩ מַצְלִיבִ֨ים ע֤וֹד לְנַפְשָׁם֨ אֵ֣ת בֶּן־הָֽאֱלֹהִ֔ים וְשָׂמִ֥ים אֹת֖וֹ לְרַֽאֲוָֽה.
6:7	כִּ֣י הָאָ֡רֶץ הַשֹּׁתָה֩ אֶת־הַמָּטָ֨ר הַבָּ֤א עָלֶ֨יהָ֨ פְּעָמִ֣ים רַבּ֔וֹת וְיוֹלֶ֤דֶת דֶּ֨שֶׁא֨ כָּשֵׁ֣ר לָאֵ֔לֶּה אֲשֶׁ֛ר הִ֥יא נֶֽעֱבָדָ֖ה בַּֽעֲבוּרָ֑ם תִּרְוֶ֥ה בְרָכָ֖ה מֵאֵ֥ת הָֽאֱלֹהִֽים.
6:8	אַ֗ךְ הַמּֽוֹצִיאָ֨ה קוֹצִ֤ים וְדַרְדָּרִים֨ נִקְלָ֣ה הִ֔יא וּקְרֹבָ֖ה לִמְאֵרָ֑ה אֲשֶׁ֥ר אַֽחֲרִיתָ֖הּ לִשְׂרֵפָֽה.
6:9	גאַ֡ךְ נֽוֹכָחִים֩ אֲנַ֨חְנוּ עֲלֵיכֶ֤ם אֲהוּבִים֨ דְּבָרִ֔ים הַטּוֹבִ֥ים מֵאֵ֖לֶּה וַֽחֲבוּרֵ֣י הַיְשׁוּעָ֑ה גַּ֛ם אִם־כָּ֖כָה אֲנַ֥חְנוּ מְדַבְּרִֽים.
6:10	כִּ֣י ׀ לֹֽא־עַוָּ֣ל הָֽאֱלֹהִ֗ים לִשְׁכֹּ֨חַ֨ אֶת־פָּֽעָלְכֶ֔ם וְאֵת֨ עֲמַ֣ל הָאַהֲבָ֔ה אֲשֶׁ֥ר הִרְאֵיתֶ֖ם עַל־שְׁמ֑וֹ כִּ֛י שֵֽׁרַתֶּ֥ם אֶת־הַקְּדוֹשִׁ֖ים וּמְשָֽׁרְתִ֥ים אַתֶּֽם.
6:11	אַ֖ךְ נִכְסָפִ֣ים אֲנָ֑חְנוּ כִּ֣י כָל־אֶחָ֣ד מִ֠כֶּם יַרְאֶ֞ה אֶת־עֶ֣צֶם ׀ הַֽחֲרִיצ֣וּת הַזֹּ֗את לִמְלֹ֛א בִּטְחַ֥ת הַתִּקְוָ֖ה עַד־הַקֵּֽץ.
6:12	לְמַ֖עַן לֹ֣א תֵֽעָצֵ֑לוּ כִּ֣י אִם־תִּהְי֤וּ מִתְהַלְּכִים֨ בְּתַבְנִ֣ית אֵ֔לֶּה אֲשֶׁ֛ר בֶּֽאֱמוּנָ֥ה וְאֹֽרֶךְ־ר֖וּחַ יֽוֹרְשִׁ֥ים אֶת־הַֽהַבְטָחֽוֹת.
6:13	כִּ֛י בְּהַבְטִ֥יחַ הָֽאֱלֹהִ֖ים לְאַבְרָהָ֑ם בַּֽאֲשֶׁ֞ר לֹֽא־הָיָ֨ה ל֤וֹ לְהִשָּׁבֵ֨עַ֨ בְּגָד֣וֹל מִמֶּ֔נּוּ נִשְׁבַּ֖ע בְּנַפְשֽׁוֹ.
6:14	לֵאמֹ֑ר כִּ֚י בָּרֵ֣ךְ אֲבָֽרֶכְךָ֔ וְהַרְבָּ֥ה אַרְבֶּ֖ה אֹתָֽךְ.
6:15	וּבְכֵ֗ן כְּהֽוֹחִילוֹ֨ בְּאֹ֣רֶךְ ר֔וּחַ הִשִּׂ֖יג אֶת־הַֽהַבְטָחָֽה.
6:16	כִּ֚י אָמְנָ֔ם בְּנֵ֥י אָדָ֛ם בַּגָּד֥וֹל מֵהֶ֖ם נִשְׁבָּעִ֑ים וְקֵ֤ץ כָּל־רִיב֨ לָהֶ֔ם הִ֥יא הַשְּׁבוּעָ֖ה לְקִיּֽוּם.
6:17	בַּֽעֲב֣וּר זֹ֔את בִּרְצ֣וֹת הָֽאֱלֹהִ֗ים לְהַרְא֤וֹת עַל־יֶ֨תֶר֨ אֶת־יֹֽרְשֵׁ֣י הַֽהַבְטָחָ֔ה אֵ֖ת בְּלִֽי־הִתמֽוֹטְט֣וּת עֲצָת֑וֹ הִתְמַצֵּ֖עַ בִּשְׁבוּעָֽה.
6:18	לְמַ֡עַן עַל־יְדֵי֩ שְׁנֵ֨י דְבָרִ֜ים בְּלִֽי־מִתְמֽוֹטְטִ֗ים אֲשֶׁר־בָּ֞ם בִּלְתִּ֨י יְכֹ֤לֶת לֵֽאלֹהִים֨ לְכַזֵּ֔ב תִּהְיֶ֛ה נֶֽחֳמָ֥ה חֲזָקָ֖ה לָ֑נוּ הַנָּסִ֛ים לְהַֽחֲזִ֥יק בַּתִּקְוָ֖ה הַמּֽוּשָׂמָ֥ה לְפָנֵֽינוּ.
6:19	אֲשֶׁ֥ר הִיא֨ יֵ֣שׁ לָ֔נוּ כְּמ֖וֹ עוֹגֵ֣ן הַנָּ֑פֶשׁ יַצִּ֣יב וְקַיָּ֔ם וּבָ֖א לִפְנִ֥ימָה לַפָּרֹֽכֶת.
6:20	אֲשֶׁר־שָׁ֗ם חָרָ֧ץ לְפָנִ֛ים בָּ֥א פְנִ֖ימָה בַּֽעֲדֵ֑נוּ יֵשׁ֕וּעַ הַנִּהְיָ֞ה כֹּהֵ֤ן גָּדוֹל֨ לְעוֹלָ֔ם עַל־דִּבְרָתִ֖י מַלְכִּי־צֶֽדֶק.
7:1	כִּ֣י זֶ֤ה מַלְכִּי־צֶ֨דֶק֨ מֶ֣לֶךְ שָׁלֵ֔ם כֹּהֵ֖ן לְאֵ֣ל עֶלְי֑וֹן אֲשֶׁ֨ר בָּ֜א לִקְרַ֣את אַבְרָהָ֗ם בְּשׁוּבוֹ֨ מֵֽהַכּ֣וֹת אֶת־הַמְּלָכִ֔ים וּבֵרַ֖ךְ אֹתֽוֹ.
7:2	אֲשֶׁר֩ חִלֵּ֨ק ל֧וֹ אַבְרָהָ֛ם גַּ֥ם מַֽעֲשֵׂ֖ר מִכֹּ֑ל בָּרִֽאשֹׁנָ֣ה ׀ אָמְנָ֗ם ה֤וּא מְתֻרְגָּם֨ מֶ֣לֶךְ צְדָקָ֔ה וְאַֽחֲרֵי־כֵ֗ן גַּ֚ם מֶ֣לֶךְ שָׁלֵ֔ם ה֖וּא מֶ֥לֶךְ שָׁלֽוֹם.
7:3	בִּבְלִ֣י אָ֡ב בִּבְלִ֣י אֵ֠ם בִּבְלִ֨י הִתְיַֽחֲס֜וּת בְּאֵין֩ הֱי֨וֹת ל֜וֹ תְּחִלַּ֣ת יָמִ֗ים גַּם־לֹא֨ אַֽחֲרִ֣ית חַיִּ֔ים כִּ֥י אִם־נִדְמֶ֖ה לְבֶן־הָֽאֱלֹהִ֑ים ה֛וּא עוֹמֵ֥ד כֹּהֵ֖ן לְעוֹלָֽם.
7:4	אַ֣ךְ הַבִּ֔יטוּ מַה־גָּד֖וֹל זֶ֑ה אֲ֠שֶׁר גַּ֤ם אַבְרָהָם֨ רֹ֣אשׁ הָֽאָב֔וֹת נָ֥תָן ל֛וֹ מַֽעֲשֵׂ֖ר מִן־הַמִּבְחָֽר.
7:5	גַּ֣ם אָמְנָ֡ם אֵ֣לֶּה מִבְּנֵ֣י לֵ֠וִי הָמְּקַבְּלִ֨ים אֶת־הַכְּהֻנָּ֜ה יֵֽשׁ־לָהֶ֣ם פְּקֻדָּ֗ה לְהַעְשִׂ֤יר אֶת־הָעָם֨ כְּפִ֣י הַתּוֹרָ֗ה זֶ֥ה ה֖וּא אֶת־אֲחֵיהֶ֑ם אַ֛ף כִּ֥י יָֽצְא֖וּ מֵֽחֶלְצֵ֥י אַבְרָהָֽם.
7:6	אַ֣ךְ זֶ֗ה אֲשֶׁ֨ר אֵינֶ֤נוּ מְיֻחָס֨ מֵהֶ֔ם הֶעְשִׂ֖יר אֶת־אַבְרָהָ֑ם וּבֵרַ֗ךְ אֵ֛ת אֲשֶׁ֥ר יֵֽשׁ־ל֖וֹ הַֽהַבְטָחֽוֹת.
7:7	וּבִבְלִ֖י כָּל־דְּבַ֣ר הִתְנַגְּד֑וּת הַמִּצְעָ֣ר מְבֹרָ֔ךְ מֵאֵ֖ת הַטּ֥וֹב מִמֶּֽנּוּ.
7:8	וְאָמְנָ֗ם בָּזֶה֨ לֹֽקְחִ֣ים מַעְשֵׂ֔ר בְּנֵ֥י אָדָ֖ם אֲשֶׁ֣ר יָמ֑וּתוּ אַ֣ךְ שָׁ֔ם הַמּוּעָ֥ד עָלָ֖יו כִּי־ה֥וּא חָֽי.
7:9	וּכְמ֖וֹ לְדַבֵּ֣ר אֹ֑מֶר בְּאַבְרָהָ֗ם גַּ֥ם לֵוִ֛י הַלֹּקֵ֥חַ מַֽעֲשֵׂ֖ר עֻשָּֽׂר.
7:10	כִּ֛י ע֥וֹד הָיָ֖ה בְּחֶלְצֵ֣י אָבִ֑יו כַּֽאֲשֶׁ֛ר בָּ֥א מַלְכִּי־צֶ֖דֶק לִקְרָאתֽוֹ.
7:11	לָכֵ֕ן אִם־אָמְנָ֗ם מִ֠כְלוֹל עַל־יְדֵ֨י הַכְּהֻנָּ֣ה הַלְוִיִּית֘ הָיָה֒ כִּ֣י הָעָ֔ם תַּחְתֶּ֖יהָ קִבֵּ֣ל אֶת־הַתּוֹרָ֑ה מָ֣ה ׀ צֹ֣רֶךְ ע֗וֹד לָק֨וּם כֹּהֵ֤ן אַחֵר֨ עַל־דִּבְרָתִי֨ מַלְכִּי־צֶ֔דֶק וְלֹ֥א לְהֵֽאָמֵ֖ר עַל־דִּבְרָתִ֥י אַֽהֲרֹֽן.
7:12	כִּ֚י בְּהֵֽעָתֵ֣ק הַכְּהֻנָּ֔ה מִצֹּ֕רֶךְ גַּ֛ם הַעְתָּקַ֥ת הַתּוֹרָ֖ה נִהְיָֽה.
7:13	כִּ֗י אֲשֶׁ֨ר נֶֽאֱמְר֤וּ עָלָיו֨ אֵ֔לֶּה יְלִ֛יד שֵׁ֥בֶט אַחֵ֖ר ה֑וּא אֲשֶׁ֣ר מִמֶּ֔נּוּ אִ֛ישׁ לֹ֥א שָׁמַ֖ר אֶת־מִשְׁמֶ֥רֶת הַמִּזְבֵּֽחַ.
7:14	כִּ֚י גָּל֣וּי ה֔וּא כִּ֥י מִֽיהוּדָ֖ה זָרַ֣ח אַדֹנֵ֑ינוּ אֲשֶׁר֨ עַל־הַשֵּׁ֣בֶט הַזֶּ֔ה מֹשֶׁ֛ה לֹֽא־דִּבֶּ֥ר מְא֖וּמָה עַל־אֹד֥וֹת כְּהֻנָּֽה.
7:15	וְע֥וֹד יוֹתֵ֖ר גָּל֣וּי ה֑וּא אִ֗ם כְּפִ֤י דְמוּת֙ מַלְכּ־צֶ֔דֶק יָק֖וּם כֹּהֵ֥ן אַחֵֽר.
7:16	אֲשֶׁ֗ר לֹ֣א כְפִ֥י תוֹרַ֛ת הַמִּצְוָ֥ה הַבְּשָׂרִ֖ית נִהְיָ֑ה כִּ֛י אִם־כְּפִ֥י גְבוּרַ֖ת חַיִּ֥ים בִּלְתִּֽי־סָפִֽים.
7:17	כִּ֖י ה֣וּא מֵעִ֑יד כִּ֗י אַתָּ֤ה כֹהֵן֨ לְעוֹלָ֔ם עַל־דִּבְרָתִ֖י מַלְכִּי־צֶֽדֶק.
7:18	כִּ֣י אָמְנָ֔ם בִּטּ֛וּל הַמִּצְוָ֥ה הַקּוֹדֶ֖מֶת נִהְיָ֑ה בַּֽעֲב֥וּר רִפְיוֹנָ֖הּ וּבְלִי־תֽוֹעַלְתָּֽהּ.
7:19	כִּ֣י הַתּוֹרָ֔ה לֹ֥א כִלְּתָ֖ה מְא֑וּמָה אַ֗ךְ הֲבָאַ֤ת תִּקְוָה֨ ט֣וֹבָה יֶ֔תֶר אֲשֶׁ֥ר בָּ֖הּ נִקְרַ֥ב אֶל־הָֽאֱלֹהִֽים.
7:20	וּבַֽאֲשֶׁ֕ר לֹ֥א הָֽיְתָ֖ה בִּבְלִ֥י שְׁבוּעָֽה.
7:21	כִּ֣י אָמְנָ֔ם הֵ֛מָּה בִּבְלִ֥י שְׁבוּעָ֖ה נִהְי֣וּ כֹֽהֲנִ֑ים אַ֥ךְ זֶה֨ בִּשְׁבוּעָ֔ה עַל־יְדֵ֞י הָֽאֹמֵ֣ר אֵלָ֗יו נִשְׁבַּ֤ע יהֹוֻּ֨הַּ וְלֹ֣א יִנָּחֵ֔ם אַתָּ֤ה כֹהֵן֨ לְעוֹלָ֔ם עַל־דִּבְרָתִ֖י מַלְכִּי־צֶֽדֶק.
7:22	לְפִ֥י כֵן֨ נִהְיָ֣ה יֵשׁ֔וּעַ עָרֵ֕ב בְּרִ֖ית ט֥וֹבָה יָֽתֶר.
7:23	וְאָמְנָ֣ם הֵ֔מָּה נִהְי֖וּ כֹּֽהֲנִ֣ים רַבִּ֑ים בַּֽעֲב֧וּר הִמָּֽנְעָ֛ם עַל־יְדֵ֥י הַמָּ֖וֶת לְהִשָּׁאֵֽר.
7:24	אַ֣ךְ זֶ֗ה בַּֽעֲבוּר֨ הִשָּֽׁאֲר֣וֹ לְעוֹלָ֔ם יֵ֥שׁ ל֛וֹ כְּהֻנָּ֖ה בִּלְתִּ֥י עֹבָֽרֶת.
7:25	לָכֵ֗ן גַּ֣ם יָכֹ֥ל הוּא֨ לְהוֹשִׁ֣יעַ עַד־כָּלִ֔יל אֶת־הַנִּגָּשִׁ֥ים אֶל־הָֽאֱלֹהִ֖ים עַל־יָדָ֑יו בַּֽחֲיוֹת֣וֹ תָמִ֔יד לְמַ֖עַן הַפְגִּ֥יעַ בַּֽעֲדָֽם.
7:26	כִּ֚י יָאָ֣ה לָ֔נוּ כֹּהֵ֥ן גָּד֖וֹל כָּזֶ֑ה קָד֞וֹשׁ בְּלִֽי־רָעָ֗ה בִּלְתִּ֤י נִכְתָּם֨ נִבְדָּ֣ל מִן־הַֽחַטָּאִ֔ים וְנִהְיָ֖ה גָּב֥וֹהַּ מִן־הַשָּׁמָֽיִם.
7:27	אֲשֶׁ֡ר אֵֽין־ל֣וֹ צֹרֶךְ֩ י֨וֹם י֜וֹם כְּמ֣וֹ ׀ הַכֹּֽהֲנִ֣ים הַגְּדוֹלִ֗ים לְהַקְרִ֨יב בָּרִֽאשֹׁנָ֤ה זְבָחִים֨ עַל־חֲטָאָ֔יו וְאַֽחֲרֵי־כֵ֖ן עַל־חַטֹּ֣אות הָעָ֑ם כִּ֣י זֹ֚את עָשָׂה֨ פַּ֣עַם אַחַ֔ת בְּהַקְרִ֖יב אֶת־נַפְשֽׁוֹ.
7:28	כִּ֣י הַתּוֹרָ֗ה שָׁ֚תָה כֹּֽהֲנִ֣ים גְּדוֹלִ֔ים אֲנָשִׁ֖ים בַּֽעֲלֵ֣י רִפְי֑וֹן אַ֣ךְ ׀ דְּבַ֣ר הַשְּׁבוּעָ֗ה אֲשֶׁ֥ר הִיא֨ אַֽחֲרֵ֣י הַתּוֹרָ֔ה אֶת־הַבֵּ֖ן הַמְּשֻׁלָּ֥ם לְעוֹלָֽם.
8:1	אַ֕ךְ רֹ֖אשׁ הַדְּבָרִ֣ים הַנֶּֽאֱמָרִ֑ים כֹּהֵ֗ן גָּד֤וֹל כָּזֶה֨ יֵ֣שׁ לָ֔נוּ אֲשֶׁ֣ר יָשַׁ֔ב לִימִ֛ין כִּסֵּ֥א הַגְּדֻלָּ֖ה בַּשָּׁמָֽיִם.
8:2	מְשָׁרֵ֥ת הַמִּקְדָּשִׁ֖ים וְהָאֹ֣הֶל הָֽאֲמִתִּ֑י אֲשֶׁ֛ר תָּקַ֥ע יְהֹוָ֖ה וְלֹ֥א אָדָֽם.
8:3	כִּ֚י כָּל־כֹּהֵ֣ן גָּד֔וֹל מוּשָׁ֗ת לְמַ֛עַן הַקְרִ֥יב מְנָח֖וֹת וּזְבָחִ֑ים לָכֵ֣ן ׀ צֹ֣רֶךְ ה֗וּא כִּ֚י גַּ֣ם זֶ֔ה יִהְיֶ֥ה ל֛וֹ מְא֖וּמָה מָ֥ה יַקְרִֽיב.
8:4	כִּ֗י אִם־אָמְנָם֨ הָיָ֣ה עַל־הָאָ֔רֶץ לֹ֥א הָיָ֖ה כֹּהֵ֑ן בִּהְי֣וֹת כֹּֽהֲנִ֔ים הַמַּקְרִיבִ֛ים כְּפִ֥י הַתּוֹרָ֖ה אֶת־הַמְּנָחֽוֹת.
8:5	אֲשֶׁ֣ר ׀ הֵ֣ם מְשָֽׁרְתִ֗ים לְצוּרַ֨ת הַדְּבָרִ֣ים הַשָּֽׁמַיְמִיִּים֘ וְצִלָּם֒ כַּֽאֲשֶׁ֞ר לֻמַּ֤ד מֹשֶׁה֨ בְּמַֽעֲנֵ֣ה אֱלֹהִ֔ים בְּחַשְּׁב֖וֹ לְכַלּ֣וֹת אֶת־הַמִּשְׁכָּ֑ן כִּ֠י רְאֵ֣ה ׀ ה֣וּא אָמַ֗ר תַּֽעֲשֶׂה֨ אֶת־הַכֹּ֔ל כַּתַּבְנִ֕ית אֲשֶׁ֛ר אַתָּ֥ה מָרְאֶ֖ה בָּהָֽר.
8:6	אַ֤ךְ עַתָּה֨ הִשִּׂ֔יג שֵׁר֖וּת נַֽעֲלֶ֣ה יָ֑תֶר כְּפִ֣י מָה֩ אֲשֶׁ֨ר ה֜וּא גַּ֤ם מְמַצֵּ֨עַ֨ בְּרִ֣ית ט֣וֹבָה יֶ֔תֶר הַמְּכ֣וֹנָנָ֔ה עַל־הַבְטָח֖וֹת טוֹב֥וֹת יָֽתֶר.
8:7	כִּ֗י אִ֚ם הָרִֽאשׁוֹנָ֣ה הַהִ֔יא הָֽיְתָ֖ה בִּבְלִ֣י שִׁמְצָ֑ה לֹ֥א בֻקַּ֖שׁ מְק֥וֹם שֵׁנִֽית.
8:8	כִּ֛י בְּגַנּ֥וֹת אֹתָ֖ם אָמָ֑ר הִנֵּ֞ה יָמִ֤ים בָּאִים֨ נְאֻ֣ם יהֹוֻּ֔הַּ וְכָֽרַתִּ֗י אֶת־בֵּ֧ית יִשְׂרָאֵ֛ל וְאֶת־בֵּ֥ית יְהוּדָ֖ה בְּרִ֥ית חֲדָשָֽׁה.
8:9	לֹ֣א כַבְּרִ֗ית אֲשֶׁ֤ר כָּרַ֨תִּי֨ אֶת־אֲבוֹתָ֔ם בְּיוֹם֨ הֶֽחֱזִיקִ֣י בְיָדָ֔ם לְהֽוֹצִיאָ֖ם מֵאֶ֣רֶץ מִצְרָ֑יִם יַ֗עַן הֵ֜מָּה לֹ֤א עָֽמְדוּ֨ בִּבְרִיתִ֔י גַּ֧ם אָֽנֹכִ֛י בָּחַ֥לְתִּי בָ֖ם נְאֻם־יְהֹוָֽה.
8:10	כִּ֣י זֹ֣את הַבְּרִ֡ית אֲשֶׁ֣ר אֶכְרוֹת֩ אֶת־בֵּ֨ית יִשְׂרָאֵ֜ל אַֽחֲרֵ֨י הַיָּמִ֤ים הָהֵם֨ נְאֻ֣ם יְהֹוָ֔ה אֶתֵּ֤ן אֶת־תּֽוֹרָתִי֨ בְּקִרְבָּ֔ם וְעַל־לִבָּ֖ם אֶכְתֳּבֶ֑נָּה וְהָיִ֤יתִי לָהֶם֨ לֵֽאלֹהִ֔ים וְהֵ֖מָּה יִהְיוּ־לִ֥י לְעָֽם.
8:11	וְלֹ֧א יְלַמְּד֣וּ ע֗וֹד אִ֣ישׁ אֶת־רֵעֵ֜הוּ וְאִ֤ישׁ אֶת־אָחִיו֨ לֵאמֹ֔ר דְּע֖וּ אֶת־יְהֹוָ֑ה כִּ֗י כֻּלָּם֨ יֵֽדְע֣וּ אֹתִ֔י לְמִקְּטַנָּ֖ם וְעַד־גְּדוֹלָֽם.
8:12	כִּ֛י אֶהְיֶ֥ה רַח֖וּם לַֽעֲוֹֽנוֹתָ֑ם וּלְחַטּאוֹתָ֥ם וּלְפִשְׁעֵיהֶ֖ם לֹ֥א אֶזְכָּר־עֽוֹד.
8:13	בְּאָמְר֣וֹ ׀ חֲדָשָׁ֔ה זִקֵּ֖ן אֶת־הָרִֽאשֹׁנָ֑ה אַ֚ךְ הַמְּזֻקָּנָ֣ה וְהַנּֽוֹשָׁנָ֔ה קְרוֹבָ֖ה לְכִלָּיֽוֹן.
9:1	אָמְנָ֣ם אֵפ֔וֹא גַּ֣ם הָרִֽאשׁוֹנָ֗ה הָי֤וּ לָהּ֨ פִּקּוּדֵ֣י שֵׁר֔וּת וּמִקְדַּ֖שׁ הָֽעוֹלָ֥ם הַזֶּֽה.
9:2	כִּ֚י אֹ֣הֶל כּוֹנַ֔ן הָֽרִאשׁ֗וֹן אֲשֶׁ֥ר בּ֛וֹ הַמְּנוֹרָ֖ה וְהַשֻּׁלְחָ֖ן וְלֶ֣חֶם הַפָּנִ֑ים אֲשֶׁ֥ר נִקְרָ֖א קֹֽדֶשׁ.
9:3	אַ֗ךְ מֵאַֽחֲרֵי֨ הַפָּרֹ֣כֶת הַשֵּׁנִ֔ית הָאֹ֛הֶל הַנִּקְרָ֖א קֹ֥דֶשׁ הַקֳּדָשִֽׁים.
9:4	אֲ֠שֶׁר הָ֤יְתָה לוֹ֨ מִקְטֶ֣רֶת הַזָּהָ֔ב וַֽאֲר֣וֹן הַבְּרִ֔ית מְצֻפֶּ֥ה זָהָ֛ב כָּלִ֖יל מִסָּבִ֑יב אֲשֶׁ֣ר בּוֹ֩ צִנְצֶ֨נֶת הַזָּהָ֜ב הַמְּכַלכֶּ֣לֶת אֶת־הַמָּ֗ן וּמַטֵּ֧ה אַֽהֲרֹ֛ן הַפֹּרֵ֖חַ וְלוּח֥וֹת הַבְּרִֽית.
9:5	וּמִמַּ֣עַל ל֔וֹ כְּרוּבֵ֥י כָב֖וֹד מְצִלִּ֣ים עַל־הַכַּפֹּ֑רֶת אֲשֶׁ֣ר עַל־אֹֽדוֹתָ֗ם אֵ֤ין עַתָּה֨ לְדַבֵּ֔ר כְּפִ֖י חֵ֥לֶק וָחֵֽלֶק.
9:6	וְכַֽאֲשֶׁ֥ר אֵ֛לֶּה כָּ֖כָה כּוֹנָ֑נוּ אָמְנָ֗ם אֶל־הָאֹ֨הֶל הָֽרִאשׁ֜וֹן בָּ֣אוּ פְנִ֤ימָה תָּמִיד֨ הַכֹּ֣הֲנִ֔ים מְכַלִּ֖ים אֶת־הָֽעֲבֹדֽוֹת.
9:7	אַ֣ךְ אֶל־הַשֵּׁנִ֗י פַּ֤עַם אַחַת֨ בַּשָּׁנָ֔ה הַכֹּהֵ֥ן הַגָּד֖וֹל לְבַדּ֑וֹ לֹ֣א ׀ בִּבְלִ֣י דָ֗ם אֲשֶׁר֨ הִקְרִ֣יב בַּֽעֲד֔וֹ וּבְעַ֖ד שְׁגִיא֥וֹת הָעָֽם.
9:8	בְּגַלּ֛וֹת ר֥וּחַ הַקֹּ֖דֶשׁ זֹ֑את כִּ֠י ע֣וֹד לֹ֤א הוֹפִ֨יעַ֨ דֶּ֣רֶךְ הַמִּקְדָּשִׁ֔ים כָּל־עוֹד֨ הָאֹ֣הֶל הָֽרִאשׁ֔וֹן הָ֥יָה ל֖וֹ מָכֽוֹן.
9:9	אֲשֶׁ֥ר הוּא֨ מָשָׁ֔ל עַל־הַזְּמָ֥ן הַנּוֹכָ֖ח הַזֶּ֑ה אֲשֶׁר־בּ֞וֹ מְנָח֤וֹת וּזְבָחִים֨ מָקְרָבִ֔ים אֲשֶׁר֨ אֵינָ֣ם יְכֹלִ֔ים לְהַשְׁלִ֛ים לְפִ֥י הָֽרַעְי֖וֹן אֵ֥ת הָֽעֹבֵֽד.
9:10	רַ֛ק בְּמַֽאֲכָלִ֥ים וּבְמַשְׁקִ֖ים וּבִטְבִיל֣וֹת שׁוֹנ֑וֹת פִּקּוּדֵ֣י הַבָּשָׂ֔ר מֽוּשָׂמִ֥ים עָלָ֖יו עַד־עֵ֥ת הַֽהַיְשָׁרָֽה.
9:11	אַ֣ךְ הַמָּשִׁ֡יחַ בַּֽאֲתוֹתוֹ֩ כֹהֵ֨ן גָּד֜וֹל אֲשֶׁ֣ר ׀ לַטּוּב֣וֹת הָֽעֲתִיד֗וֹת עַל־יְדֵ֨י הָֽאֹ֜הֶל הַגָּד֤וֹל וְהַשָּׁלֵם֨ יֶ֔תֶר בִּלְתִּ֖י עֲשׂ֣וּי יָדָ֑יִם זֶ֣ה ה֔וּא לֹ֖א מִן־הַבְּרִיאָ֥ה הַזֹּֽאת.
9:12	גַּם־לֹ֕א עַל־יְדֵ֛י דַּ֥ם עַתּוּדִ֖ים וַֽעֲגָלִ֑ים כִּ֣י אִם־עַל־יְדֵ֞י דָּמ֣וֹ שֶׁלּ֗וֹ בָּ֣א פְנִ֜ימָה פַּ֤עַם אַחַת֨ אֶל־הַמִּקְדָּ֔שׁ וּמָצָ֖א גְּאֻלַּ֥ת עוֹלָֽם.
9:13	כִּ֗י אִם־דַּ֨ם פָּרִ֤ים וְעַתּוּדִים֨ וְאֵ֣פֶר פָּרָ֔ה בִּזְרֹ֖ק עַל־הַטְּמֵאִ֑ים מְקַדֵּ֖שׁ לְטָֽהֳרַ֥ת הַבָּשָֽׂר.
9:14	כַּמָּ֣ה י֠וֹתֵר דַּ֣ם הַמָּשִׁ֜יחַ אֲשֶׁ֣ר ׀ בְּר֣וּחַ נֶ֗צַח הִקְרִ֧יב אֶת־נַפְשׁ֛וֹ בִּבְלִ֥י מ֖וּם לָֽאֱלֹהִ֑ים יְטַהֵ֤ר אֶת־רַעְיוֹנֵ֨נוּ֨ מִפְּעָלִ֣ים מֵתִ֔ים לְמַ֥עַן נַֽעֲבוֹ֖ד אֶת־אֱלֹהִ֥ים חַיִּֽים.
9:15	וּבַֽעֲב֣וּר זֹ֔את מְמַצֵּ֛עַ בְּרִ֥ית חֲדָשָׁ֖ה ה֑וּא לְמַ֡עַן כַּֽאֲשֶׁר֩ נִהְיָ֨ה הַמָּ֜וֶת לִפְד֤וּת הַפְּשָׁעִים֨ תַּ֚חַת הַבְּרִ֣ית הָרִֽאשֹׁנָ֔ה יִקְח֣וּ הַמְּקֹֽרָאִ֔ים אֶת־הַבְטָחַ֖ת יְרֻשַּׁ֥ת עוֹלָֽם.
9:16	כִּ֚י בַּֽאֲשֶׁ֣ר בְּרִ֗ית שָׁ֛ם צָרִ֥יךְ לְהוּבָ֖א מ֥וֹת הַמֵּקִֽים.
9:17	כִּ֣י בְרִ֔ית עַל־מֵתִ֖ים מְקֻיָּמָ֑ה בַּֽאֲשֶׁ֞ר לֹ֤א תֶֽחֱזַק֨ לְעוֹלָ֔ם כָּל־ע֖וֹד הַמֵּקִ֥ים חָֽי.
9:18	לָכֵ֗ן גַּם־לֹ֧א הָרִֽאשׁוֹנָ֛ה נֶחְנְכָ֖ה בִּבְלִ֥י דָֽם.
9:19	כִּ֞י כַּֽאֲשֶׁ֣ר ׀ נֶֽאֱמְרָ֣ה כָל־מִצְוָ֗ה כְּפִ֧י הַתּוֹרָ֛ה בְּפִ֥י מֹשֶׁ֖ה לְכָל־הָעָ֑ם לָקַ֡ח אֶת־דַּם֩ הָֽעֲגָלִ֨ים וְהָֽעַתּוּדִ֜ים עִם־מַ֗יִם וְצֶ֨מֶר תּוֹלַ֤עַת שָׁנִי֨ וְאֵז֔וֹב וְהִזָּ֗ה גַּ֚ם אֶת־הַסֵּ֔פֶר וְגַ֖ם אֵ֥ת כָּל־הָעָֽם.

10:21	וְכֹהֵ֣ן גָּד֔וֹל עַ֖ל בֵּ֥ית הָֽאֱלֹהִֽים.

10:24	וְנִתְבּֽוֹנְנָ֖ה אִ֣ישׁ אֶת־רֵעֵ֑הוּ לְהַקְנָאַ֥ת אַֽהֲבָ֖ה וּמַֽעֲשִׂ֥ים טוֹבִֽים.

10:26	כִּ֚י בְּחָטְאֵ֣נוּ בְרָצ֔וֹן אַֽחֲרֵ֥י קַ֖חַת אֶת־הַכָּרַ֣ת הָֽאֱמֶ֑ת אֵ֣ין ע֥וֹד נִשְׁאָ֛ר זֶ֖בַח בְּעַ֥ד חֲטָאִֽים.
10:27	אַ֗ךְ צִפִּיַּ֥ת מִשְׁפָּ֛ט נֽוֹרָאָ֖ה אֶחָ֑ת וְקִנְאַ֣ת אֵ֔שׁ הָֽעֲתִידָ֖ה לֶֽאֱכֹ֥ל אֶת־הַמִּתְקֽוֹמְמִֽים.

10:29	כַּמָּ֣ה יוֹתֵ֡ר מְדַמִּ֣ים אַ֠תֶּם יֵֽחָשֵׁ֨ב שֹׁוֶ֤ה לְעֹ֨נֶשׁ֨ רַ֣ע יֶ֔תֶר הָֽרֹמֵ֖ס אֶת־בֶּ֣ן הָֽאֱלֹהִ֑ים וְהַֽחֹשֵׁ֨ב חֹ֜ל אֶת־דַּ֤ם הַבְּרִית֨ אֲשֶׁ֣ר נִקְדַּ֣שׁ בּ֔וֹ וְהַמְּחָרֵ֖ף אֶת־ר֥וּחַ הַחָֽסֶד.
10:30	כִּ֤י יָדַ֨עְנוּ֨ אֵ֣ת הָֽאֹמֵ֔ר לִ֥י נָקָ֛ם אֲנִ֥י אֲשַׁלֵּ֖ם נְאֻ֣ם יְהֹוָ֑ה וְע֕וֹד יָדִ֥ין יְהֹוָ֖ה עַמּֽוֹ.
10:31	נוֹרָ֕א לִנְפּ֛וֹל בִּידֵ֖י אֱלֹהִ֥ים חַיִּֽים.
10:32	אַ֥ךְ זִכְר֖וּ אֶת־הַיָּמִ֣ים הָרִֽאשֹׁנִ֑ים אֲשֶׁ֤ר בָּהֶם֨ בִּהְיֽוֹתְכֶ֣ם נְאוֹרִ֔ים כִּלְכַּלְתֶּ֛ם מִלְחֶ֥מֶת עִנּוּיִ֖ים רַבָּֽה.
10:33	בָּזֶ֣ה אָמְנָ֔ם בַּֽאֲשֶׁ֛ר נִהְיֵיתֶ֥ם לְרַֽאֲוָ֖ה בַּֽחֲרָפ֣וֹת וּבִמְצוּק֑וֹת וּבָזֶ֕ה בַּֽאֲשֶׁ֣ר נִהְיֵיתֶ֔ם חַבְרֵ֖י הַמִּתְהַלְּכִ֥ים כָּֽכָה.

12:27	אַ֥ךְ זֹאת֨ ע֣וֹד אַחַ֔ת מְחַוָּ֗ה אֶת־הַעְתָּקַ֧ת הַדְּבָרִ֛ים אֲשֶׁ֥ר יִנּ֖וֹעוּ כְּמ֣וֹ עֲשׂוּיִ֑ים לְמַ֨עַן֨ יַֽעַמְד֣וּ הַדְּבָרִ֔ים אֲשֶׁ֖ר לֹ֥א יִנּֽוֹעוּ.
12:28	לָכֵ֕ן בְּקַחְתֵּ֥נוּ מַלְכ֖וּת בִּלְתִּ֣י מֽוּנָעָ֑ה נֹֽאחֲזָ֣ה חֶ֔סֶד אֲשֶׁ֨ר עַל־יָד֜וֹ נַֽעֲבֹ֤ד אֶת־הָֽאֱלֹהִים֨ עֲבוֹדָ֣ה רְצוּיָ֔ה בְּבֹ֥שֶׁת פָּנִ֖ים וּבְיִרְאָֽה.
12:29	כִּ֚י גַּ֣ם אֱלֹהִ֔ים אֵ֥שׁ אֹֽכְלָ֖ה הֽוּא.
13:1	אַֽהֲבַ֥ת הָֽאַחִ֖ים תַּֽעֲמֽוֹד.

13:3	זִכְרוּ֨ אֶת־הָֽאֲסִירִ֔ים כְּמ֖וֹ חַבְרֵ֣י מַֽאֲסָ֑ר וְאֵת֨ אֲחֻזֵ֣י הַמְּצוּקָ֔ה כְּמ֥וֹ גַם־אַתֶּ֖ם הוִֹ֥ים בַּגְּוִיָּֽה.

13:5	הַֽהִתְהַלְּכ֗וּת בִּבְלִי֨ אַ֣הֲבַת כֶּ֔סֶף מִסְתַּפְּקִ֖ים בַּדְּבָרִ֣ים הַנּֽוֹכָחִ֑ים כִּ֚י ה֣וּא אָמַ֔ר לֹ֥א אַרְפְּךָ֖ וְלֹ֥א אֶֽעֶזְבֶֽךָּ.
13:6	עַ֣ד כִּֽי־תִהְיֶ֣ה לָ֛נוּ בִּטְחָ֖ה לֵאמֹ֑ר יְהֹוָ֨ה לִ֤י עֶזְרָה֨ לֹ֣א אִירָ֔א מַה־יַּֽעֲשֶׂ֥ה לִ֖י אָדָֽם.
13:7	זִכְרוּ֨ אֶת־מְנַֽהֲלֵיכֶ֔ם אֲשֶׁ֛ר דִּבְּר֥וּ אֲלֵיכֶ֖ם אֶת־דְּבַ֣ר הָֽאֱלֹהִ֑ים אֲשֶׁ֤ר תִּתְבּֽוֹנְנוּ֨ אֶת־מוֹצָ֣א הִתְהַלְּכוּתָ֔ם וּלְכ֖וּ בְּתַבְנִ֥ית אֱמֽוּנָתָֽם.
13:8	יֵשׁ֥וּעַ הַמָּשִׁ֖יחַ תְּמ֣וֹל וְהַיּ֑וֹם זֶ֥ה ה֖וּא גַּ֥ם לְעֽוֹלָמִֽים.
13:9	אַל־תִּנָּֽשְׂאוּ֨ הֵ֣נָּה וָהֵ֔נָּה עַל־יְדֵ֛י לִמּוּדִ֥ים שֹׁנִ֖ים וְזָרִ֑ים כִּ֣י ׀ ט֣וֹב ה֗וּא אֲשֶׁ֨ר יְקֻיַּ֤ם הַלֵּב֨ בְּחֶ֔סֶד לֹ֣א בְמַֽאֲכָלִ֔ים אֲשֶׁ֛ר לֹ֥א הוֹעִ֖ילוּ לַמִּתְהַלְּכִ֥ים בָּהֶֽם.
13:10	יֵ֥שׁ לָ֖נוּ מִזְבֵּ֑חַ אֲשֶׁר֨ לֶֽאֱכֹ֣ל מִמֶּ֔נּוּ אֵ֥ין שָׁלְטָ֖ן לָעֹֽבְדִ֥ים אֶת־הַמִּשְׁכָּֽן.
13:11	כִּ֣י ׀ הַֽחַיּ֣וֹת הָהֵ֗ן אֲשֶׁ֨ר דָּמָ֜ן מוּבָ֨א אֶל־הַקֹּ֤דֶשׁ פְּנִ֨ימָה֨ עַל־חֵ֔טְא עַל־יְדֵ֖י הַכֹּהֵ֣ן הַגָּד֑וֹל גְּוִיּ֣וֹתֵיהֶ֔ן נִשְׂרָפ֖וֹת מִח֥וּץ לַמַּֽחֲנֶֽה.
13:12	לָכֵ֖ן גַּ֣ם יֵשׁ֑וּעַ לְמַ֨עַן יְקַדֵּ֤שׁ אֶת־הָעָם֨ בְּדָמ֣וֹ שֶׁלּ֔וֹ מִח֥וּץ לַשַּׁ֖עַר עֻנָּֽה.
13:13	עַל־כֵּ֕ן נֵֽצְאָ֥ה אֵלָ֖יו אֶל־מִח֣וּץ לַמַּֽחֲנֶ֑ה וְנִשְּׂאָ֖ה אֶת־חֶרְפָּתֽוֹ.
13:14	כִּ֣י אֵ֥ין לָ֛נוּ פֹּ֖ה עִ֣יר עֹמָ֑דֶת כִּ֛י אִם־אֵ֥ת הָֽעֲתִידָ֖ה אֲנַ֥חְנוּ מְבַקְּשִֽׁים.
13:15	לָ֠כֵן עַל־יָד֞וֹ נַקְרִ֨יבָה זֶ֧בַח תְּהִלָּ֛ה תָּמִ֖יד לָֽאֱלֹהִ֑ים זֶ֣ה ה֔וּא פְּרִ֥י שְׂפָתַ֖יִם מוֹד֥וֹת לִשְׁמֽוֹ.
13:16	אַ֣ךְ אֶת־עֲשִׂיַּ֧ת הַטּוֹבָ֛ה וְאֵ֥ת הַֽחִלּ֖וּק אַל־תִּשְׁכָּ֑חוּ כִּ֛י בִּזְבָחִ֥ים כָּאֵ֖לֶּה חָפֵ֥ץ הָֽאֱלֹהִֽים.
13:17	שִׁמְע֣וּ אֶל־מְנַֽהֲלֵיכֶם֘ וְהִכָּֽנְע֣וּ תַחְתָּם֒ כִּ֣י הֵ֜מָּה שֹֽׁקְדִ֤ים עַל־נַפְשֽׁוֹתֵיכֶם֨ כְּנֽוֹתְנֵ֣י חֶשְׁבּ֔וֹן לְמַ֗עַן בְּשִׂמְחָ֛ה יַֽעֲשׂ֥וּ זֹ֖את וְלֹ֣א נֶֽאֱנָחִ֑ים כִּ֣י בִלְתִּ֥י מוֹעִ֛יל זֶ֖ה לָכֶֽם.
13:18	הִתְפַּלְל֖וּ בַּֽעֲדֵ֑נוּ כִּ֣י ׀ בְּטוּחִ֣ים אֲנַ֗חְנוּ כִּ֣י רַעְי֥וֹן טוֹב֨ יֵ֣שׁ לָ֔נוּ וּבְכָל־דָּבָ֕ר חֲפֵצִ֖ים לְהִתְהַלֵּ֥ךְ בְּיֹֽשֶׁר.
13:19	אַ֣ךְ בְּיוֹתֵ֔ר אֲנִ֥י מַזְהִ֖יר לַֽעֲשׂ֣וֹת זֹ֑את לְמַ֛עַן מְהֵ֥רָה יֶ֖תֶר אוּשַׁ֥ב אֲלֵיכֶֽם.
13:20	וֵֽאלֹהֵ֣י הַשָּׁל֗וֹם הַמַּֽעֲלֶה֩ מֵעִ֨ם הַמֵּתִ֜ים אֶת־רֹעֵ֤ה הַצֹּאן֨ הַגָּד֔וֹל בְּדַ֖ם בְּרִ֣ית עוֹלָ֑ם אֵ֖ת אֲדֹנֵ֥ינוּ יֵשֽׁוּעַ.
13:21	יְתַקֵּ֨ן אֶתְכֶ֜ם בְּכָל־פֹּ֣עַל ט֗וֹב לְמַעַן֘ עֲשׂ֣וֹת אֶת־רְצוֹנוֹ֒ וְיַֽעֲשֶׂ֣ה בָכֶ֗ם אֵ֚ת הָֽרָצ֣וּי לְפָנָ֔יו עַל־יְדֵ֖י יֵשׁ֣וּעַ הַמָּשִׁ֑יחַ אֲשֶׁ֨ר ל֧וֹ הַכָּב֛וֹד לְעֽוֹלְמֵ֥י עֽוֹלָמִ֖ים אָמֵֽן.
13:22	וַֽאֲנִ֞י מַזְהִ֤יר אֶתְכֶם֨ אַחַ֔י הָכִ֖ילוּ אֶת־דְּבַ֣ר הַֽהַזְהָרָ֑ה כִּ֣י בְקֹ֔צֶר כָּתַ֥בְתִּי אִגֶּ֖רֶת אֲלֵיכֶֽם.
13:23	דְּע֕וּ כִּ֛י הָאָ֥ח טִמֽוֹתִיּ֖וֹס שֻׁלַּ֣ח חָפְשִׁ֑י אֲשֶׁ֥ר אִתּ֛וֹ אִם־יָב֥וֹא מְהֵרָ֖ה אֶרְאֶ֥ה אֶתְכֶֽם.
13:24	שַֽׁאֲל֛וּ לִשְׁל֥וֹם כָּל־מְנַֽהֲלֵיכֶ֖ם וְלִשְׁל֣וֹם כָּל־הַקְּדוֹשִׁ֑ים אֵ֛לֶּה אֲשֶׁ֥ר מֵֽאִטַּלְיָ֖א שֹֽׁאֲלִ֥ים לִשְׁלֽוֹמְכֶֽם.
13:25	הַחֶ֥סֶד עִם־כֻּלְּכֶ֖ם אָמֵֽן.
