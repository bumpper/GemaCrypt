‏Romans 
1:1	פָּא֕וּל עֶ֖בֶד יֵשׁ֣וּעַ הַמָּשִׁ֑יחַ מְקֹרָ֣א שָׁלִ֔יחַ נִבְדָּ֖ל לִבְשׂוֹרַ֥ת אֱלֹהִֽים.
1:2	אֲ֠שֶׁר הִבְטִ֨יחַ אֹתָ֥הּ מִקֶּ֛דֶם עַל־פִּ֥י נְבִיאָ֖יו בַּכְּתָבִ֥ים הַקְּדוֹשִֽׁים.
1:3	עַל־אֹד֣וֹת בְּנ֔וֹ הַנּוֹלָ֛ד מִזֶּ֥רַע דָּוִ֖ד לְפִ֥י הַבָּשָֽׂר.
1:4	והַמְּפֹרָ֨שׁ ׀ בֶּן־אֱלֹהִ֜ים בְּכֹ֗חַ כְּפִי֨ ר֣וּחַ הַקְּדֻשָּׁ֔ה עַל־יְדֵ֥י הַתְּקוּמָ֖ה מֵעִ֣ם הַמֵּתִ֑ים יֵשׁ֥וּעַ הַמָּשִׁ֖יחַ אֲדֹנֵֽינוּ.
1:5	אֲשֶׁ֙ר עַל־יָד֤וֹ קִבַּ֨לְנוּ֨ חֵ֣ן וּשְׁלִיח֔וּת לְמִשְׁמַ֥עַת הָֽאֱמוּנָ֖ה בְּכָל־הַגּוֹיִ֑ם לְמַ֖עַן שְׁמֽוֹ.
1:6	אֲשֶׁ֣ר בְּתוֹכָ֗ם יֶשְׁכֶ֧ם גַּם־אַתֶּ֛ם קְרוּאֵ֖י יֵשׁ֥וּעַ הַמָּשִֽׁיחַ.
1:7	לְכֻלְּכֶ֗ם אֲשֶׁ֤ר יֶשְׁכֶם֨ בְּרוֹמָ֔א אֲהוּבֵ֣י אֱלֹהִ֔ים הַקְּדוֹשִׁ֖ים הַמְּקֹֽרָאִ֑ים חֵ֣ן לָכֶ֞ם וְשָׁל֗וֹם מֵאֵת֨ אֱלֹהִ֣ים אָבִ֔ינוּ וְהָֽאָד֖וֹן יֵשׁ֥וּעַ הַמָּשִֽׁיחַ.
1:8	בָּרִֽאשׁוֹנָ֣ה אָמְנָ֔ם אֲנִ֣י ׀ מוֹדֶ֣ה לֵֽאלֹהַ֗י עַל־יְדֵ֛י יֵשׁ֥וּעַ הַמָּשִׁ֖יחַ עַל־כֻּלְּכֶ֑ם כִּ֛י אֱמֽוּנַתְכֶ֥ם מְסֻפָּרָ֖ה בְּכָל־הַתֵּבֵֽל.
1:9	כִּ֞י עֵדִ֣י ׀ ה֣וּא הָֽאֱלֹהִ֗ים אֲ֠שֶׁר אֲנִ֨י עֹבֵ֥ד אֹת֛וֹ בְּרוּחִ֖י בִּבְשׂוֹרַ֣ת בְּנ֑וֹ אֵ֚יךְ בִּלְתִּ֣י חֲדֹ֔ל אֲנִ֛י עֹשֶׂ֥ה זֵ֖כֶר לָכֶֽם.
1:10	בְּהִתְחַנֵּ֨ן תָּמִ֜יד בִּתְפִלּוֹתַ֗י אִם־אֵ֠פוֹא עַתָּ֨ה בָאַֽחֲרִ֥ית אֶצְלַ֛ח בִּרְצ֥וֹן הָֽאֱלֹהִ֖ים לָב֥וֹא אֲלֵיכֶֽם.
1:11	כִּ֛י נִכְסָ֥ף אֲנִ֖י לִרְאֽוֹתְכֶ֑ם לְמַ֣עַן ׀ אֲחַלֵּ֣ק לָכֶ֗ם אֵי־זֶ֛ה מַתָּ֥ן רֽוּחֲנִ֖י לְהִתְכּֽוֹנֶנְכֶֽם.
1:12	וְזֹ֖את הִ֑יא לְ֠נֻחַם יַ֜חַד בָּכֶ֗ם עַל־יְדֵ֨י הָֽאֱמוּנָ֜ה אֲשֶׁ֤ר לְאִישׁ֨ וְרֵעֵ֔הוּ גַּ֥ם לָכֶ֖ם גַּ֥ם לִֽי.
1:13	אַךְ֩ לֹ֨א אֹבֶ֜ה לִהְיֽוֹתְכֶ֣ם בִּלְתִּי־יֽוֹדְעִים֘ אַחַי֒ כִּ֠י לִפְעָמִ֞ים שַׂ֤מְתִּי בְלִבִּי֨ לָב֣וֹא אֲלֵיכֶ֔ם וְנִמְנַ֖עְתִּי עַד־עָ֑תָּה לְ֠מַעַן יִהְיֶ֨ה לִ֜י אֵי־זֶ֤ה פְרִי֨ גַּ֣ם בָּכֶ֔ם כַּֽאֲשֶׁ֥ר גַּ֖ם בִּשְׁאָ֥ר הַגּוֹיִֽם.
1:14	לִֽיוָנִ֞ים וְגַ֣ם לְלֹֽעֲזִ֗ים לַֽחֲכָמִים֨ וְגַ֣ם לְבִלְתִּ֣י נְבוֹנִ֔ים חַיָּ֖ב אָֽנִי.
1:15	וּבְכֵ֗ן כְּפִי֨ אֲשֶׁ֣ר בִּ֔י חָפֵ֥ץ אֲנִ֛י לְבַשֵּׂ֥ר גַּם־אֶתְכֶ֖ם אֲשֶׁ֥ר בְּרוֹמָֽא.
1:16	כִּ֛י אֵינֶ֥נִּי ב֖וֹשׁ מִבְּשׂוֹרַ֣ת הַמָּשִׁ֑יחַ כִּ֡י גְּבוּרַת֩ אֱלֹהִ֨ים הִ֤יא ׀ לִתְשׁוּעָה֨ לְכָל־הַמַּֽאֲמִ֔ין לִיֽהוּדִ֥י בָרִֽאשׁוֹנָ֖ה וְגַ֥ם לִֽיוָנִֽי.
1:17	כִּ֗י צִדְקַ֧ת הָֽאֱלֹהִ֛ים בָּ֥הּ נִגְלָ֖ה מֵֽאֱמוּנָ֣ה לֶֽאֱמוּנָ֑ה כַּכָּת֕וּב הַצַּדִּ֥יק בֶּֽאֱמוּנָ֖ה יִחְיֶֽה.
1:18	כִּ֣י נִגְלֶ֞ה חֲר֨וֹן אַ֤ף אֱלֹהִים֨ מִשָּׁמַ֔יִם עַ֥ל כָּל־רֶ֖שַׁע וְעַוְלַ֣ת הָֽאֲנָשִׁ֑ים הָאֹֽחֲזִ֥ים אֶת־הָֽאֱמֶ֖ת בְּעַוְלָֽה.
1:19	יַ֗עַן אֵ֣ת אֲשֶׁ֤ר נוֹדָע֨ עַל־הָֽאֱלֹהִ֔ים גָּל֥וּי ה֖וּא בָּהֶ֑ם כִּ֥י הָֽאֱלֹהִ֖ים גִּלָּ֥ה לָהֶֽם.
1:20	כִּ֣י הַדְּבָרִ֣ים אֹֽדוֹתָ֡יו אֲשֶׁ֣ר בִּלְתִּֽי־נִרְאִים֩ מִבְּרִיאַ֨ת הָֽעוֹלָ֜ם נִרְאִ֣ים בָּר֗וּר בִּהְיוֹתָ֨ם מֽוּבָנִ֜ים עַל־יְדֵ֤י הַדְּבָרִים֨ הַנַּֽעֲשִׂ֔ים וְהֵ֛ם גְּבֽוּרָת֥וֹ וֵאלֹֽהוּת֖וֹ הַנִּצְחִ֑ית עַ֛ד כִּי־אֵ֥ין לָהֶ֖ם הִתְנַצְּלֽוּת.
1:21	יַ֚עַן בְּדַעְתָּ֣ם אֶת־הָֽאֱלֹהִ֔ים כֵּֽאלֹהִים֨ לֹ֣א כִבְּד֔וּהוּ גַּ֛ם לֹ֥א הוֹד֖וּ ל֑וֹ כִּ֚י אִם־הָֽבְל֣וּ בְמַחְשְׁבוֹתָ֔ם וְחָשַׁ֖ךְ לְבָבָ֥ם הָֽאֱוִילִֽי.
1:22	בְּהִתְאַמְּרָ֛ם לִהְי֥וֹת חֲכָמִ֖ים נִסְכָּֽלוּ.
1:23	וְהֵמִ֗ירוּ אֶת־כְּבוֹד֨ הָֽאֱלֹהִ֣ים בִּלְתִּֽי־נִשְׁחָ֔ת בִּדְמ֥וּת צֶ֖לֶם אָדָ֣ם נִשְׁחָ֑ת וְצֶ֥לֶם ע֛וֹף וּבַֽעֲלֵ֥י אַרְבַּע־רַגְלַ֖יִם וְרָֽמֶשׂ.
1:24	עַל־כֵּ֞ן גַּ֣ם הָֽאֱלֹהִ֗ים הִסְגִּירָ֛ם אֶל־תַּֽאֲוֹ֥ת לִבּוֹתָ֖ם לְטָמְאָ֑ה לְמַ֧עַן נַבֵּ֛ל אֶת־גְּוִיּֽוֹתֵיהֶ֖ם בֵּֽינֵיהֶֽם.
1:25	אֲשֶׁ֧ר הֶֽחֱלִ֛יפוּ אֶת־אֱמֶ֥ת הָֽאֱלֹהִ֖ים בְּכָזָ֑ב וְיָֽרְא֨וּ וְעָֽבְד֤וּ אֶת־הַבְּרִיאָה֨ עַ֣ל הַבּוֹרֵ֔א אֲשֶׁ֥ר ה֛וּא מְבֹרָ֥ךְ לְעֽוֹלָמִ֖ים אָמֵֽן.
1:26	בִּגְלַ֣ל זֹ֔את הִסְגִּירָ֥ם הָֽאֱלֹהִ֖ים אֶל־תְּשׁוּק֣וֹת קָל֑וֹן כִּ֣י ׀ גַּ֣ם נְקֵבֽוֹתֵיהֶ֗ם הֶֽחֱלִ֛יפוּ אֶת־הַתַּשְׁמִ֥ישׁ הַטִּבְעִ֖י בְּנֶ֥גֶד הַטָּֽבַע.
1:27	וּכְמוֹ־כֵן֘ גַּ֣ם הַזְּכָרִים֒ בְּעָזְבָ֞ם אֶת־תַּשְׁמִ֤ישׁ הָֽאִשָּׁה֨ הַטִּבְעִ֔י בָּֽעֲר֥וּ בְעַגְבָתָ֖ם אִ֣ישׁ אֶל־רֵעֵ֑הוּ זְכָרִ֤ים עִם־זְכָרִים֨ פֹּֽעֲלִ֣ים בֹּ֔שֶׁת וְאֵ֨ת גְּמ֤וּל תְּעוּתָם֨ אֲשֶׁ֣ר הָיָ֣ה רָא֔וּי הֵ֥ם לֹֽקְחִ֖ים בְּעַצְמָֽם.
1:28	וְכַֽאֲשֶׁר֩ לֹ֨א בָֽחֲר֜וּ לִנְצֹ֤ר אֶת־הָֽאֱלֹהִים֨ בְּדַעְתָּ֔ם כֵּ֣ן הִסְגִּ֥יר אֹתָ֛ם הָֽאֱלֹהִ֖ים אֶל־לֵ֣ב נִתְעָ֑ב לַֽעֲשׂוֹת֨ אֶת־הַדְּבָרִ֔ים אֲשֶׁ֖ר אֵינָ֥ם נָאוִֽים.
1:29	בְּהִמָּֽלְאָ֣ם כָּל־עַוְלָ֗ה זְנוּת֨ רִשְׁעָ֔ה חֶמְדַּ֥ת בֶּ֖צַע אָ֑וֶן מְלֵאִ֗ים קִנְאָ֥ה ׀ רֶ֛צַח מְרִיבָ֥ה ׀ מִרְמָ֖ה מְזִמָּֽה.
1:30	מְלַֽחֲשִׁים֨ מַלְשִׁינִ֔ים שֽׂנְאֵ֥י אֱלֹהִ֖ים מְחָֽרְפִ֑ים זֵדִ֣ים ׀ יֽהִירִ֔ים חֹֽרְשֵׁ֥י רָע֖וֹת מֹרִ֥ים בַּיּֽוֹלְדִֽים.
1:31	לֹֽא־נְבוֹנִ֣ים ׀ בּֽוֹגְדִ֔ים חַסְרֵ֖י אַֽהֲבָ֣ה טִבְעִ֑ית בִּלְתִּֽי־מִתְרַצִּ֖ים בִּלְתִּֽי־מְרַֽחֲמִֽים.
1:32	אֲשֶׁ֨ר הֵ֜ם בְּדַעְתָּ֣ם ׀ אֶת־מִשְׁפַּ֣ט הָֽאֱלֹהִ֗ים כִּ֛י הַפֹּֽעֲלִ֥ים כָּאֵ֖לֶּה שֹׁוִ֣ים לַמָּ֑וֶת לֹ֤א לְבַד֨ עֹשִׂ֣ים אֶת־אֵ֔לֶּה כִּ֛י אִם־גַּ֥ם נְאוֹתִ֖ים אֶל־הָֽעֹשִֽׂים.
2:1	עַל־כֵּ֗ן אֵֽינְךָ֤ מְנֻקֶּה֨ הָֽאָדָ֔ם כָּל־אִ֖ישׁ הַשּׁוֹפֵ֑ט כִּ֡י בְּזֹאת֩ אֲשֶׁ֨ר אַתָּ֜ה שֹׁפֵ֣ט אֶת־הָֽאַחֵ֗ר אֶת־עַצְמְךָ֨ אַתָּ֣ה מַרְשִׁ֔יעַ כִּ֗י אֶת־עֶ֨צֶם֨ הַדְּבָרִ֣ים הָאֵ֔לֶּה אַתָּ֥ה פֹעֵ֖ל הַשּׁוֹפֵֽט.
2:2	וְיָדַ֗עְנוּ כִּי֩ מִשְׁפַּ֨ט הָֽאֱלֹהִ֥ים הוּא֨ כְּפִ֣י הָֽאֱמֶ֔ת עַ֖ל הַפֹּֽעֲלִ֣ים כָּאֵֽלֶּה.
2:3	אַ֞ךְ הַֽחֹשֵׁ֣ב ׀ אַתָּ֣ה הָֽאָדָ֗ם הַשּׁוֹפֵט֨ אֶת־הַפֹּֽעֲלִ֣ים כָּאֵ֔לֶּה וְעֹשֶׂ֖ה אֶת־עַצְמָ֑ם כִּ֛י אַתָּ֥ה תָנ֖וּס מִמִּשְׁפַּ֥ט הָֽאֱלֹהִֽים.
2:4	א֠וֹ הֲבֹזֶ֨ה אַתָּ֜ה אֶת־עֹ֧שֶׁר טוּב֛וֹ וְאֹ֥רֶךְ אַפּ֖וֹ וְאֹ֣רֶךְ רוּח֑וֹ מִבְּלִ֣י דַ֗עַת כִּ֛י ט֥וּב הָֽאֱלֹהִ֖ים מוֹלִֽיכְךָ֥ לִתְשׁוּבָֽה.
2:5	אַ֡ךְ כְּפִ֣י קָ֠שֽׁוּתְךָ וּלְבָ֨בְךָ֜ אֲשֶׁ֣ר ׀ אֵינֶ֣נּוּ שָׁ֗ב אַתָּ֨ה צֹבֵ֤ר לְעַצְמְךָ֨ חָר֔וֹן לְי֨וֹם חָר֧וֹן ׀ וְהִתְגַּלּ֛וּת מִשְׁפַּ֥ט צֶ֖דֶק הָֽאֱלֹהִֽים.
2:6	אֲשֶׁ֛ר יְשַׁלֵּ֥ם לְכָל־אִ֖ישׁ כְּמַֽעֲשָֽׂיו.
2:7	אָמְנָ֣ם ׀ לָאֵ֗לֶּה הַמְּבַקְשִׁ֞ים בְּתוֹחֶ֨לֶת פְּעֻלָּ֥ה טּוֹבָ֛ה כָּב֥וֹד וִיקָ֖ר וְאַל־שָׁ֑חַת חַיֵּ֖י עֽוֹלָמִֽים.
2:8	אַ֣ךְ לָאֵלֶּה֩ אֲשֶׁ֨ר הֵ֜ם מִמְּרִיבָ֗ה וּבִלְתִּ֤י נִפְתִּים֨ אֶל־הָֽאֱמֶ֔ת כִּ֥י אִם־נִפְתִּ֖ים אֶל־הָֽעַוְלָ֑ה רֹ֖גֶז וְחֵמָֽה.
2:9	צָרָ֣ה וְצוּקָ֔ה עַ֛ל כָּל־נֶ֥פֶשׁ אָדָ֖ם הַפֹּעֵ֣ל אֶת־הָרָ֑ע עַל־נֶ֤פֶשׁ יְהוּדִי֨ בָּרִֽאשֹׁנָ֔ה גַּ֖ם נֶ֥פֶשׁ יְוָנִֽי.
2:10	אַ֗ךְ כָּב֥וֹד וִיקָ֛ר וְשָׁל֖וֹם לְכָל־הַפֹּעֵ֣ל אֶת־הַטּ֑וֹב גַּ֧ם יְהוּדִ֛י בָּרִֽאשֹׁנָ֖ה גַּ֥ם יְוָנִֽי.
2:11	כִּ֗י אֵ֛ין מַשּׂ֥וֹא פָנִ֖ים עִם־הָֽאֱלֹהִֽים.
2:12	כִּ֗י כֹּ֣ל אֲשֶׁ֤ר בְּאֵֽין־תּוֹרָה֨ חָֽטְא֔וּ גַּ֛ם בְּאֵ֥ין תּוֹרָ֖ה יֹאבֵ֑דוּ וְכֹ֗ל אֲשֶׁ֤ר בְּתוֹרָה֙ חָֽטְא֔וּ עַל־יְדֵ֥י תוֹרָ֖ה יִשָּׁפֵֽטוּ.
2:13	כִּ֗י לֹ֚א שֹֽׁמְעֵ֣י הַתּוֹרָ֔ה צַדִּיקִ֖ים לִפְנֵ֣י הָֽאֱלֹהִ֑ים כִּ֛י אִם־עֹשֵׂ֥י הַתּוֹרָ֖ה יִצָּדֵֽקוּ.
2:14	כִּ֗י כַּֽאֲשֶׁ֨ר גּוֹיִ֜ם בִּבְלִ֨י הֱי֤וֹת לָהֶם֨ תּוֹרָ֔ה מִטֶּ֕בַע עֹשִׂ֖ים אֶת־דִּבְרֵ֣י הַתּוֹרָ֑ה אֵ֗לֶּה בִּבְלִ֨י הֱי֤וֹת לָהֶם֨ תּוֹרָ֔ה לְעַצְמָ֖ם הֵ֥ם תּוֹרָֽה.
2:15	אֲשֶׁר֩ הֵ֨מָּה מַרְאִ֜ים אֶת־פֹּ֤עַל הַתּוֹרָה֨ כָּת֣וּב בְּלִבּוֹתָ֔ם בְּהָעִ֖יד גַּ֣ם רַעְיוֹנָ֑ם בַּֽאֲשֶׁ֣ר גַּ֣ם מַחְשְׁבֽוֹתֵיהֶ֗ם מַשְׂטִינ֛וֹת א֥וֹ מְנַקּ֖וֹת אֵ֥לֶּה אֶת־אֵֽלֶּה.
2:16	בַּיּ֗וֹם אֲשֶׁ֨ר יִשְׁפֹּ֤ט הָֽאֱלֹהִים֨ אֶת־תַּֽעֲלֻמּוֹת֨ בְּנֵי֣ הָֽאָדָ֔ם לְפִ֖י בְּשֽׂוֹרָתִ֑י עַל־יְדֵ֖י יֵשׁ֥וּעַ הַמָּשִֽׁיחַ.
2:17	הִנֵּ֗ה אַתָּה֨ יְהוּדִ֣י מְכֻנֶּ֔ה וְשֹׁקֵ֖ט עַל־הַתּוֹרָ֑ה וְאַתָּ֥ה מִתְהַלֵּ֖ל בֵּֽאלֹהִֽים.
2:18	וְאַתָּ֤ה יוֹדֵ֨עַ֨ אֶת־רְצוֹנ֔וֹ וּבֹחֵ֖ן אֶת־הַמְּעֻלִּ֑ים בַּֽאֲשֶׁ֛ר אַתָּ֥ה מְלֻמָּ֖ד מִן־הַתּוֹרָֽה.
2:19	וּבֹטֵ֣חַ אַתָּ֔ה לִהְיֽוֹתְךָ֥ עַצְמְךָ֖ נֹהֵ֣ג עִוְרִ֑ים א֕וֹר לָאֵ֖לֶּה אֲשֶׁ֥ר בַּחֹֽשֶׁךְ.
2:20	מְיַסֵּ֣ר אֱוִילִ֔ים מְלַמֵּ֖ד עֹֽלְלִ֑ים אֲשֶׁ֣ר יֵּֽשׁ־לְךָ֗ תַּבְנִ֥ית הַדַּ֛עַת וְהָֽאֱמֶ֖ת בַּתּוֹרָֽה.
2:21	לָכֵ֗ן הַמְּלַמֵּד֨ אֶת־אַחֵ֔ר הַאִ֥ם אֶת־עַצְמְךָ֖ לֹ֣א תְלַמֵּ֑ד הִמַּגִּיד֨ לְבִלְתִּ֣י גְנֹ֔ב הַאִ֥ם אַתָּ֖ה גֹּנֵֽב.
2:22	הָֽאֹמֵר֨ לְבִלְתִּ֣י נְאֹ֔ף הַֽאִם־אַתָּ֖ה נֹאֵ֑ף הַמְּתָעֵ֣ב אֶת־הָֽאֱלִילִ֔ים הַֽאִם־אַתָּ֖ה בֹּזֵ֥ז אֶת־הַקֳּדָשִֽׁים.
2:23	אֲשֶׁ֥ר אַתָּ֖ה מִתְהַלֵּ֣ל בַּתּוֹרָ֑ה הַאִ֗ם עַל־יְדֵי֨ עֲבֵרַ֣ת הַתּוֹרָ֔ה אֶת־הָֽאֱלֹהִ֖ים אַתָּ֥ה מְנַבֵּֽל.
2:24	כִּ֚י שֵׁ֣ם הָֽאֱלֹהִ֔ים בַּֽעֲבֽוּרְכֶ֖ם מְגֻדָּ֣ף בַּגּוֹיִ֑ם כַּֽאֲשֶׁ֖ר נִכְתָּֽב.
2:25	כִּ֣י הַמִּילָ֗ה אָמְנָם֨ מוֹעֶ֣לֶת הִ֔יא אִם־אַתָּ֥ה עֹשֶׂ֖ה אֶת־הַתּוֹרָ֑ה אַ֗ךְ אִ֤ם תִּהְיֶה֙ עֹבֵ֣ר הַתּוֹרָ֔ה מִילָֽתְךָ֖ לְעָרְלָ֥ה נִהְיָֽתָה.
2:26	לָכֵ֞ן אִ֤ם בֶּן־הָֽעָרְלָה֨ שֹׁמֵר֨ אֶת־צִדְק֣וֹת הַתּוֹרָ֔ה הַֽאִם־לֹ֥א עָרְלָת֖וֹ לְמוּלָ֥ה תֵֽחָשֵֽׁב.
2:27	וְיִשְׁפֹּ֞ט בֶּן־הָֽעָרְלָ֧ה מִן־הַטֶּ֛בַע הַגֹּמֵ֥ר אֶת־הַתּוֹרָ֖ה אֹתָ֑ךְ אֲשֶׁ֗ר עַל־יְדֵי֨ הָא֣וֹת וְהַמִּילָ֔ה עֹבֵ֥ר הַתּוֹרָ֖ה אָֽתָּה.
2:28	כִּ֣י לֹ֥א זֶ֛ה אֲשֶׁ֥ר בְּגָל֖וּי יְה֣וּדִי ה֑וּא גַּם־לֹ֣א זֹ֗את אֲשֶׁ֧ר בְּגָל֛וּי בַּבָּשָׂ֖ר מִילָ֥ה הִֽיא.
2:29	כִּ֣י אִם־זֶ֞ה אֲשֶׁ֣ר בַּסֵּתֶר֘ יְהוּדִ֣י הוּא֒ וּמִילַ֥ת הַלֵּ֛ב בָּר֖וּחַ לֹ֣א בָא֑וֹת אֲשֶׁ֣ר ׀ תְּהִלָּת֗וֹ אֵינֶ֨נָּה֨ מֵאֵ֣ת אֲנָשִׁ֔ים כִּ֖י אִם־מֵאֵ֥ת הָֽאֱלֹהִֽים.
3:1	מָ֥ה אֵפ֖וֹא יִתְר֣וֹן הַיְּהוּדִ֑י א֕וֹ מָ֖ה תּוֹעֶ֥לֶת הַמִּילָֽה.
3:2	רַ֖ב בְּכָל־דָּ֑רֶךְ בָּרִֽאשׁוֹנָ֣ה אָמנָ֔ם יַ֗עַן כִּ֛י אִתָּ֥ם הָפְקְד֖וּ אִמְר֥וֹת הָֽאֱלֹהִֽים.
3:3	כִּ֣י מָ֔ה אִ֥ם אֲחָדִ֖ים לֹ֣א הֶֽאֱמִ֑ינוּ הַאִ֛ם תְּבַטֵּ֥ל לֹֽא־אֱמֽוּנָתָ֖ם אֶת־אֱמוּנַ֥ת הָֽאֱלֹהִֽים.
3:4	חָלִ֕ילָה אַ֗ךְ יְהִ֤י הָֽאֱלֹהִים֨ אֱמֶ֔ת וְכָל־אָדָ֖ם כֹּזֵ֑ב כַּכָּת֗וּב לְמַ֨עַן֨ תִּצְדַּ֣ק בִּדְבָרֶ֔יךָ וְתִזְכֶּ֖ה בְּהִשָּֽׁפְטֶֽךָ.
3:5	אַ֗ךְ אִם־עַוְלָתֵ֤נוּ תְשַׁבֵּ֨חַ֨ אֶת־צִדְקַ֣ת הָֽאֱלֹהִ֔ים מַה־נֹּאמַ֕ר הַאִ֧ם מְעַוֵּ֛ל הָֽאֱלֹהִ֖ים הַמֵּבִ֣יא חָר֑וֹן כְּפִ֥י אָדָ֖ם אָֽנֹכִ֥י מְדַבֵּֽר.
3:6	חָלִ֑ילָה אִם־לֹ֕א אֵ֛יךְ יִשְׁפֹּ֥ט הָֽאֱלֹהִ֖ים אֶת־הַתֵּבֵֽל.
3:7	כִּ֚י אִם־אֱמֶ֣ת הָֽאֱלֹהִ֔ים בִּכְזָבִ֖י הֶעְדִּ֣יפָה לִכְבוֹד֑וֹ לָ֣מָּה ׀ ע֗וֹד גַּ֧ם אָֽנֹכִ֛י כְּחֹטֵ֖א אֲנִ֥י נִשְׁפָּֽט.
3:8	וְלֹ֕א כַּֽאֲשֶׁ֣ר ׀ אֲנַ֣חְנוּ מְגֻדָּפִ֗ים וְכַֽאֲשֶׁ֨ר אֲחָדִ֤ים מְמַלְלִים֨ כִּ֣י אֲנַ֣חְנוּ אֹֽמְרִ֔ים נַֽעֲשֶׂ֣ה אֶת־הָרָ֔ע לְמַ֥עַן יָב֖וֹא הַטּ֑וֹב אֲשֶׁ֥ר דִּינָ֖ם צֶ֥דֶק הֽוּא.
3:9	מָ֣ה אֵפ֗וֹא הֲיֵ֥שׁ לָ֨נוּ֨ יִתְר֔וֹן לֹ֖א כָּל־מְא֑וּמָה כִּ֣י ׀ מִקֹּ֣דֶם הֶֽאֱשַׁ֗מְנוּ גַּ֤ם יְהוּדִים֨ וְגַ֣ם יְוָנִ֔ים לִהְיוֹתָ֥ם כֻּלָּ֖ם תַּ֥חַת חֵֽטְא.
3:10	כַּכָּת֕וּב כִּ֚י אֵ֣ין צַדִּ֔יק אֵ֖ין גַּ֥ם אֶחָֽד.
3:11	אֵ֖ין מַשְׂכִּ֑יל אֵ֖ין דֹּרֵ֥שׁ אֶת־אֱלֹהִֽים.
3:12	הַכֹּ֣ל סָ֔רוּ יַחְדָּ֖ו נֶֽאֱלָ֑חוּ אֵ֚ין עֹשֵׂ֣ה ט֔וֹב אֵ֖ין גַּם־אֶחָֽד.
3:13	קֶ֤בֶר פָּת֨וּחַ֨ גְּרֹנָ֔ם בִּלְשֹֽׁנוֹתָ֖ם נִכֵּ֑לוּ חֲמַ֥ת עַכְשׁ֖וּב תַּ֥חַת שְׂפָתֵֽימוֹ.
3:14	אֲשֶׁ֥ר פִּיהֶ֛ם מָלֵ֥א אָלָ֖ה וּמְרִירֽוּת.
3:15	רַגְלֵיהֶ֕ם מְמַֽהֲר֖וֹת לִשְׁפֹּ֥ךְ דָּֽם.
3:16	שֶׁ֥בֶר וָשֹׁ֖ד בִּמְסִלּוֹתָֽם.
3:17	וְדֶ֥רֶךְ שָׁל֖וֹם לֹ֥א יָדָֽעוּ.
3:18	אֵ֛ין פַּ֥חַד אֱלֹהִ֖ים לְנֶ֥גֶד עֵֽינֵיהֶֽם.
3:19	אַ֣ךְ יָדַ֗עְנוּ כִּ֠י כָּל־אֲשֶׁ֤ר הַתּוֹרָה֨ אֹמֶ֔רֶת אֶ֣ל אֲשֶׁ֥ר הֵ֛ם בַּתּוֹרָ֖ה הִ֣יא אֹמָ֑רֶת לְמַ֨עַן֨ יִסָּכֵ֣ר כָּל־פֶּ֔ה וְתִהְיֶ֥ה כָל־הַתֵּבֵ֖ל אֲשֵׁמָ֥ה לָֽאֱלֹהִֽים.
3:20	לָכֵ֕ן מִפָּֽעֳלֵ֣י הַתּוֹרָ֔ה לֹֽא־יִצְדַּ֥ק כָּל־בָּשָׂ֖ר לְפָנָ֑יו כִּ֛י עַל־יְדֵ֥י הַתּוֹרָ֖ה דַּ֥עַת הַחֵֽטְא.
3:21	אַ֣ךְ עַתָּ֔ה בִּבְלִ֣י תוֹרָ֔ה צִדְקַ֥ת הָֽאֱלֹהִ֖ים נִגְלָ֑תָה בִּהְיוֹתָ֣הּ מֽוּעָדָ֔ה מֵאֵ֥ת הַתּוֹרָ֖ה וְהַנְּבִיאִֽים.
3:22	וְצִדְקַ֣ת הָֽאֱלֹהִ֗ים הִ֞יא עַל־יְדֵ֤י אֱמוּנַת֨ יֵשׁ֣וּעַ הַמָּשִׁ֔יחַ אֶל־הַכֹּ֥ל וְעַל־הַכֹּ֖ל הַמַּֽאֲמִינִ֑ים כִּ֖י אֵ֥ין הַבְדֵּֽל.
3:23	כִּ֖י הַכֹּ֣ל חָטָ֑אוּ וְהֵ֥ם חֲסֵרֵ֖י כְּב֥וֹד הָֽאֱלֹהִֽים.
3:24	בְּהִצָּֽדְקָ֥ם חִנָּ֖ם בְּחַסְדּ֑וֹ עַל־יְדֵ֣י הַפְּד֔וּת אֲשֶׁ֥ר הִ֖יא בְּיֵשׁ֥וּעַ הַמָּשִֽׁיחַ.
3:25	אֲשֶׁ֣ר הָֽאֱלֹהִ֡ים שָׁת֩ אֹת֨וֹ מִקֶּ֧דֶם לְכַפֹּ֛רֶת עַל־יְדֵ֥י הָֽאֱמוּנָ֖ה בְּדָמ֑וֹ לְגִלּ֣וּי צִדְקָת֗וֹ בַּֽעֲב֞וּר פָּסְח֤וֹ עַל־הַֽחֲטָאִים֨ הַנִּהְיִ֣ים מִקֶּ֔דֶם בְּאֹ֥רֶךְ אַ֖ף הָֽאֱלֹהִֽים.
3:26	לְגִּלּ֥וּי צִדְקָת֖וֹ בָּעֵ֣ת הַזֹּ֑את לְמַ֨עַן֨ הֱיוֹת֣וֹ צַדִּ֔יק וּמַצְדִּ֛יק אֵ֥ת אֲשֶׁ֖ר מֵֽאֱמוּנַ֥ת יֵשֽׁוּעַ.
3:27	אַיֵּ֤ה אֵפוֹא֨ הַֽהִתְהַלְל֔וּת הִ֖יא נִגְזְרָ֣ה לָ֑הּ עַל־יְדֵ֣י ׀ אֵי־זוּ֣ תוֹרָ֗ה הַאִם֨ אֲשֶׁ֣ר לַפְּעָלִ֔ים לֹ֕א כִּ֥י אִם־עַל־יְדֵ֖י תּוֹרַ֥ת הָֽאֱמוּנָֽה.
3:28	לָכֵ֖ן אֲנַ֣חְנוּ מְחַשְּׁבִ֑ים כִּ֗י בָּֽאֱמוּנָה֨ נִצְדָּ֣ק אִ֔ישׁ בִּבְלִ֖י פָּֽעֳלֵ֥י הַתּוֹרָֽה.
3:29	הַאִם֩ אֱלֹהֵ֨י הַיְּהוּדִ֤ים לְבַד֨ הָֽאֱלֹהִ֔ים הַֽאִם־לֹ֖א גַּ֣ם שֶׁלַּגּוֹיִ֑ם כֵּ֖ן גַּ֥ם שֶׁלַּגּוֹיִֽם.
3:30	כִּ֣י אָמְנָ֔ם אֶחָ֖ד הָֽאֱלֹהִ֑ים אֲשֶׁ֨ר יַצְדִּ֜יק אֶת־בְּנֵ֤י הַמִּילָה֨ מֵֽאֱמוּנָ֔ה וְאֵ֛ת בְּנֵ֥י הָֽעָרְלָ֖ה עַל־יְדֵ֥י הָֽאֱמוּנָֽה.
3:31	הַֽאִם־אֵפ֗וֹא מְבַטְּלִ֥ים אֲנַ֛חְנוּ אֶת־הַתּוֹרָ֖ה עַל־יְדֵ֣י הָֽאֱמוּנָ֑ה חָלִ֕ילָה כִּ֛י אִם־מְקִימִ֥ים אֲנַ֖חְנוּ אֶת־הַתּוֹרָֽה.
4:1	מָ֣ה אֵפ֗וֹא אֹֽמְרִ֤ים אֲנַ֨חְנוּ֨ עַל־אַבְרָהָ֣ם אָבִ֔ינוּ כִּ֥י מָצָ֖א כְּפִ֥י הַבָּשָֽׂר.
4:2	כִּ֗י אִם־אַבְרָהָם֨ מִפְּעָלִ֣ים נִצְדַּ֔ק יֵ֥שׁ ל֖וֹ תְּהִלָּ֑ה אַ֕ךְ לֹ֖א לִפְנֵ֥י הָֽאֱלֹהִֽים.
4:3	כִּ֛י מָ֥ה אֹמֵ֖ר הַכְּתָ֑ב הֶֽאֱמִ֤ין אַבְרָהָם֨ בֵּֽאלֹהִ֔ים וַתֵּחָ֥שֶׁב ל֖וֹ לִצְדָקָֽה.
4:4	אַ֚ךְ אֶל־הַפֹּעֵ֔ל הַשָּׂכָ֗ר אֵינֶ֤נּוּ נֶחְשָׁב֨ כְּפִ֣י חֶ֔סֶד כִּ֖י אִם־כְּפִ֥י חֽוֹב.
4:5	אַ֞ךְ לַֽאֲשֶׁ֣ר ׀ אֵינֶ֣נּוּ פֹעֵ֗ל רַ֤ק מַֽאֲמִין֨ בַּמַּצְדִּ֣יק אֶת־הָֽרָשָׁ֔ע נֶחְשְׁבָ֥ה אֱמֽוּנָת֖וֹ לִצְדָקָֽה.
4:6	כַּֽאֲשֶׁ֣ר גַּם־דָּוִ֗ד אֹמֵר֨ אֶת־אֹ֣שֶׁר הָֽאָדָ֔ם אֲשֶׁ֨ר הָֽאֱלֹהִ֜ים חְֹשֵׁ֥ב ל֛וֹ צְדָקָ֖ה בִּבְלִ֥י פְעָלִֽים.
4:7	אַשְׁרֵי֨ נְשׂוּאֵ֣י פְשָׁעִ֔ים וּכְסוּיֵ֖י חֲטָאֽוֹת.
4:8	אַשְׁרֵ֣י אָדָ֗ם לֹֽא־יַחְשֹׁ֧ב יְהוָֹ֛ה ל֖וֹ עָוֹֽן.
4:9	וְהָאֹ֣שֶׁר הַזֶּ֗ה הַאִ֤ם אֵפוֹא֨ עַל־הַמִּילָ֔ה א֖וֹ גַּ֣ם עַל־הָֽעָרְלָ֑ה כִּ֣י אָמַ֔רְנוּ כִּ֣י נֶחְשְׁבָ֧ה לְאַבְרָהָ֛ם הָֽאֱמוּנָ֖ה לִצְדָקָֽה.
4:10	אֵ֤יךְ אֵפוֹא֨ נֶחְשְׁבָ֔ה אִ֚ם בִּהְיוֹת֣וֹ בַמִּילָ֔ה א֖וֹ בָּֽעָרְלָ֑ה לֹ֚א בַּמִּילָ֔ה כִּ֖י אִם־בָּֽעָרְלָֽה.
4:11	וְאֵת֩ א֨וֹת הַמִּילָ֜ה לָקַ֗ח חוֹתַם֨ צִדְקַ֣ת הָֽאֱמוּנָ֔ה אֲשֶׁ֥ר הִ֖יא בָּֽעָרְלָ֑ה לְמַ֨עַן הֱיוֹת֜וֹ אֲבִ֤י כָל־הַמַּֽאֲמִינִים֨ בָּֽעָרְלָ֔ה לְמַ֧עַן תֵּֽחָשֵׁ֛ב גַּ֥ם לָהֶ֖ם הַצְּדָקָֽה.
4:12	וַֽאֲבִ֣י הַמִּילָ֗ה לֹ֤א לְבַד֨ לָאֵ֔לֶּה אֲשֶׁ֥ר הֵ֖ם מִן־הַמִּילָ֑ה כִּ֣י אִם־גַּ֠ם אֶל־הַמִּתְהַלְּכִ֞ים בְּעִקְּב֣וֹת הָֽאֱמוּנָ֗ה אֲשֶׁ֛ר לְאַבְרָהָ֥ם אָבִ֖ינוּ בָּֽעָרְלָֽה.
4:13	כִּ֡י לֹא֩ עַל־יְדֵ֨י הַתּוֹרָ֜ה הַֽהַבְטָחָ֤ה לְאַבְרָהָם֨ א֣וֹ לְזַרְע֔וֹ לִהְיוֹת֖וֹ יוֹרֵ֣שׁ הַתֵּבֵ֑ל כִּ֛י אִם־עַל־יְדֵ֖י צִדְקַ֥ת הָֽאֱמוּנָֽה.
4:14	כִּ֗י אִם־אֵ֨לֶּה אֲשֶׁ֤ר מִן־הַתּוֹרָה֨ יֽוֹרְשִׁ֣ים הֵ֔ם נָב֖וֹקָה הָֽאֱמוּנָ֑ה וּבֻטְּלָ֖ה הַֽהַבְטָחָֽה.
4:15	כִּ֥י הַתּוֹרָ֖ה פֹּעֶ֣לֶת חָר֑וֹן כִּ֗י בַּֽאֲשֶׁר֨ אֵ֣ין תּוֹרָ֔ה שָׁ֖ם אֵ֥ין עֲבֵרָֽה.
4:16	בַּֽעֲב֥וּר זֹאת֨ מֵֽאֱמוּנָ֣ה הִ֔יא לְמַ֥עַן תִּהְיֶ֖ה לְפִ֣י הַחָ֑סֶד לְמַעַן֩ הֱי֨וֹת הַֽהַבְטָחָ֤ה נְכוֹנָה֨ לְכָל־הַזֶּ֔רַע לֹ֣א לְבַ֞ד לַֽאֲשֶׁ֣ר ׀ ה֣וּא מִן־הַתּוֹרָ֗ה כִּ֠י אִם־גַּ֨ם לַֽאֲשֶׁ֥ר הוּא֨ מֵֽאֱמוּנַ֣ת אַבְרָהָ֔ם אֲשֶׁ֥ר ה֖וּא אֲבִ֥י כֻלָּֽנוּ.
4:17	כַּכָּת֕וּב כִּ֛י אַב־הֲמ֥וֹן גּוֹיִ֖ם נְתַתִּ֑יךָ לִפְנֵ֣י הָֽאֱלֹהִ֗ים אֲשֶׁר֙ הֶֽאֱמִ֣ין בּ֔וֹ הַמְּחַיֶּ֣ה אֶת־הַמֵּתִ֔ים וְהַקּוֹרֵ֛א לְלֹ֥א הוִֹ֖ים כְּהוִֹֽים.
4:18	אֲשֶׁ֞ר מִח֤וּץ לַתִּקְוָה֨ עַל־תִּקְוָ֣ה הֶֽאֱמִ֔ין לְמַ֣עַן הֱיוֹת֔וֹ אֲבִ֖י גּוֹיִ֣ם רַבִּ֑ים כְּפִ֣י הַנֶּֽאֱמָ֔ר כֹּ֖ה יִהְיֶ֥ה זַרְעֶֽךָ.
4:19	וּמִבְּלִ֨י רְפוֹת֜וֹ בָּֽאֱמוּנָ֗ה לֹ֤א הִתְבּוֹנֵן֨ אֶת־גְּוִיָּת֣וֹ שֶׁלּ֔וֹ אֲשֶׁר֨ כְּבָ֣ר מֵ֔תָה בִּהְיוֹת֖וֹ כְּבֶן־מֵאָ֣ה שָׁנָ֑ה גַּ֥ם לֹ֕א אֶת־תְּמוּתַ֖ת רֶ֥חֶם שָׂרָֽה.
4:20	וּבְהַבְטָחַ֧ת הָֽאֱלֹהִ֛ים לֹ֥א סִפֵּ֖ק עַל־יְדֵ֣י לֹֽא־אֵמ֑וּן כִּ֚י אִם־הִתְגַּבֵּ֣ר בָּֽאֱמוּנָ֔ה בְּתֵ֥ת כָּב֖וֹד לָֽאֱלֹהִֽים.
4:21	בִּהְי֨וֹת ל֜וֹ מְלֹ֣א בִטְחָ֗ה כִּ֚י אֵ֣ת אֲשֶׁ֣ר הִבְטִ֔יחַ יָכֹ֖ל גַּ֥ם לַֽעֲשֽׂוֹת.
4:22	לָכֵ֕ן גַּ֛ם נֶחְשְׁבָ֥ה ל֖וֹ לִצְדָקָֽה.
4:23	וְלֹ֤א נִכְתַּב֨ לְמַֽעֲנ֣וֹ לְבַדּ֔וֹ כִּ֥י נֶחְשְׁבָ֖ה לֽוֹ.
4:24	כִּ֚י אִם־גַּ֣ם לְמַֽעֲנֵ֔נוּ אֲשֶׁ֛ר הִ֥יא עֲתִידָ֖ה לְהֵֽחָשֵׁ֣ב לָ֑נוּ הַמַּֽאֲמִינִ֗ים בַּמֵּקִ֛ים אֶת־יֵשׁ֥וּעַ אֲדֹנֵ֖ינוּ מֵעִ֥ם הַמֵּתִֽים.
4:25	אֲשֶׁ֥ר נִמְסַ֖ר בַּֽעֲב֣וּר פְּשָׁעֵ֑ינוּ וְהוּקַ֖ם בַּֽעֲב֥וּר הִצְטַדְּקוּתֵֽנוּ.
5:1	לָכֵ֕ן בִּהְיוֹתֵ֥נוּ נִצְדָּקִ֖ים מֵֽאֱמוּנָ֑ה יֵ֣שׁ לָ֤נוּ שָׁלוֹם֨ אֶל־הָֽאֱלֹהִ֔ים עַל־יְדֵ֥י אֲדֹנֵ֖ינוּ יֵשׁ֥וּעַ הַמָּשִֽׁיחַ.
5:2	אֲשֶׁ֣ר עַל־יָד֡וֹ גַּם֩ יֵ֨שׁ לָ֜נוּ הִתְקָֽרְב֤וּת ׀ בָּֽאֳמוּנָה֨ אֶל־הַחֶ֣סֶד הַזֶּ֔ה אֲשֶׁ֛ר אֲנַ֥חְנוּ עֹֽמְדִ֖ים בּ֑וֹ וּמִתְהַלְלִ֣ים אֲנַ֔חְנוּ בְּתִקְוַ֖ת כְּב֥וֹד הָֽאֱלֹהִֽים.
5:3	וְלֹ֤א זֹאת֨ לְבַ֔ד כִּ֛י אִם־גַּ֥ם נִתְהַלֵּ֖ל בַּצָּר֑וֹת בְּדַעְתֵּ֕נוּ כִּ֥י הַצָּרָ֖ה פֹּעֶ֥לֶת תּוֹחֶֽלֶת.
5:4	וְהַתּוֹחֶ֖לֶת בֹּ֑חַן וְהַבֹּ֖חַן תִּקְוָֽה.
5:5	וְהַתִּקְוָ֖ה אֵינֶ֣נָּה מְבִישָׁ֑ה כִּ֞י אַֽהֲבַ֤ת הָֽאֱלֹהִים֨ נִשְׁפְּכָ֣ה בְלִבּוֹתֵ֔ינוּ בְּר֥וּחַ הַקֹּ֖דֶשׁ הַנִּתָּ֥ן לָֽנוּ.
5:6	כִּ֗י בְּע֤וֹד הֱיוֹתֵ֨נוּ֨ רָפִ֔ים הַמָּשִׁ֕יחַ בְּעִתָּ֥ה ׀ מֵ֖ת בְּעַ֥ד הָֽרְשָׁעִֽים.
5:7	כִּ֚י בְּעַ֣ד צַדִּ֔יק בְּק֖וֹשִׁי יָמ֣וּת אִ֑ישׁ כִּ֚י בְּעַ֣ד הַטּ֔וֹב אוּלַ֥י אִ֛ישׁ גַּ֥ם יִתְעָרֵ֖ב לָמֽוּת.
5:8	אַ֚ךְ הָֽאֱלֹהִ֔ים מְשַׁבֵּ֖חַ אֶת־אַֽהֲבָת֣וֹ אֵלֵ֑ינוּ כִּ֗י בְּעוֹד֨ הֱיוֹתֵ֣נוּ חַטָּאִ֔ים הַמָּשִׁ֖יחַ מֵ֥ת בַּֽעֲדֵֽנוּ.
5:9	לָכֵ֞ן כַּמָּ֣ה יוֹתֵ֗ר בְּהִצָּֽדְקֵ֤נוּ עַתָּה֙ בְּדָמ֔וֹ נִוָּשֵׁ֥עַ עַל־יָ֖דוֹ מִן־הֶֽחָרֽוֹן.
5:10	כִּ֠י אִ֞ם כַּֽאֲשֶׁ֨ר הָיִ֤ינוּ אֽוֹיְבִים֨ נִרְצֵ֣ינוּ אֶל־הָֽאֱלֹהִ֔ים עַל־יְדֵ֖י מ֣וֹת בְּנ֑וֹ כַּמָּ֤ה יוֹתֵר֨ בִּהְיוֹתֵ֣נוּ נִרְצִ֔ים נִוָּשֵׁ֖עַ בְּחַיָּֽיו.
5:11	וְלֹ֣א ׀ זֹ֣את לְבַ֗ד כִּ֤י אִם־גַּם֨ נְתְהַלֵּ֣ל בָּֽאֱלֹהִ֔ים עַל־יְדֵ֥י אֲדֹנֵ֖ינוּ יֵשׁ֣וּעַ הַמָּשִׁ֑יחַ אֲשֶׁ֧ר עַל־יָד֛וֹ לָקַ֥חְנוּ עַתָּ֖ה אֶת־הַֽהִתְרַצּֽוּת.
5:12	לָ֠כֵן כַּֽאֲשֶׁ֞ר עַל־יְדֵ֣י ׀ אָדָ֣ם אֶחָ֗ד הַחֵטְא֨ בָּ֣א אֶל־הָֽעוֹלָ֔ם וְעַל־יְדֵ֥י הַחֵ֖טְא הַמָּ֑וֶת וּבְכֵ֞ן הַמָּ֤וֶת עָבַר֨ עַל־כָּל־בְּנֵ֣י אָדָ֔ם בַּֽאֲשֶׁ֖ר הַכֹּ֥ל חָטָֽאוּ.
5:13	כִּ֚י עַד־הַתּוֹרָ֔ה חֵ֖טְא הָיָ֣ה בָֽעוֹלָ֑ם אַ֣ךְ חֵ֞טְא אֵ֤ין נֶחְשָׁב֨ לְאִ֔ישׁ בְּאֵ֖ין הֱי֥וֹת תּוֹרָֽה.
5:14	אוּלָ֗ם מָשַׁ֤ל הַמָּ֨וֶת֨ מֵֽאָדָ֣ם עַד־מֹשֶׁ֔ה גַּ֠ם עַ֣ל אֲשֶׁ֧ר לֹא־חָֽטְא֛וּ בִּדְמ֖וּת פֶּ֣שַׁע אָדָ֑ם אֲשֶׁ֣ר ה֔וּא תַּבְנִ֥ית זֶ֖ה הֶֽעָתִֽיד.
5:15	אַ֚ךְ לֹ֣א כַפֶּ֔שַׁע כֵּ֖ן גַּ֣ם הַֽחֲנִינָ֑ה כִּ֗י אִ֣ם בְּפֶ֤שַׁע הָֽאֶחָד֨ הָֽרַבִּ֣ים מֵ֔תוּ כַּמָּ֨ה יוֹתֵ֜ר חֶ֣סֶד הָֽאֱלֹהִ֗ים וְ֠הַמַּתָּנָה בְּחֶ֨סֶד הָֽאָדָ֧ם הָֽאֶחָ֛ד יֵשׁ֥וּעַ הַמָּשִׁ֖יחַ הֶעְדִּ֥יף עַל־הָֽרַבִּֽים.
5:16	וְלֹ֗א כְּמ֨וֹ עַל־יְדֵ֧י הָֽאֶחָ֛ד הַֽחֹטֵ֖א הַמַּתָּנָ֑ה כִּ֣י אָמְנָ֞ם הַמִּשְׁפָּ֤ט ׀ מֵֽאֶחָד֨ לְהַרְשָׁעָ֔ה אַ֚ךְ הַֽחֲנִינָ֔ה מִפְּשָׁעִ֥ים רַבִּ֖ים לְהִצְטַדְּקֽוּת.
5:17	כִּ֚י אִם־בְּפֶ֣שַׁע הָֽאֶחָ֔ד הַמָּ֛וֶת מָשַׁ֖ל עַל־יְדֵ֣י הָֽאֶחָ֑ד כַּמָּ֣ה יוֹתֵ֡ר הַלֹּֽקְחִים֩ אֶת־שִׁפְעַ֨ת הַחֶ֜סֶד וְאֶת־מַתְּנַ֣ת הַצְּדָקָ֗ה יִמְשְׁל֧וּ ׀ בְּחַיִּ֛ים עַל־יְדֵ֥י הָֽאֶחָ֖ד יֵשׁ֥וּעַ הַמָּשִֽׁיחַ.
5:18	לָכֵן֘ אֵפוֹא֒ כַּֽאֲשֶׁ֗ר עַל־יְדֵי֨ פֶּ֣שַׁע הָֽאֶחָ֔ד עַ֛ל כָּל־בְּנֵ֥י הָֽאָדָ֖ם לְהַרְשָׁעָ֑ה כֵּ֗ן גַּ֚ם בְּצִדְקַ֣ת הָֽאֶחָ֔ד עַ֛ל כָּל־בְּנֵ֥י הָֽאָדָ֖ם לְהִצְטַדְּק֥וּת הַֽחַיִּֽים.
5:19	כִּ֗י כַּֽאֲשֶׁ֞ר עַל־יְדֵ֤י מְרִי֨ הָֽאָדָ֣ם הָֽאֶחָ֔ד חַטָּאִ֖ים הֽוּשְׁת֣וּ הָֽרַבִּ֑ים כֵּ֗ן גַּ֚ם עַל־יְדֵי֨ מִשְׁמַ֣עַת הָֽאֶחָ֔ד צַדִּיקִ֖ים יֽוּשְׁת֥וּ הָֽרַבִּֽים.
5:20	אַ֗ךְ הַתּוֹרָה֨ בָּ֣אָה עַל־זֹ֔את לְמַ֖עַן יִרְבֶּ֣ה הַפָּ֑שַׁע אַ֗ךְ בַּֽאֲשֶׁ֤ר רָבָה֨ הַחֵ֔טְא שָׁ֛ם הֶעְדִּ֥יף יוֹתֵ֖ר הַחָֽסֶד.
5:21	לְמַ֜עַן כַּֽאֲשֶׁ֥ר מָשַׁ֛ל הַחֵ֖טְא לַמָּ֑וֶת כֵּ֣ן ׀ גַּ֣ם הַחֶ֗סֶד יִמְשֹׁ֞ל עַל־יְדֵ֤י צְדָקָה֨ לְחַיֵּ֣י עֽוֹלָמִ֔ים עַל־יְדֵ֛י יֵשׁ֥וּעַ הַמָּשִׁ֖יחַ אֲדֹנֵֽינוּ.
6:1	מָ֥ה אֵפ֖וֹא נֹאמַ֑ר הֲנָלִ֣ין בַּחֵ֔טְא לְמַ֥עַן יִרְבֶּ֖ה הַחָֽסֶד.
6:2	חָלִ֑ילָה אֲנַ֜חְנוּ אֲשֶׁ֥ר מַ֨תְנוּ֨ לַחֵ֔טְא אֵיכָ֛ה נִחְיֶ֥ה ב֖וֹ עֽוֹד.
6:3	הַֽאִם־לֹ֣א יְדַעְתֶּ֔ם כִּ֣י כָל־אֵ֗לֶּה אֲשֶׁ֤ר נִטְבַּ֨לְנוּ֨ בְּיֵשׁ֣וּעַ הַמָּשִׁ֔יחַ בְּמוֹת֖וֹ נִטְבָּֽלְנוּ.
6:4	לָכֵן֨ נִקְבַּ֣רְנוּ עִמּ֔וֹ עַל־יְדֵ֖י הַטְּבִילָ֣ה לַמָּ֑וֶת לְמַ֡עַן כַּֽאֲשֶׁר֩ הוּקַ֨ם הַמָּשִׁ֤יחַ מֵֽעִם־הַמֵּתִים֨ בִּכְב֣וֹד הָאָ֔ב כֵּ֣ן גַּ֥ם אֲנַ֛חְנוּ בְּחִדּ֥וּשׁ הַֽחַיִּ֖ים נִתְהַלֵּֽכָה.
6:5	כִּ֗י אִם־נִהְיִ֧ינוּ נְטוּעִ֛ים אִתּ֖וֹ בִּדְמ֣וּת מוֹת֑וֹ אָכֵן֨ נִהְיֶ֣ה כֵ֔ן גַּ֖ם בִּדְמ֥וּת תְּקֽוּמָתֽוֹ.
6:6	בְּדַעְתֵּ֣נוּ זֹ֗את כִּ֣י הָֽאָדָ֧ם הַיָּשָׁ֛ן שֶׁלָּ֖נוּ נִצְלַ֣ב אִתּ֑וֹ לְמַ֤עַן תְּבֻטַּל֨ גְּוִיַּ֣ת הַחֵ֔טְא לְבִלְתִּ֛י נַֽעֲבֹ֥ד ע֖וֹד אֶת־הַחֵֽטְא.
6:7	כִּ֗י זֶ֚ה אֲשֶׁ֣ר מֵ֔ת נִצְדַּ֖ק מִן־הַחֵֽטְא.
6:8	אַ֛ךְ אִ֥ם מִ֖תְנוּ עִ֣ם הַמָּשִׁ֑יחַ מַֽאֲמִינִ֣ים אֲנַ֔חְנוּ כִּ֛י גַּם־נִחְיֶ֥ה יַ֖חַד עִמּֽוֹ.
6:9	בְּדַעְתֵּ֕נוּ כִּ֣י הַמָּשִׁ֗יחַ בִּהְיוֹת֤וֹ מוּקָם֨ מֵעִ֣ם הַמֵּתִ֔ים לֹ֥א יָמ֖וּת ע֑וֹד הַמָּ֕וֶת לֹֽא־יִמְשֹׁ֥ל בּ֖וֹ עֽוֹד.
6:10	כִּ֗י מָ֚ה אֲשֶׁ֣ר ה֣וּא מֵ֔ת לַחֵ֛טְא ה֥וּא מֵ֖ת פַּ֣עַם אֶחָ֑ת וּמָה֨ אֲשֶׁ֣ר ה֣וּא חַ֔י חַ֥י ה֖וּא לָֽאֱלֹהִֽים.
6:11	כְּמוֹ־כֵ֗ן גַּ֤ם אַתֶּם֨ חַשְּׁב֣וּ אֶת־עַצְמְכֶ֔ם לִהְי֥וֹת מֵתִ֖ים אָמְנָ֣ם ׀ לַחֵ֑טְא אַ֤ךְ חַיִּים֨ לָֽאֱלֹהִ֔ים בְּיֵשׁ֥וּעַ הַמָּשִׁ֖יחַ אֲדֹנֵֽינוּ.
6:12	לָכֵ֕ן אַל־יִמְשֹׁ֥ל הַחֵ֖טְא בִּגְוִיַּתְכֶ֣ם בַּת־תְּמוּתָ֑ה לִשְׁמֹ֖עַ אֶל־תַּֽאֲוֹתֶֽיהָ.
6:13	וְ֠גַם לֹ֤א תַצִּ֨יגוּ֨ אֶת־אֶבְרֵיכֶ֔ם לִכְלֵי־נֶ֥שֶׁק עַוְלָ֖ה לַחֵ֑טְא כִּ֠י אִם־הַצִּ֨יגוּ אֶת־עַצְמְכֶ֜ם לָֽאֱלֹהִ֗ים כְּמ֤וֹ הַֽחַיִּים֨ מֵעִ֣ם הַמֵּתִ֔ים וְאֶבְרֵיכֶ֛ם לִכְלֵי־נֶ֥שֶׁק צְדָקָ֖ה לָֽאֱלֹהִֽים.
6:14	כִּ֛י הַחֵ֥טְא לֹֽא־יִמְשֹׁ֖ל בָּכֶ֑ם כִּ֣י אַתֶּ֗ם אֵֽינְכֶם֨ תַּ֣חַת הַתּוֹרָ֔ה כִּ֖י אִם־תַּ֥חַת הַחָֽסֶד.
6:15	מָה֘ אֵפוֹא֒ הֲנֶֽחֱטָ֗א יַ֣עַן אֵ֤ין אֲנַ֨חְנוּ֨ תַּ֣חַת הַתּוֹרָ֔ה כִּ֖י אִם־תַּ֣חַת הַחָ֑סֶד חָלִֽילָה.
6:16	הֲלֹ֣א יְדַעְתֶּ֗ם כִּ֡י אֲשֶׁ֣ר אַתֶּ֣ם מַצִּיגִים֩ אֶת־עַצְמְכֶ֨ם אֵלָ֤יו ׀ עֲבָדִים֨ לְמִשְׁמַ֔עַת עֲבָדִ֤ים אַתֶּם֨ לָזֶ֔ה אֲשֶׁ֛ר אַתֶּ֥ם שֹֽׁמְעִ֖ים ל֑וֹ אִ֣ם עַבְדֵ֥י חֵטְא֙ לַמָּ֔וֶת א֛וֹ עַבְדֵ֥י מִשְׁמַ֖עַת לִצְדָקָֽה.
6:17	אַ֖ךְ תּוֹדָ֣ה לָֽאֱלֹהִ֑ים כִּ֣י הֱיִיתֶ֞ם עַבְדֵ֣י הַחֵ֗טְא אַ֣ךְ שְׁמַעְתֶּ֤ם מִן־הַלֵּב֨ אֶל־תַּבְנִ֣ית הַלֶּ֔קַח אֲשֶׁ֥ר נִמְסַרְתֶּ֖ם אֵלֶֽיהָ.
6:18	אַ֖ךְ חֻפַּשְׁתֶּ֣ם מִן־הַחֵ֑טְא וְהָֽעֳבַדְתֶּ֖ם אֶת־הַצְּדָקָֽה.
6:19	כְּדֶ֤רֶךְ אֱנוֹשׁ֨ אָֽנֹכִ֣י מְדַבֵּ֔ר בַּֽעֲב֖וּר רִפְי֣וֹן בְּשַׂרְכֶ֑ם כִּ֡י כַּֽאֲשֶׁר֩ הִצַּגְתֶּ֨ם אֶת־אֶבְרֵיכֶ֜ם עֲבָדִ֣ים אֶל־הַטֻּמְאָ֗ה וְאֶל־הָאָ֨וֶן֨ לָאָ֔וֶן כֵּ֣ן עַתָּ֗ה הַצִּ֧יגוּ אֶת־אֶבְרֵיכֶ֛ם עֲבָדִ֥ים אֶל־הַצְּדָקָ֖ה לִקְדֻשָּֽׁה.
6:20	כִּ֗י כַּֽאֲשֶׁ֤ר הֱיִיתֶם֨ עַבְדֵ֣י הַחֵ֔טְא חָפְשִׁ֥ים הֱיִיתֶ֖ם אֶל־הַצְּדָקָֽה.
6:21	מַה־פְּרִ֣י אֵפ֗וֹא הָיָ֥ה אָז֨ לָכֶ֔ם בָּאֵ֗לֶּה אֲשֶׁ֥ר עַתָּ֛ה אַתֶּ֥ם בּוֹשִׁ֖ים מֵהֶ֑ם כִּ֛י אַֽחֲרִ֥ית אֵ֖לֶּה מָֽוֶת.
6:22	אַ֣ךְ עַ֠תָּה בִּהְיֽוֹתְכֶ֨ם מְחֻפָּשִׁ֤ים מִן־הַחֵטְא֨ וּמָֽעֳבָדִ֣ים אֶת־הָֽאֱלֹהִ֔ים יֵֽשׁ־לָכֶ֥ם פִּרְיְכֶ֖ם לִקְדֻשָּׁ֑ה וְהָאַֽחֲרִ֖ית חַיֵּ֥י עֽוֹלָמִֽים.
6:23	כִּ֛י שְׂכַ֥ר הַחֵ֖טְא מָ֑וֶת אַ֗ךְ חֲנִינַ֤ת הָֽאֱלֹהִים֨ חַיֵּ֣י עֽוֹלָמִ֔ים בְּיֵשׁ֥וּעַ הַמָּשִׁ֖יחַ אֲדֹנֵֽינוּ.
7:1	הֲלֹ֤א יְדַעְתֶּם֨ אַחַ֔י כִּ֛י אֶל־הַיּֽוֹדְעִ֥ים אֶת־הַתּוֹרָ֖ה אָֽנֹכִ֣י מְדַבֵּ֑ר כִּ֤י הַתּוֹרָה֨ מֹשֶׁ֣לֶת בָּֽאָדָ֔ם כָּל־הָעֵ֖ת אֲשֶׁר־ה֥וּא חָֽי.
7:2	כִּ֣י הָֽאִשָּׁ֗ה אֲשֶׁ֥ר הִיא֨ תַּ֣חַת אִ֔ישׁ קְשׁ֥וּרָה הִ֛יא בְּאִישָׁ֥הּ הַחַ֖י עַל־פִּ֣י הַתּוֹרָ֑ה אַ֚ךְ אִם־מֵ֣ת הָאִ֔ישׁ הֻתְּרָ֖ה מִתּוֹרַ֥ת הַבָּֽעַל.
7:3	לָכֵ֣ן אָמְנָ֗ם אִם־תִּהְיֶ֨ה לְאִ֤ישׁ אַחֵר֨ בַּֽחֲי֣וֹת הַבַּ֔עַל נֹאֶ֖פֶת תִּקָּרֵ֑א אַ֞ךְ אִם־מֵ֣ת הָאִ֗ישׁ חָפְשִׁיָּ֥ה הִיא֨ מִן־הַתּוֹרָ֔ה לְבִלְתִּ֤י הֱיוֹתָהּ֨ נֹאֶ֔פֶת אִם־תִּהְיֶ֖ה לְאִ֥ישׁ אַחֵֽר.
7:4	וְלָכֵ֣ן אַחַ֗י גַּם־אַתֶּם֨ מַתֶּ֣ם לַתּוֹרָ֔ה עַל־יְדֵ֖י גְּוִיַּ֣ת הַמָּשִׁ֑יחַ לְמַ֣עַן ׀ הֱיֽוֹתְכֶ֣ם לְאַחֵ֗ר אֶל־הַמּוּקָם֨ מֵעִ֣ם הַמֵּתִ֔ים לְמַ֛עַן נַֽעֲשֶׂ֥ה פְרִ֖י לָֽאֱלֹהִֽים.
7:5	כִּ֞י כַּֽאֲשֶׁ֣ר ׀ הָיִ֣ינוּ בַבָּשָׂ֗ר חוּשֵׁ֤י הַֽחֲטָאִים֨ אֲשֶׁ֣ר עַל־יְדֵ֣י הַתּוֹרָ֔ה פָּֽעֲל֖וּ בַּֽאֲבָרֵ֑ינוּ לְמַ֛עַן הָבִ֥יא פְרִ֖י לַמָּֽוֶת.
7:6	אַ֤ךְ עַתָּה֨ הֻתַּ֣רְנוּ מִן־הַתּוֹרָ֔ה בְּמוּתֵ֕נוּ אֶל־אֲשֶׁ֖ר נֶֽאֱחַ֣זְנוּ בָ֑הּ עַד֤ לַֽעֲבֹד֙ בְּחִדּ֣וּשׁ הָר֔וּחַ וְלֹ֖א בְּיֹ֥שֶׁן הָאֽוֹת.
7:7	מָ֣ה אֵפ֞וֹא נֹאמַ֗ר הַאִ֣ם הַתּוֹרָה֘ חֵטְא֒ חָלִ֗ילָה אוּלָ֤ם אֶת־הַחֵטְא֨ לֹ֣א יָדַ֔עְתִּי זֽוּלָתִ֖י עַל־יְדֵ֣י הַתּוֹרָ֑ה כִּ֣י גַ֤ם אֶת־הַֽחֶמְדָּה֨ לֹ֣א יָדַ֔עְתִּי לוּלֵ֛א אָֽמְרָ֥ה הַתּוֹרָ֖ה לֹ֥א תַחְמֹֽד.
7:8	אַ֣ךְ הַחֵ֗טְא בְּקַ֤חַת עֲלִילָה֨ עַל־יְדֵ֣י הַמִּצְוָ֔ה פָּ֥עַל בִּ֖י כָּל־חֶמְדָּ֑ה כִּ֚י בִּבְלִ֣י תוֹרָ֔ה חֵ֖טְא מֵֽת.
7:9	וַֽאֲנִ֥י הָיִ֛יתִי בִּבְלִ֥י תוֹרָ֖ה לְפָנִ֑ים אַ֣ךְ ׀ בְּב֣וֹא הַמִּצְוָ֗ה הַחֵטְא֨ חָ֣יָה ע֔וֹד וַֽאֲנִ֖י מַֽתִּי.
7:10	וַתִּמָּצֵ֣א לִ֗י הַמִּצְוָ֛ה אֲשֶׁ֥ר לַֽחַיִּ֖ים לַמָּֽוֶת.
7:11	כִּ֣י הַחֵ֔טְא בְּקַ֧חַת עֲלִילָ֛ה עַל־יְדֵ֥י הַמִּצְוָ֖ה הִשִּׁיאָ֑נִי וּבָ֖הּ הֱמִיתָֽנִי.
7:12	וּבְכֵ֕ן הַתּוֹרָ֖ה קְדוֹשָׁ֣ה הִ֑יא וְהַ֨מִּצְוָ֔ה קְדוֹשָׁ֥ה וִֽישָׁרָ֖ה וְטוֹבָֽה.
7:13	הֲכִי֩ הַדָּבָ֨ר הַטּ֜וֹב נִהְיָ֣ה לִי֘ לְמָוֶת֒ חָלִ֗ילָה כִּ֚י אִם־הַחֵ֔טְא לְמַ֨עַן יֵֽרָאֶ֜ה חֵ֗טְא בְּפָֽעֳל֥וֹ לִי֨ אֶת־הַמָּ֔וֶת עַל־יְדֵ֖י הַדָּבָ֣ר הַטּ֑וֹב לְמַ֨עַן יִהְיֶ֥ה הַחֵ֛טְא חַטָּ֥א עַל־יֶ֖תֶר עַל־יְדֵ֥י הַמִּצְוָֽה.
7:14	כִּ֣י יָדַ֔עְנוּ כִּ֥י הַתּוֹרָ֖ה רֽוּחֲנִ֣ית הִ֑יא אַ֤ךְ אֲנִי֨ בְּשָׂרִ֣י אָֽנֹכִ֔י נִמְכָּ֖ר תַּ֥חַת הַחֵֽטְא.
7:15	כִּ֠י אֵ֣ת אֲשֶׁ֥ר אֲנִ֛י פֹּעֵ֖ל אֵינֶ֣נִּי יוֹדֵ֑עַ כִּ֡י לֹ֣א אֵת֩ אֲשֶׁ֨ר אֲנִ֤י חָפֵץ֨ זֹ֣את אֲנִ֣י פֹעֵ֔ל כִּ֤י אִם־אֵת֨ אֲשֶׁ֣ר אֲנִ֣י שֹׂנֵ֔א זֹ֖את אֲנִ֥י עֹשֶֽׂה.
7:16	וְאִ֗ם אֵ֚ת אֲשֶׁ֣ר אֵינֶ֣נִּי חָפֵ֔ץ זֹ֖את אֲנִ֣י עֹשֶׂ֑ה אֲנִ֤י מֵעִיד֨ אֶל־הַתּוֹרָ֔ה כִּ֥י טוֹבָ֖ה הִֽיא.
7:17	וְעַתָּ֕ה לֹ֥א ע֛וֹד אָֽנֹכִ֖י פֹּעֵ֣ל אֹתָ֑הּ כִּ֥י אִם־הַחֵ֖טְא הַשֹּׁכֵ֥ן בִּֽי.
7:18	כִּ֣י יָדַ֔עְתִּי כִּ֞י אֵ֣ין ׀ שֹׁכֵ֣ן בִּ֗י זֶ֥ה ה֛וּא בִּבְשָׂרִ֖י דָּבָ֣ר ט֑וֹב כִּ֗י חָפֹץ֨ רֹבֵ֣ץ בִּ֔י אַ֛ךְ פָּעֹ֥ל אֶת־הַטּ֖וֹב אֵינֶ֥נִּי מֹצֵֽא.
7:19	כִּ֧י אֶת־הַטּ֛וֹב אֲשֶׁר־אֲנִ֥י חָפֵ֖ץ אֵינֶ֣נִּי עֹשֶׂ֑ה כִּ֣י אִם־אֶת־הָרָ֗ע אֲשֶׁר֨ אֵינֶ֣נִּי חָפֵ֔ץ אֶת־זֶ֖ה אֲנִ֥י פֹעֵֽל.
7:20	וְאִ֗ם אֵ֚ת אֲשֶׁ֣ר אֵינֶ֣נִּי חָפֵ֔ץ זֹ֖את אֲנִ֣י עֹשֶׂ֑ה לֹ֣א ע֤וֹד אָֽנֹכִי֨ פֹּעֵ֣ל אֹתָ֔הּ כִּ֥י אִם־הַחֵ֖טְא הַשֹּׁכֵ֥ן בִּֽי.
7:21	לָכֵ֕ן אֲנִ֥י מוֹצֵ֖א אֶת־הַתּוֹרָ֑ה כִּ֣י לִ֗י הֶֽחָפֵץ֨ לַֽעֲשׂ֣וֹת אֶת־הַטּ֔וֹב הָרָ֖ע רֹבֵ֥ץ בִּֽי.
7:22	כִּ֛י אֲנִ֥י מִתְעַנֵּ֖ג בְּתוֹרַ֣ת הָֽאֱלֹהִ֑ים לְפִ֖י הָֽאָדָ֥ם הַפְּנִימִֽי.
7:23	אַ֗ךְ אֲנִ֨י רֹאֶ֜ה תּוֹרָ֤ה אַחֶ֨רֶת֨ בַּֽאֲבָרַ֔י הַ֨נִּלְחָמָ֔ה נֶ֖גֶד תּוֹרַ֣ת שִׂכְלִ֑י וְהַשֹּׁבָ֥ה אֹתִ֛י אֶל־תּוֹרַ֥ת הַחֵ֖טְא אֲשֶׁ֥ר בַּֽאֲבָרָֽי.
7:24	אֱנ֥וֹשׁ אֻמְלָ֖ל אָנֹ֑כִי מִ֣י יַצִּילֵ֔נִי מִגְּוִיַּ֖ת הַמָּ֥וֶת הַזֶּֽה.
7:25	אֲנִי֨ מוֹדֶ֣ה לָֽאֱלֹהִ֔ים עַל־יְדֵ֛י יֵשׁ֥וּעַ הַמָּשִׁ֖יחַ אֲדֹנֵ֑ינוּ וּבְכֵ֞ן אֲנִ֣י עַצְמִ֗י אָמְנָ֨ם בַּשֵּׂ֜כֶל אֲנִ֤י עֹבֵד֨ אֶת־תּוֹרַ֣ת הָֽאֱלֹהִ֔ים אַ֚ךְ בַּבָּשָׂ֔ר אֵ֖ת תּוֹרַ֥ת הַחֵֽטְא.
8:1	לָכֵ֗ן עַתָּה֨ אֵ֣ין הַרְשָׁעָ֔ה לָאֵ֕לֶּה אֲשֶׁ֖ר בְּיֵשׁ֣וּעַ הַמָּשִׁ֑יחַ הַמִּתְהַלְּכִ֗ים לֹ֚א כְּפִ֣י הַבָּשָׂ֗ר כִּ֖י אִם־כְּפִ֥י הָרֽוּחַ.
8:2	כִּ֠י תּוֹרַ֨ת ר֤וּחַ הַֽחַיִּים֙ בְּיֵשׁ֣וּעַ הַמָּשִׁ֔יחַ חִפְּשָׁ֣ה אֹתִ֗י מִתּוֹרַ֥ת הַחֵ֖טְא וְהַמָּֽוֶת.
8:3	כִּ֠י אֲשֶׁ֞ר בְּאֵ֤ין יְכֹ֨לֶת֨ הַתּוֹרָ֔ה בַּֽאֲשֶׁ֥ר רָֽפְתָ֖ה עַל־יְדֵ֣י הַבָּשָׂ֑ר הָֽאֱלֹהִ֡ים בִּשְׁלֹחַ֩ אֶת־בְּנ֨וֹ שֶׁלּ֜וֹ בִּדְמ֨וּת בְּשַׂ֤ר הַחֵטְא֨ וּבַֽעֲב֣וּר הַחֵ֔טְא הִרְשִׁ֥יעַ אֶת־הַחֵ֖טְא בַּבָּשָֽׂר.
8:4	לְמַ֧עַן תִּמָּלֵ֛א צִדְקַ֥ת הַתּוֹרָ֖ה בָּ֑נוּ הַמִּתְהַלְּכִ֗ים לֹ֚א כְּפִ֣י הַבָּשָׂ֔ר כִּ֖י אִם־כְּפִ֥י הָרֽוּחַ.
8:5	כִּ֗י אֲשֶׁ֥ר הֵם֨ כְּפִ֣י הַבָּשָׂ֔ר אֶת־דִּבְרֵ֥י הַבָּשָׂ֖ר הֵ֣ם מַשְׂכִּילִ֑ים אַ֗ךְ אֲשֶׁ֥ר הֵם֨ כְּפִ֣י הָר֔וּחַ אֵ֖ת דִּבְרֵ֥י הָרֽוּחַ.
8:6	כִּ֛י הַשְׂכֵּ֥ל הַבָּשָׂ֖ר מָ֑וֶת וְהַשְׂכֵּ֣ל הָר֔וּחַ חַיִּ֖ים וְשָׁלֽוֹם.
8:7	כִּ֚י הַשְׂכֵּ֣ל הַבָּשָׂ֔ר אֵיבָ֥ה הִ֖יא לָֽאֱלֹהִ֑ים כִּ֣י ׀ אֵינֶ֣נּוּ מוּשָׁ֗ת תַּ֚חַת תּוֹרַ֣ת הָֽאֱלֹהִ֔ים כִּ֖י גַּ֥ם לֹֽא־יוּכָֽל.
8:8	וַֽאֲשֶׁ֥ר הֵם֨ בַּבָּשָׂ֔ר לֹ֥א יֽוּכְל֖וּ לְהֵֽרָצ֥וֹת לָֽאֱלֹהִֽים.
8:9	אַ֣ךְ אַתֶּ֞ם לֹ֣א בַבָּשָׂ֤ר אַתֶּם֨ כִּ֣י אִם־בָּר֔וּחַ אִ֣ם אָמְנָ֔ם ר֥וּחַ הָֽאֱלֹהִ֖ים שֹׁכֵ֣ן בָּכֶ֑ם אַ֣ךְ אִ֗ישׁ אִם־ר֤וּחַ הַמָּשִׁ֨יחַ֨ אֵ֣ין ל֔וֹ זֶ֖ה אֵינֶ֥נּוּ שֶׁלּֽוֹ.
8:10	וְאִ֤ם הַמָּשִׁ֨יחַ֨ בָּכֶ֔ם אָמְנָ֕ם הַגְּוִיָּ֥ה מֵתָ֖ה בַּֽעֲב֣וּר חֵ֑טְא אַ֛ךְ הָר֥וּחַ חַיִּ֖ים בַּֽעֲב֥וּר צְדָקָֽה.
8:11	וְאִ֡ם רוּחַ֩ זֶ֨ה הַמֵּקִ֧ים אֶת־יֵשׁ֛וּעַ מֵעִ֥ם הַמֵּתִ֖ים שֹׁכֵ֣ן בָּכֶ֑ם הַמֵּקִ֨ים אֶת־הַמָּשִׁ֜יחַ מֵעִ֣ם הַמֵּתִ֗ים יְחַיֶּ֞ה גַּ֤ם אֶת־גְּוִיּֽוֹתֵיכֶם֙ בְּנֽוֹת־תְּמוּתָ֔ה בַּֽעֲב֥וּר רוּח֖וֹ הַשֹּׁכֵ֥ן בָּכֶֽם.
8:12	לָכֵ֤ן אֵפוֹא֨ אַחַ֔י חַיָּבִ֖ים אֲנָ֑חְנוּ לֹ֣א לַבָּשָׂ֔ר לִחְי֖וֹת כְּפִ֥י הַבָּשָֽׂר.
8:13	כִּ֗י אִ֣ם כְּפִ֧י הַבָּשָׂ֛ר אַתֶּ֥ם חַיִּ֖ים תָּמ֑וּתוּ אַ֠ךְ אִם־בָּר֜וּחַ אֶת־פָּֽעֳלֵ֧י הַגְּוִיָּ֛ה תְּמֽוֹתְת֖וּ תִּחְיֽוּן.
8:14	כִּ֗י כֹּ֣ל אֲשֶׁ֥ר הֵ֛ם מְנֻֽהָגִ֖ים בְּר֣וּחַ אֱלֹהִ֑ים אֵ֕לֶּה בְּנֵ֥י אֱלֹהִ֖ים הֵֽמָּה.
8:15	כִּ֣י לֹ֧א לְקַחְתֶּ֛ם ר֥וּחַ עַבְד֖וּת ע֣וֹד לְיִרְאָ֑ה כִּ֣י אִם־לְקַחְתֶּ֗ם ר֤וּחַ תְּשׂוּמַ֣ת בָּנִ֔ים אֲשֶׁ֥ר בּ֛וֹ אֲנַ֥חְנוּ צֹֽעֲקִ֖ים אַבָּ֥א ׀ הָאָֽב.
8:16	הָר֣וּחַ עַצְמ֗וֹ מֵעִיד֨ עִ֣ם רוּחֵ֔נוּ כִּ֛י בְּנֵ֥י אֱלֹהִ֖ים אֲנָֽחְנוּ.
8:17	וְאִ֤ם בָּנִים֨ גַּ֣ם יֽוֹרְשִׁ֔ים יֽוֹרְשֵׁ֣י אֱלֹהִ֔ים וְיֽוֹרְשִׁ֥ים חֲבֵרִ֖ים עִ֣ם הַמָּשִׁ֑יחַ אִ֣ם אָמְנָ֗ם אֲנַ֤חְנוּ מְעֻנִּים֨ עִמּ֔וֹ לְמַ֛עַן גַּ֥ם נְכֻבַּ֖ד עִמּֽוֹ.
8:18	כִּ֖י מְחַשֵּׁ֣ב אָ֑נִי כִּ֞י עִנּוּיֵ֣י ׀ הַזְּמָ֣ן הַזֶּ֗ה אֵינָ֤ם שֹׁוִים֨ אֶל־הַכָּב֔וֹד הֶֽעָתִ֖יד לְהִגָּל֥וֹת בָּֽנוּ.
8:19	כִּ֣י תוֹחֶ֤לֶת הַבְּרִיאָה֨ הָֽעֹרֶ֔גֶת מְצַפָּ֕ה לִהִתְגַּלּ֖וּת בְּנֵ֥י הָֽאֱלֹהִֽים.
8:20	כִּ֛י תַּ֥חַת הַהֶ֖בֶל הַבְּרִיאָ֣ה הוּשָׁ֑תָה לֹ֣א בְרָצ֔וֹן כִּ֧י אִם־בַּֽעֲב֛וּר הַשָּׁ֥ת תַּחְתָּ֖יו בְּתִקְוָֽה.
8:21	כִּ֞י גַּ֣ם ׀ הַבְּרִיאָ֣ה עַצְמָ֗הּ תְּחֻפַּשׁ֨ מֵֽעַבְד֣וּת הַשַּׁ֔חַת אֶל־חֹ֕פֶשׁ כְּב֖וֹד בְּנֵ֥י הָֽאֱלֹהִֽים.
8:22	כִּ֣י יָדַ֗עְנוּ כִּ֣י כָל־הַבְּרִיאָ֧ה נֶֽאֱנָחָ֛ה וְחָלָ֥ה יַחְדָּ֖יו עַד־עָֽתָּה.
8:23	וְלֹ֨א לְבַ֜ד הִ֗יא כִּ֣י אִם־גַּ֣ם אֲ֠נַחְנוּ אֲשֶׁ֨ר יֵ֥שׁ לָ֨נוּ רֵאשִׁ֣ית הָר֔וּחַ גַּ֛ם אֲנַ֥חְנוּ בְעַצְמֵ֖נוּ נֶֽאֱנָחִ֣ים בְּקִרְבֵּ֑נוּ בְּצַפּ֛וֹת אֶל־תְּשׂוּמַ֥ת הַבָּנִ֖ים אֶל־פְּד֥וּת גְּוִיָּתֵֽנוּ.
8:24	כִּ֣י בַתִּקְוָה֘ נוֹשַׁעְנוּ֒ אַ֚ךְ תִּקְוָ֣ה הַנִּרְאָ֔ה אֵינֶ֖נָּה תִּקְוָ֑ה כִּ֗י אֵ֚ת אֲשֶׁ֣ר אִ֣ישׁ רֹאֶ֔ה לָ֛מָּה גַּ֥ם יְקַוֶּ֖ה לֽוֹ.
8:25	אַ֗ךְ אִם־אֵת֩ אֲשֶׁ֨ר אֵינֶנּ֤וּ רֹאִים֨ אֲנַ֣חְנוּ מְקַוִּ֔ים בְּתוֹחֶ֖לֶת נְצַפֶּ֥ה לֽוֹ.
8:26	וּכְמוֹ־כֵ֗ן גַּ֤ם הָר֨וּחַ֨ עֹזֵ֣ר עִם־רִפְיוֹנֵ֔ינוּ כִּ֠י אֵ֣ת אֲשֶׁ֧ר נִתְפַּלֵּ֛ל כַּֽאֲשֶׁ֥ר צָרִ֖יךְ לֹ֣א יָדָ֑עְנוּ אַ֣ךְ הָר֤וּחַ עַצְמוֹ֨ מַפְגִּ֣יעַ בַּֽעֲדֵ֔נוּ בַּֽאֲנָח֖וֹת עֲצוּמ֥וֹת מִדַּבֵּֽר.
8:27	וְהַֽחֹקֵ֣ר אֶת־הַלְּבָב֔וֹת יוֹדֵ֖עַ אֶת־הַשְׂכֵּ֣ל הָר֑וּחַ כִּ֚י כְּפִ֣י אֱלֹהִ֔ים ה֥וּא מַפְגִּ֖יעַ בְּעַ֥ד הַקְּדֹשִֽׁים.
8:28	וְיָדַ֗עְנוּ כִּ֚י לָאֹֽהֲבִ֣ים אֶת־הָֽאֱלֹהִ֔ים כָּל־הַדְּבָרִ֧ים פֹּֽעֲלִ֛ים יַ֖חַד לְט֑וֹב לָאֵ֗לֶּה אֲשֶׁ֥ר הֵ֛ם קְרוּאִ֖ים כְּפִ֥י עֲצָתֽוֹ.
8:29	כִּ֠י אֵ֞ת אֲשֶׁ֣ר ׀ יְדָעָ֣ם מִקֶּ֗דֶם גַּ֤ם מִקֶּ֨דֶם֨ יְעָדָ֔ם לִהְי֛וֹת שָׁוֵ֥י הַתַּבְנִ֖ית לְצֶ֣לֶם בְּנ֑וֹ לְ֠מַעַן הֱיוֹתוֹ֨ בְּכ֔וֹר בֵּ֖ין אַחִ֥ים רַבִּֽים.
8:30	אַ֗ךְ אֵ֞ת אֲשֶׁ֣ר ׀ יְעָדָ֣ם מִקֶּ֗דֶם אֹתָם֨ גַּ֣ם קָרָ֔א אַ֗ךְ אֵ֚ת אֲשֶׁ֣ר קְרָאָ֔ם אֹתָ֖ם גַּ֣ם הִצְדִּ֑יק אַ֗ךְ אֵ֚ת אֲשֶׁ֣ר הִצְדִּיקָ֔ם אֹתָ֖ם גַּ֥ם פֵּאֵֽר.
8:31	וּמָ֥ה אֵפ֖וֹא נֹאמַ֣ר אֶל־אֵ֑לֶּה אִ֤ם הָֽאֱלֹהִים֨ בַּֽעֲדֵ֔נוּ מִ֖י נֶגְדֵּֽנוּ.
8:32	זֶ֠ה אֲשֶׁ֞ר גַּ֣ם עַל־בְּנ֤וֹ שֶׁלּוֹ֨ לֹ֣א חָמַ֔ל כִּ֛י אִם־מָסַ֥ר אֹת֖וֹ בַּֽעֲדֵ֣נוּ כֻלָּ֑נוּ אֵ֞יךְ לֹ֣א ׀ גַּ֣ם אִתּ֗וֹ יתֵּ֥ן לָ֛נוּ חִנָּ֖ם אֶת־הַכֹּֽל.
8:33	מִ֚י יָבִ֣יא שִׂטְנָ֔ה נֶ֖גֶד בְּחִירֵ֣י אֱלֹהִ֑ים הָֽאֱלֹהִ֖ים ה֥וּא הַמַּצְדִּֽיק.
8:34	מִ֥י ה֖וּא הַמַּרְשִׁ֑יעַ הַמָּשִׁ֜יחַ ה֣וּא ׀ אֲשֶׁ֣ר מֵ֗ת וְאַ֥ף גַּם֨ הוּקַ֔ם אֲשֶׁ֨ר גַּ֥ם הוּא֨ לִימִ֣ין הָֽאֱלֹהִ֔ים אֲשֶׁ֥ר גַּ֖ם מַפְגִּ֥יעַ בַּֽעֲדֵֽנוּ.
8:35	מִ֛י יַפְרִ֥יד אֹתָ֖נוּ מֵאַֽהֲבַ֣ת הַמָּשִׁ֑יחַ הֲצָרָ֣ה ׀ א֣וֹ מְצוּקָ֗ה א֤וֹ רְדִיפָה֨ א֣וֹ רָעָ֔ב א֥וֹ עֵירֹ֛ם א֥וֹ סַכָּנָ֖ה א֥וֹ חָֽרֶב.
8:36	כַּכָּת֑וּב כִּ֗י עָלֶ֨יךָ֨ הֹרַ֣גְנוּ כָל־הַיּ֔וֹם נֶחְשַׁ֖בְנוּ כְּצֹ֥אן טִבְחָֽה.
8:37	אַ֣ךְ בְּכָל־אֵ֔לֶּה מְנַצְּחִ֥ים אֲנַ֖חְנוּ לְמוֹתָ֑ר עַל־יְדֵ֥י זֶ֖ה הָֽאֹהֵ֥ב אֹתָֽנוּ.
8:38	כִּ֛י אֲנִ֥י יוֹדֵ֖עַ בֶּ֑טַח כִּ֠י לֹא־מָ֨וֶת וְלֹֽא־חַיִּ֜ים וְלֹ֣א מַלְאָכִ֗ים וְלֹ֤א מֶמְשָׁלוֹת֨ וְלֹ֣א גְבוּר֔וֹת וְלֹ֥א נֽוֹכָחִ֖ים וְלֹ֥א עֲתִידִֽים.
8:39	וְלֹ֥א רוֹם֨ וְלֹ֣א עֹ֔מֶק וְלֹ֖א בְּרִיאָ֣ה אַחֶ֑רֶת יוּכַ֗ל לְהַפְרִ֤יד אֹתָ֙נוּ֨ מֵאַֽהֲבַ֣ת הָֽאֱלֹהִ֔ים אֲשֶׁ֛ר בְּיִשׁ֥וּעַ הַמָּשִׁ֖יחַ אֲדֹנֵֽינוּ.
9:1	אֱמֶ֞ת אָֽנֹכִ֤י אֹמֵר֨ בַּמָּשִׁ֔יחַ אֵינֶ֖נִּי מְשַׁקֵּ֑ר בְּהָעִ֥יד לִ֛י גַּ֥ם רַעְיוֹנִ֖י בְּר֥וּחַ הַקֹּֽדֶשׁ.
9:2	כִּ֗י יֵ֥שׁ לִי֨ יָג֣וֹן גָּד֔וֹל וּמַכְאֹ֥ב בִּלְתִּי־סָ֖ר בְּלִבִּֽי.
9:3	כִּ֤י חָפַ֨צְתִּי֨ אֲנִ֣י בְעַצְמִ֔י לִהְי֥וֹת חֵ֖רֶם מִן־הַמָּשִׁ֑יחַ בְּעַ֣ד אַחַ֔י בְּנֵ֥י מֽוֹלַדְתִּ֖י כְּפִ֥י הַבָּשָֽׂר.
9:4	אֲשֶׁ֥ר הֵ֖ם יִשְׂרְאֵלִ֑ים אֲשֶׁ֨ר לָהֶ֜ם תְּשׂוּמַ֤ת הַבָּנִים֨ וְהַכָּב֣וֹד וְהַבְּרִית֔וֹת וּמַתַּ֧ן הַתּוֹרָ֛ה וְהָֽעֲבוֹדָ֖ה וְהַֽהַבְטָחֽוֹת.
9:5	אֲשֶׁ֤ר לָהֶם֨ הָֽאָב֔וֹת וְֽאֲשֶׁ֥ר מֵהֶ֛ם הַמָּשִׁ֖יחַ לְפִ֣י הַבָּשָׂ֑ר הַֽהֹוֶ֨ה עַל־הַכֹּ֧ל אֱלֹהִ֛ים מְבֹרָ֥ךְ לְעֽוֹלָמִ֖ים אָמֵֽן.
9:6	אַ֚ךְ לֹֽא־יָכֹ֣ל לִהְי֔וֹת כִּ֥י נָפַ֖ל דְּבַ֣ר הָֽאֱלֹהִ֑ים כִּ֣י לֹ֥א כֻלָּ֛ם אֲשֶׁ֥ר מִיִּשְׂרָאֵ֖ל יִשְׂרָאֵ֥ל הֵֽם.
9:7	וְגַם־לֹ֗א יַ֥עַן הֵ֛ם זֶ֥רַע אַבְרָהָ֖ם כֻּלָּ֣ם בָּנִ֑ים כִּ֚י אִם־בְּיִצְחָ֔ק יִקָּרֵ֥א לְךָ֖ זָֽרַע.
9:8	זֶ֣ה ה֗וּא לֹ֚א בְּנֵ֣י הַבָּשָׂ֔ר הֵ֖ם בְּנֵ֣י הָֽאֱלֹהִ֑ים כִּ֛י אִם־בְּנֵ֥י הַֽהַבְטָחָ֖ה נֶחְשָׁבִ֥ים לְזָֽרַע.
9:9	כִּ֛י דְּבַ֥ר הַֽהַבְטָחָ֖ה זֶ֣ה ה֑וּא לַמּוֹעֵ֤ד הַזֶּה֨ אָב֔וֹא וְיִהְיֶ֥ה לְשָׂרָ֖ה בֵּֽן.
9:10	וְלֹ֥א לְבַ֖ד זֹ֑את כִּ֣י ׀ אִם־גַּ֣ם רִבְקָ֗ה בִּהְי֨וֹת לָ֧הּ הֵֽרָי֛וֹן מֵֽאֶחָ֖ד יִצְחָ֥ק אָבִֽינוּ.
9:11	כִּ֞י בְּע֣וֹד ׀ לֹ֣א יֻלְּד֗וּ וּ֠בְעוֹד לֹ֣א פָֽעֲל֥וּ מְא֛וּמָה ט֖וֹב א֣וֹ רָ֑ע לְמַעַן֩ עֲצַ֨ת הָֽאֳלֹהִ֜ים לְפִ֤י הַבְּחִירָה֨ תַּֽעֲמֹ֔ד לֹ֣א מִפְּעָלִ֔ים כִּ֖י אִם־מִן־הַקּוֹרֵֽא.
9:12	נֶֽאֱמַ֣ר לָ֔הּ כִּ֥י רַ֖ב יַֽעֲבֹ֥ד צָעִֽיר.
9:13	כַּכָּת֑וּב וָֽאֹהַב֨ אֶת־יַֽעֲקֹ֔ב וְאֶת־עֵשָׂ֖ו שָׂנֵֽאתִי.
9:14	מָ֤ה אֵפוֹא֨ נֹאמַ֔ר הַֽאִם־עַוְלָ֖ה אֵ֣צֶל הָֽאֱלֹהִ֑ים חָלִֽילָה.
9:15	כִּ֥י אֶל־מֹשֶׁ֖ה ה֣וּא אֹמֵ֑ר וְחַנֹּתִי֨ אֶת־אֲשֶׁ֣ר אָחֹ֔ן וְרִֽחַמְתִּ֖י אֶת־אֲשֶׁ֥ר אֲרַחֵֽם.
9:16	לָכֵ֣ן אֵפ֗וֹא לֹ֤א מִן־הָֽרֹצֶ֔ה גַּ֥ם לֹ֖א מִן־הָרָ֑ץ כִּ֥י אִם־מִן־הָֽאֱלֹהִ֖ים הַֽחֹנֵֽן.
9:17	כִּ֣י הַכְּתָב֘ אֹמֵ֣ר לְפַרְעֹה֒ כִּ֗י בַּֽעֲב֥וּר זֹאת֨ הֶֽעֱמַדְתִּ֔יךָ בַּֽעֲב֛וּר הַרְאוֹתִ֥י בְךָ֖ אֶת־כֹּחִ֑י וּלְמַ֛עַן יְסֻפַּ֥ר שְׁמִ֖י בְּכָל־הָאָֽרֶץ.
9:18	ולָכֵ֣ן אֵפ֔וֹא אֵ֥ת אֲשֶׁ֛ר ה֥וּא רֹצֶ֖ה חֹנֵ֑ן וְאֵ֥ת אֲשֶׁ֛ר ה֥וּא רֹצֶ֖ה מַקְשֶֽׁה.
9:19	וְאִם־תֹּאמַ֤ר אֵפוֹא֨ אֵלַ֔י לָ֛מָּה ה֥וּא מוֹכִ֖יחַ ע֑וֹד כִּ֛י מִ֥י הִתְנֵגֵּ֖ד לִרְצוֹנֽוֹ.
9:20	אָכֵ֞ן הָ֣הּ אֱנ֗וֹשׁ מִ֤י אֵפוֹא֙ אַתָּ֔ה הָֽעֹנֶ֖ה בִּפְנֵ֣י הָֽאֱלֹהִ֑ים הֲיֹ֤אמַר יֵ֨צֶר֨ לְ֣יוֹצְר֔וֹ לָ֥מָּה עֲשִׂיתַ֖נִי כָּֽכָה.
9:21	הַאֵ֥ין שָׁלְטָ֛ן לַיּוֹצֵ֖ר עַל־הַחֹ֑מֶר לַֽעֲשׂ֞וֹת מֵעֶ֣צֶם ׀ הַגֹּ֣לֶם הַה֗וּא כְּלִ֤י אֶחָד֨ לְכָב֔וֹד וְאַחֵ֖ר לְקָלֽוֹן.
9:22	אַ֣ךְ ׀ אִ֣ם הָֽאֱלֹהִ֗ים בְּחָפְצוֹ֨ לְהַרְא֣וֹת אֶת־הֶֽחָר֔וֹן וּלְהוֹדִ֖יעַ אֶת־גְּבֽוּרָת֑וֹ נָשָׂ֞א בְּאֹ֤רֶךְ אַפַּ֨יִם֨ רַ֔ב אֶת־כְּלֵ֥י הֶֽחָר֖וֹן הַמְּתֻכָּנִ֥ים לַֽאֲבַדּֽוֹן.
9:23	וּלְמַ֨עַן יוֹדִ֜יעַ אֶת־עֹ֤שֶׁר כְּבוֹדוֹ֙ עַל־כְּלֵ֣י חֲנִינָ֔ה אֲשֶׁ֛ר הֵכִ֥ין מִקֶּ֖דֶם לְתִפְאָֽרֶת.
9:24	אֹתָ֕נוּ אֲשֶׁ֖ר גַּ֣ם קָרָ֑א לֹ֤א לְבַד֨ מִן־הַיְּהוּדִ֔ים כִּ֖י אִם־גַּ֥ם מִן־הַגּוֹיִֽם.
9:25	כַּֽאֲשֶׁ֗ר גַּ֛ם ה֥וּא אֹמֵ֖ר בְּהוֹשֵׁ֑עַ אֶקְרָ֤א לְלֹֽא־עַמִּי֨ עַמִּ֔י וּלְלֹֽא־רֻחָ֖מָה רֻחָֽמָה.

13:2	לָכֵ֗ן הָֽעֹרֵךְ֨ נֶ֣גֶד הַמֶּמְשָׁלָ֔ה לְמַֽעֲרֶ֥כֶת הָֽאֱלֹהִ֖ים ה֣וּא מִתְנַגֵּ֑ד אַ֚ךְ הַמִּתְנַגְּדִ֔ים יִקְּח֥וּ דִ֖ין לְנַפְשָֽׁם.
13:3	כִּ֣י הַמֹּֽשְׁלִ֗ים אֵינָ֥ם פַּ֨חַד֨ לַמַּֽעֲשִׂ֣ים הַטּוֹבִ֔ים כִּ֖י אִם־אֶל־הָֽרָעִ֑ים אַ֠ךְ אִם־תִּרִצֶ֞ה לְבִלְתִּ֤י פְחֹד֨ מִפְּנֵ֣י הַמֶּמְשָׁלָ֔ה עֲשֵׂ֣ה אֶת־הַטּ֔וֹב וְתִהְיֶ֥ה לְךָ֛ תְּהִלָּ֖ה מִמֶּֽנָּה.
13:4	כִּ֠י מְשָֽׁרְתַ֨ת אֱלֹהִ֥ים הִיא֘ לְךָ֣ לַטּוֹב֒ אַךְ֩ אִם־תַּֽעֲשֶׂ֨ה אֶת־הָרָ֜ע פְּחַ֗ד כִּ֚י לֹ֣א לְחִנָּ֗ם אֶת־הַחֶ֖רֶב הִ֣יא נֹשֵׂ֑את כִּ֣י מְשָֽׁרְתַ֤ת אֱלֹהִים֨ הִ֗יא וְנוֹקֶ֨מֶת֨ לְעֶבְרָ֔ה לַפֹּעֵ֖ל אֶת־הָרָֽע.
13:5	לָכֵ֖ן צָרִ֣יךְ לְהִכָּנֵ֑עַ לֹ֤א לְבַד֨ בַּֽעֲב֣וּר הָֽעֶבְרָ֔ה כִּ֥י אִם־גַּ֖ם בַּֽעֲב֥וּר הָֽרַעְיֽוֹן.
13:6	בַּֽעֲב֣וּר זֹ֔את גַּ֥ם אֶת־הַמַּ֖ס אַתֶּ֣ם מְשַׁלְּמִ֑ים כִּ֣י מְשָֽׁרְתֵ֤י אֱלֹהִים֨ הֵ֔מָּה הַמַּשְׁגִּיחִ֕ים עַל־עֶ֖צֶם הַדָּבָ֥ר הַזֶּֽה.
13:7	לָכֵ֕ן שַׁלְּמ֥וּ לַכֹּ֖ל אֶת־הַֽחֹב֑וֹת לַֽאֲשֶׁר֩ ל֨וֹ הַמַּ֜ס אֶת־הַמַּ֗ס לַֽאֲשֶׁ֨ר ל֤וֹ הַמֶּ֨כֶס֨ אֶת־הַמֶּ֔כֶס לַֽאֲשֶׁ֨ר ל֤וֹ הַיִּרְאָה֨ אֶת־הַיִּרְאָ֔ה לַֽאֲשֶׁ֥ר ל֛וֹ הַכָּב֖וֹד אֶת־הַכָּבֽוֹד.
13:8	אַל־תָּח֤וּבוּ לְאִישׁ֨ מְא֔וּמָה זֽוּלָתִ֕י לְאַֽהֲבָ֖ה אִ֣ישׁ אֶת־רֵעֵ֑הוּ כִּ֚י הָֽאֹהֵ֣ב אֶת־הָרֵ֔עַ מִלֵּ֖א אֶת־הַתּוֹרָֽה.
13:9	כִּ֣י זֹ֔את לֹ֣א תִנְאַ֞ף לֹ֤א תִרְצַח֨ לֹ֣א תִגְנֹ֔ב לֹ֧א תַֽעֲנֶ֛ה עֵ֥ד שֶׁ֖קֶר לֹ֣א תַחְמֹ֑ד וְאִם־יֵ֞שׁ אֵי־ז֣וּ ׀ מִצְוָ֣ה אַחֶ֗רֶת מְכֻלְכָּלָ֨ה הִ֤יא בְקֹ֨צֶר֨ בַּדָּבָ֣ר הַזֶּ֔ה וְאָֽהַבְתָּ֥ לְרֵֽעֲךָ֖ כָּמֽוֹךָ.
13:10	הָאַֽהֲבָ֗ה אֵינֶ֧נָּה פֹעֶ֛לֶת רָ֖ע לָרֵ֑עַ לָכֵ֕ן מְלֹ֥א הַתּוֹרָ֖ה הָאַֽהֲבָֽה.
13:11	וְזֹ֞את בְּדַעְתֵּ֣נוּ אֶת־הָעֵ֗ת כִּ֚י עַתָּ֣ה הַשָּׁעָ֔ה לַֽהֲקִיצֵ֖נוּ מִשֵּׁנָ֑ה כִּ֤י עַתָּה֨ קְרֹבָ֣ה יְשֽׁוּעָתֵ֔נוּ יוֹתֵ֥ר מֵעֵ֖ת אֲשֶׁ֥ר הֶֽאֱמַֽנּוּ.
13:12	הַלַּ֨יְלָה֨ צָלַ֣ח הַרְבֵּ֔ה וְהַיּ֖וֹם קָרֵ֑ב לָכֵ֗ן נָסִ֨ירָה֨ אֶת־פָּֽעֳלֵ֣י הַחֹ֔שֶׁךְ וְנִלְבְּשָׁ֖ה אֶת־נֶ֥שֶׁק הָאֽוֹר.
13:13	וּכְמ֥וֹ בַיּ֖וֹם נִתְהַלְּכָ֣ה בְתָמִ֑ים לֹ֣א בְזֹֽלְל֞וֹת וּבְשִׁכָּרוֹ֗ן לֹ֤א בְמִשְׁכָּבוֹת֨ וּבְזִמּ֔וֹת לֹ֥א בִמְרִיבָ֖ה וּבְקִנְאָֽה.
13:14	כִּ֛י אִם־לִבְשׁ֥וּ אֶת־הָֽאָד֖וֹן יֵשׁ֣וּעַ הַמָּשִׁ֑יחַ וְאֵ֨ל הַשְׁגָּחַ֧ת הַבָּשָׂ֛ר אַל־תַּֽעֲשׂ֖וּ אֶל־הַתַּֽאֲוֽוֹת.
14:1	אֶת־הָֽרָפֶ֥ה בָֽאֱמוּנָ֖ה קַבֵּ֑לוּ אַ֥ךְ לֹ֖א לְהַבְחָנַ֥ת הַדֵּעֽוֹת.
14:2	אָכֵ֕ן הָֽאֶחָ֣ד מַֽאֲמִ֔ין כִּ֥י יֹאכַ֖ל כָּל־דָּבָ֑ר אַ֚ךְ הָֽרָפֶ֔ה יְרָק֖וֹת ה֥וּא אֹכֵֽל.
14:3	הָֽאֹכֵ֗ל אַל־יִבְזֶה֨ אֵ֚ת אֲשֶׁ֣ר אֵינֶ֣נּוּ אֹכֵ֔ל וַֽאֲשֶׁר֨ אֵינֶ֣נּוּ אֹכֵ֔ל אַל־יִשְׁפֹּ֖ט אֵ֣ת הָֽאֹכֵ֑ל כִּ֥י הָֽאֱלֹהִ֖ים קִבֵּ֥ל אֹתֽוֹ.
14:4	אַתָּ֣ה ׀ מִ֣י אַתָּ֗ה הַשֹּׁפֵט֨ עֶ֚בֶד בֵּ֣ית אַחֵ֔ר לַֽאֲדֹנָ֥יו שֶׁלּ֛וֹ ה֥וּא עֹמֵ֖ד א֣וֹ נֹפֵ֑ל אַ֚ךְ יָֽעֳמַ֔ד כִּ֛י יָכֹ֥ל הָֽאֱלֹהִ֖ים לְהַֽעֲמִ֥יד אֹתֽוֹ.
14:5	זֶ֣ה שֹׁפֵ֗ט י֚וֹם מֵעַ֣ל י֔וֹם וְזֶ֥ה שֹׁפֵ֖ט כָּל־י֑וֹם כָּל־אִ֛ישׁ יִמְלָ֥א בִטְחָ֖ה בְּדַעְתּ֥וֹ שֶׁלּֽוֹ.
14:6	הַמַּשְׂכִּ֣יל אֶת־הַיּ֗וֹם לַֽיהוָֹה֨ ה֣וּא מַשְׂכִּ֔יל וַֽאֲשֶׁ֨ר אֵינֶ֤נּוּ מַשְׂכִּיל֨ אֶת־הַיּ֔וֹם לַֽיהוָֹ֖ה אֵינֶ֣נּוּ מַשְׂכִּ֑יל הָֽאֹכֵ֞ל לַֽיהוָֹ֣ה ׀ ה֣וּא אֹכֵ֗ל כִּ֣י ה֤וּא מוֹדֶה֨ לָֽאֱלֹהִ֔ים וַֽאֲשֶׁ֣ר ׀ אֵינֶ֣נּוּ אֹכֵ֗ל לַֽיהוָֹה֨ אֵינֶ֣נּוּ אֹכֵ֔ל וּמוֹדֶ֖ה לָֽאֱלֹהִֽים.
14:7	כִּ֣י אֵ֥ין אִ֛ישׁ מִמֶּ֖נּוּ חַ֣י לְעַצְמ֑וֹ וְאֵ֥ין אִ֖ישׁ מֵ֥ת לְעַצְמֽוֹ.
14:8	כִּ֗י אִ֤ם נִחְיֶה֨ לַֽיהוָֹ֣ה נִחְיֶ֔ה וְאִם־נָמ֖וּת לַֽיהוָֹ֣ה נָמ֑וּת לָכֵ֗ן אִם־נִחְיֶה֨ וְאִ֣ם נָמ֔וּת לַֽיהוָֹ֖ה אֲנָֽחְנוּ.
14:9	כִּ֚י לָזֹ֔את הַמָּשִׁ֗יחַ גַּ֥ם מֵ֛ת גַּ֥ם קָ֖ם וַיֶּ֑חִי לְמַ֣עַן יִמְשֹׁ֔ל גַּ֥ם עַל־הַמֵּתִ֖ים וְגַ֥ם עַל־הַֽחַיִּֽים.
14:10	אַ֣ךְ אַתָּ֗ה מָ֣ה אַתָּ֤ה שֹׁפֵט֙ אֶת־אָחִ֔יךָ א֤וֹ גַּ֣ם אַתָּ֔ה מָ֛ה אַתָּ֥ה בֹזֶ֖ה אֶת־אָחִ֑יךָ כִּ֚י כֻּלָּ֣נוּ נָֽעֳמַ֔ד לִפְנֵ֥י כִסֵּ֖א מִשְׁפַּ֥ט הַמָּשִֽׁיחַ.
14:11	כִּ֣י כָת֔וּב חַ֥י אֲנִ֖י נְאֻ֣ם יְהוָֹ֑ה כִּי־לִי֨ תִּכְרַ֣ע כָּל־בֶּ֔רֶךְ וְכָל־לָשׁ֖וֹן תּוֹדֶ֥ה לָֽאֱלֹהִֽים.
14:12	לָכֵ֣ן אֵפ֔וֹא כָּל־אִ֣ישׁ מִמֶּ֔נּוּ יִתֵּ֧ן עַל־נַפְשׁ֛וֹ חֶשְׁבּ֖וֹן לָֽאֱלֹהִֽים.
14:13	לָכֵ֕ן לֹֽא־נִשְׁפְּטָ֥ה ע֖וֹד אִ֣ישׁ אֶת־רֵעֵ֑הוּ כִּ֤י אִם־אֶת־זֹאת֨ שִׁפְט֣וּ בְיוֹתֵ֔ר לְבִלְתִּ֨י שׂ֧וּם דְּבַר־נֶ֛גֶף לָאָ֖ח א֥וֹ מִכְשֽׁוֹל.
14:14	אֲנִ֣י יוֹדֵ֗עַ גּ֣ם נוֹכָ֤ח אֲנִי֨ בָּֽאָד֣וֹן יֵשׁ֔וּעַ כִּ֣י אֵ֥ין מְא֛וּמָה חֹ֖ל בְּעַצְמ֑וֹ זֽוּלָתִ֞י לַמְּחַשֵּׁ֤ב מְא֨וּמָה֨ לִהְי֣וֹת חֹ֔ל לָזֶ֖ה חֹ֥ל הֽוּא.
14:15	אַ֗ךְ אִם־בִּגְלַ֥ל אֹ֨כֶל֨ נֶעְצָ֣ב אָחִ֔יךָ לֹ֥א ע֛וֹד כְּפִ֥י אַֽהֲבָ֖ה אַתָּ֣ה מִתְהַלֵּ֑ךְ בְּאָכְלְךָ֨ אַל־תְּאַבֵּ֣ד אֹת֔וֹ אֲשֶׁ֥ר בַּֽעֲד֖וֹ הַמָּשִׁ֥יחַ מֵֽת.
14:16	לָכֵ֖ן אַל־יְגֻדַּ֥ף טֽוּבְכֶֽם.
14:17	כִּ֗י אֵ֛ין מַלְכ֥וּת הָֽאֱלֹהִ֖ים אֹ֣כֶל וּמַשְׁקֶ֑ה כִּ֣י אִם־צְדָקָ֧ה וְשָׁל֛וֹם וְשִׂמְחָ֖ה בְּר֥וּחַ הַקֹּֽדֶשׁ.
14:18	כִּ֗י הָֽעֹבֵ֤ד אֶת־הַמָּשִׁ֨יחַ֨ בָּאֵ֔לֶּה רָצ֥וּי ה֖וּא לָֽאֱלֹהִ֑ים וּבָח֖וּן לָֽאֲנָשִֽׁים.
14:19	לָכֵ֣ן אֵפ֔וֹא נִרְדְּפָ֖ה אֶת־דִּבְרֵ֣י הַשָּׁל֑וֹם וְאֵת֨ דִּבְרֵ֣י הַֽהִבָּנ֔וּת אֲשֶׁ֛ר בֵּ֥ין אִ֖ישׁ לְרֵעֵֽהוּ.
14:20	בִּגְלַ֣ל אֹ֔כֶל אַל־תַּשְׁחִ֖ית אֶת־פֹּ֣עַל הָֽאֱלֹהִ֑ים אָמְנָ֗ם כָּל־דְּבָרִים֨ טְהוֹרִ֔ים אַ֛ךְ רָעָ֥ה לָאִ֖ישׁ הָֽאֹכֵ֥ל בְּמִכְשֹֽׁל.
14:21	ט֗וֹב לְבִלְתִּי֨ אֲכֹ֣ל בָּשָׂ֔ר גַּם־לֹ֖א לִשְׁתּ֣וֹת יָ֑יִן וְגַם־לֹ֞א אֵ֣ת אֲשֶׁ֤ר אָחִ֨יךָ֨ מִתְנַגֵּ֣ף בּ֔וֹ א֥וֹ נִכְשָׁ֖ל א֥וֹ נֶֽחֳלֶֽה.
14:22	אַתָּ֗ה אִם־יֵ֤שׁ לְךָ֨ אֱמוּנָ֔ה אֲחֹ֥ז אֹתָ֛הּ לְעַצְמְךָ֖ לִפְנֵ֣י הָֽאֱלֹהִ֑ים אַשְׁרֵ֜הוּ אֲשֶׁ֤ר אֵינֶ֨נּוּ֨ שֹׁפֵ֣ט אֶת־עַצְמ֔וֹ בַּֽאֲשֶׁ֖ר ה֥וּא בֹחֵֽן.
14:23	אַ֣ךְ הַמְּסַפֵּ֗ק אִם־ה֤וּא אֹכֵל֨ נִרְשָׁ֔ע כִּ֖י לֹ֣א מֵֽאֱמוּנָ֑ה אַ֗ךְ כָּל־אֲשֶׁ֛ר לֹ֥א מֵֽאֱמוּנָ֖ה חֵ֥טְא הֽוּא.
15:1	וַֽאֲנַ֣חְנוּ הַיְּכוֹלִ֗ים חַיָּבִ֤ים אֲנַ֨חְנוּ֨ לִסְבֹּ֔ל אֶת־רִפְיוֹנ֖וֹת בִּלְתִּ֣י יְכוֹלִ֑ים וְלֹ֥א לְהֵֽרָצ֖וֹת לְנַפְשֵֽׁנוּ.
15:2	כִּ֚י כָּל־אֶחָ֣ד מֵֽאִתָּ֔נוּ אֶל־רֵעֵ֥הוּ יֵֽרָצֶ֖ה לַטּ֑וֹב לְהִבָּנֽוּת.
15:3	כִּ֚י גַּ֣ם הַמָּשִׁ֔יחַ לֹ֥א נִרְצָ֖ה לְנַפְשׁ֑וֹ כִּ֚י אִם־כַּכָּת֔וּב וְחֶרְפּ֥וֹת חֽוֹרְפֶ֖יךָ נָֽפְל֥וּ עָלָֽי.
15:4	כִּ֗י כָּל־אֲשֶׁ֤ר נִכְתַּב֨ מִקֶּ֔דֶם לְלִמּוּדֵ֖נוּ נִכְתָּ֑ב לְמַ֗עַן בַּתּוֹחֶ֨לֶת֨ וּבְנֶֽחָמַ֣ת הַכְּתָבִ֔ים תִּהְיֶ֥ה לָ֖נוּ הַתִּקְוָֽה.
15:5	וֵֽאלֹהֵ֣י ׀ הַתּוֹחֶ֣לֶת וְהַנֶּֽחָמָה֩ יִתֵּ֨ן לָכֶ֜ם לְהַשְׂכִּ֗יל אֶת־עֶ֛צֶם הַדָּבָ֥ר הַה֖וּא אִ֣ישׁ לְרֵעֵ֑הוּ כְּפִ֖י יֵשׁ֥וּעַ הַמָּשִֽׁיחַ.

15:7	לָכֵ֕ן קַבְּל֖וּ אִ֣ישׁ אֶת־רֵעֵ֑הוּ כַּֽאֲשֶׁ֧ר גַּם־הַמָּשִׁ֛יחַ קִבֵּ֥ל אֹתָ֖נוּ לִכְב֥וֹד הָֽאֱלֹהִֽים.
15:8	וַֽאֲנִ֣י אֹמֵ֔ר כִּ֣י ׀ יֵשׁ֣וּעַ הַמָּשִׁ֗יחַ נִהְיָה֨ לִמְשָׁרֵת֨ בְּנֵ֣י הַמּוּלָ֔ה בַּֽעֲב֖וּר אֱמֶ֣ת הָֽאֱלֹהִ֑ים לְמַ֥עַן קַיֵּ֖ם אֶת־הַבְטְח֥וֹת הָֽאָבֽוֹת.
15:9	וּלְמַ֗עַן הַגּוֹיִ֛ם יְכַבְּד֥וּ אֶת־הָֽאֱלֹהִ֖ים עַל־רַֽחֲמָ֑יו כַּכָּת֕וּב עַל־כֵּן֨ אֽוֹדְךָ֣ בַגּוֹיִ֔ם וּלְשִׁמְךָ֖ אֲזַמֵּֽרָה.
15:10	וְע֖וֹד ה֣וּא אֹמֵ֑ר הַרְנִ֥ינוּ גוֹיִ֖ם אֵ֥ת עַמּֽוֹ.
15:11	וְע֕וֹד הַלְל֥וּ אֶת־יְהוָֹ֖ה כָּל־הַגּוֹיִ֑ם וְשַׁבְּח֖וּהוּ כָּל־הָֽאֻמִּֽים.
15:12	וְע֕וֹד יְשַׁעְיָ֖הוּ אֹמֵ֑ר וְהָיָ֣ה ׀ שֹׁ֣רֶשׁ יִשַׁ֗י וְהַקָּם֨ לִמְשֹׁ֣ל בָּֽעַמִּ֔ים אֵלָ֖יו גּוֹיִ֥ם יְקַוּֽוּ.
15:13	וֵֽאלֹהֵ֣י הַתִּקְוָ֗ה יְמַלֵּ֥א אֶתְכֶ֛ם כָּל־שִׂמְחָ֥ה וְשָׁל֖וֹם בְּהַֽאֲמִ֑ין לְמַ֤עַן תַּעְדִּ֨יפוּ֨ בַּתִּקְוָ֔ה בִּגְבוּרַ֖ת ר֥וּחַ הַקֹּֽדֶשׁ.
15:14	וְגַם֩ אֲנִ֨י עַצְמִ֜י יוֹדֵ֨עַ בֶּ֤טַח ׀ אַחַי֨ עַל־אֹדֽוֹתֵיכֶ֔ם כִּ֗י גַּ֣ם אַתֶּ֧ם עַצְמְכֶ֛ם מְלֵ֥אֵי ט֖וֹב נִמְלָאִ֣ים כָּל־דָּ֑עַת וִֽיכוֹלִ֥ים לְהַזְהִ֖יר אִ֥ישׁ אֶת־רֵעֵֽהוּ.
15:15	אַ֣ךְ ׀ בְּיֶ֣תֶר בִּטְחָ֗ה כָּתַ֨בְתִּי אֲלֵיכֶ֥ם אַחַ֛י לְמִקְצָ֖ת כְּמַזְכִּ֣יר אֶתְכֶ֑ם בַּֽעֲב֣וּר הַחֵ֔ן הַנִּתָּ֥ן לִ֖י מֵאֵ֥ת הָֽאֱלֹהִֽים.
15:16	לְמַ֨עַן הֱיוֹתִ֜י מְשָׁרֵ֨ת יֵשׁ֤וּעַ הַמָּשִׁ֨יחַ֨ לַגּוֹיִ֔ם מְכַהֵ֖ן אֶת־בְּשׂוֹרַ֣ת הָֽאֱלֹהִ֑ים לְמַעַן֩ יִהְיֶ֨ה קָרְבַּ֤ן הַגּוֹיִם֨ נִרְצֶ֗ה מְקֻדָּ֖שׁ בְּר֥וּחַ הַקֹּֽדֶשׁ.
15:17	לָכֵ֕ן יֵֽשׁ־לִ֥י תְהִלָּ֖ה בְּיֵשׁ֣וּעַ הַמָּשִׁ֑יחַ בַּדְּבָרִ֖ים אֲשֶׁ֥ר לֵֽאלֹהִֽים.
15:18	כִּי֩ לֹ֨א אֶתְעָרֵ֜ב לְדַבֵּ֣ר ׀ מְא֣וּמָה מִן־הַדְּבָרִ֗ים אֲשֶׁ֨ר לֹֽא־פָעַ֧ל הַמָּשִׁ֛יחַ עַל־יָדִ֖י לְמִשְׁמַ֣עַת הַגּוֹיִ֑ם בְּדָבָ֖ר וּבְפֹֽעַל.
15:19	בִּגְבוּרַ֤ת אוֹתוֹת֨ וּמֽוֹפְתִ֔ים בִּגְבוּרַ֖ת ר֣וּחַ הָֽאֱלֹהִ֑ים עַ֠ד אֲשֶׁ֨ר מִירֽוּשָׁלַ֤םִ וּסְבִֽיבוֹתֶ֨יהָ֨ עַד־אִלּֽוּרִק֔וֹן מִלֵּ֖אתִי אֶת־בְּשׂוֹרַ֥ת הַמָּשִֽׁיחַ.
15:20	אַ֛ךְ כָּ֖כָה הִתְאַמַּ֣צְתִּי לְבַשֵּׂ֑ר לֹ֗א בַּֽאֲשֶׁ֤ר נִקְרָא֨ שֵׁ֣ם הַמָּשִׁ֔יחַ פֶּן־אֶבְנֶ֕ה עַל־יְס֖וֹד אִ֥ישׁ אַחֵֽר.
15:21	כִּ֖י אִם־כַּכָּת֑וּב אֲ֠שֶׁר לֹֽא־סֻפַּ֨ר לָהֶ֤ם מִמֶּ֨נּוּ֨ יִרְא֔וּ וַֽאֲשֶׁ֛ר לֹ֥א שָֽׁמְע֖וּ יִתְבּוֹנָֽנוּ.
15:22	עַל־כֵּ֕ן גַּ֥ם נִמְנַ֛עְתִּי פְּעָמִ֥ים רַבּ֖וֹת מִבּ֥וֹא אֲלֵיכֶֽם.
15:23	אַ֣ךְ עַתָּ֗ה בְּאֵ֨ין לִ֥י ע֛וֹד מָק֖וֹם בַּגְּלִיל֣וֹת הָאֵ֑לֶּה וּבִהְי֨וֹת לִ֜י תְּשׁוּקָ֧ה גְדוֹלָ֛ה לָב֥וֹא אֲלֵיכֶ֖ם מִשָּׁנִ֥ים רַבּֽוֹת.
15:24	לָכֵ֗ן כַּֽאֲשֶׁ֛ר אֶסַּ֥ע לִסְפָרָ֖ד אָב֣וֹא אֲלֵיכֶ֑ם כִּ֡י אֲנִי֩ מְקַוֶּ֨ה לִרְאֽוֹתְכֶ֜ם בְּעָבְרִ֗י וּלְלֻוּ֤וֹת מֵֽאִתְּכֶם֨ שָׁ֔מָּה אִ֧ם בָּרִֽאשֹׁנָ֛ה שָׂבַ֥עְתִּי אֶתְכֶ֖ם לְמִקְצָֽת.
15:25	אַ֣ךְ עַתָּ֔ה אֲנִ֥י נוֹסֵ֖עַ יְרֽוּשָׁלָ֑מֳה לְשָׁרֵ֖ת אֶת־הַקְּדֹשִֽׁים.
15:26	כִּ֤י רָאוּ֨ מַקֶּדֽוֹנִיָּ֣א וְאַכַיָּ֔א לַֽעֲשׂ֞וֹת קְבוּצַ֤ת נְדָבוֹת֨ אַחַ֔ת לַֽעֲנִיֵּ֥י הַקְּדוֹשִׁ֖ים אֲשֶׁ֥ר בִּירֽוּשָׁלָֽםִ.
15:27	אָמְנָ֣ם רָצ֔וּ וְחַיְּבֵיהֶ֖ם הֵ֑מָּה כִּ֡י אִ֣ם הַגּוֹיִם֩ נִהְי֨וּ לֹֽקְחֵ֥י חֵ֨לֶק֙ בְּרֽוּחֲנִיּוֹתָ֔ם מְחֻיָּבִ֥ים הֵ֛ם לְשָׁרֵ֥ת אֹתָ֖ם גַּ֥ם בַּגּֽוּפִיּֽוֹת.
15:28	לָכֵן֨ אִם־כִּלִּ֣יתִי זֹ֔את וְחָתַ֥מְתִּי לָהֶ֖ם אֶת־הַפְּרִ֣י הַזֶּ֑ה אֶסַּ֥ע בְּתֽוֹכְכֶ֖ם סְפָרָֽדָה.
15:29	וְיוֹדֵ֣עַ אֲנִ֗י כִּ֚י בְּבוֹאִ֣י אֲלֵיכֶ֔ם בִּמְל֥וֹא בִרְכַּ֛ת בְּשׂוֹרַ֥ת הַמָּשִׁ֖יחַ אָבֽוֹא.
15:30	אַ֣ךְ אֲנִי֩ מַזְהִ֨יר אֶתְכֶ֜ם אַחַ֗י עַל־יְדֵ֤י אֲדֹנֵ֨ינוּ֨ יֵשׁ֣וּעַ הַמָּשִׁ֔יחַ וְעַל־יְדֵ֖י אַֽהֲבַ֣ת הָר֑וּחַ לְהִתְאַמֵּ֥ץ עִמִּ֛י בַּתְּפִלּ֥וֹת בַּֽעֲדִ֖י אֶל־הָֽאֱלֹהִֽים.
15:31	לְמַ֣עַן אֶנָּצֵ֗ל מֵאֵ֛לֶּה אֲשֶׁ֥ר בִּלְתִּי־מַֽאֲמִינִ֖ים בִּֽיהוּדָ֑ה וְשֵֽׁרוּתִי֨ אֲשֶׁ֣ר לִירֽוּשָׁלַ֔םִ יִהְיֶ֥ה רָצ֖וּי אֶל־הַקְּדוֹשִֽׁים.
15:32	לְמַ֨עַן אָב֤וֹא אֲלֵיכֶם֨ בְּשִׂמְחָ֔ה עַל־יְדֵ֖י רְצ֣וֹן אֱלֹהִ֑ים וְאֶנָּפֵ֖שׁ עִמָּכֶֽם.
15:33	וֵֽאלֹהֵ֧י הַשָּׁל֛וֹם יִהְיֶ֥ה עִם־כֻּלְּכֶ֖ם אָמֵֽן.
16:1	אֲנִי֨ מְהַלֵּ֣ל אֲלֵיכֶ֔ם אֶת־פּוֹיְבֶּ֖א אֲחוֹתֵ֑נוּ אֲשֶׁ֥ר הִ֛יא מְשָׁרֶ֥תֶת הַקְּהִלָּ֖ה אֲשֶׁ֥ר בְּקֶנְכְּרֶאָֽה.
16:2	לְ֠מַעַן תְּקַבְּל֨וּ אֹתָ֤הּ בָּֽאָדוֹן֨ כְּשֹׁוֶ֣ה לַקְּדוֹשִׁ֔ים וְתַֽעַזְר֥וּ לָהּ֨ בְּכָל־חֵ֔פֶץ אֲשֶׁ֖ר תִּצְרַ֣ךְ לָכֶ֑ם כִּ֣י הִ֗יא נִהְיְתָ֥ה עֹזֶ֛רֶת לְרַבִּ֖ים וְגַ֥ם לִֽי.
16:3	שַֽׁאֲל֤וּ לִשְׁלוֹם֨ פְּרִיסְקִלָּ֣א וַֽאֲקוּלָ֔א חַבְרֵ֥י עֲבֽוֹדָתִ֖י בַּמָּשִׁ֥יחַ יֵשֽׁוּעַ.
16:4	אֲשֶׁ֛ר בְּעַ֥ד נַפְשִׁ֖י שָׂ֣מוּ אֶת־צַוְּארֵיהֶ֑ם אֲשֶׁ֣ר לָהֶ֗ם לֹ֣א אָֽנֹכִ֤י לְבַדִּי֨ מוֹדֶ֔ה כִּ֥י אִם־גַּ֖ם כָּל־קְהִלּ֥וֹת הַגּוֹיִֽם.
16:5	וְגַ֛ם לִשְׁל֥וֹם הַקְּהִלָּ֖ה אֲשֶׁ֣ר בְּבֵיתָ֑ם שַֽׁאֲל֗וּ לִשְׁלוֹם֨ אֶפַּיְנֵט֣וֹס אֲהוּבִ֔י אֲשֶׁ֥ר יֶשְׁנ֛וֹ רֵאשִׁ֥ית אַכַיָּ֖א לַמָּשִֽׁיחַ.
16:6	שַֽׁאֲלוּ֨ לִשְׁל֣וֹם מִרְיָ֔ם אֲשֶׁ֛ר רַבּ֖וֹת עָ֥מְלָה לָֽנוּ.
16:7	שַֽׁאֲל֗וּ לִשְׁלוֹם֨ אַנְדְּרֹֽנִק֣וֹס וְיֽוּנִיָּ֔א בְּנֵ֥י מִשְׁפַּחְתִּ֖י וְחַבְרֵ֣י מַֽאֲסָרִ֑י אֲשֶׁ֨ר הֵ֤ם מְצֻיָּנִים֨ בַּשְּׁלִיחִ֔ים וַֽאֲשֶׁ֥ר גַּ֛ם לְפָנַ֥י הָי֖וּ בַּמָּשִֽׁיחַ.
16:8	שַֽׁאֲלוּ֙ לִשְׁל֣וֹם אַמְפְּלִיאַ֔ס אֲהוּבִ֖י בָּֽאָדֽוֹן.
16:9	שַֽׁאֲלוּ֙ לִשְׁל֣וֹם אֽוּרְבָּנ֔וֹס חֲבֵ֥ר עֲבֽוֹדָתֵ֖נוּ בַּמָּשִׁ֑יחַ וְלִשְׁל֖וֹם סְטָכ֥וּס אֲהוּבִֽי.
16:10	שַֽׁאֲלוּ֙ לִשְׁל֣וֹם אַפֶּלִּ֔יס הַנִּבְחָ֖ן בַּמָּשִׁ֑יחַ שַֽׁאֲלוּ֨ לִשְׁל֣וֹם אֵ֔לֶּה אֲשֶׁ֖ר מֵֽאַנְשֵׁ֥י אֲרִסְתָּבּוּלֽוֹס.
16:11	שַֽׁאֲל֧וּ לִשְׁל֛וֹם הוֹרֽוֹדִיּ֖וֹן בֶּן־מִשְׁפַּחְתִּ֑י שַֽׁאֲל֣וּ לִשְׁל֗וֹם אֵ֚לֶּה מִבֵּ֣ית נַרְקִסּ֔וֹס אֲשֶׁ֥ר הֵ֖ם בָּֽאָדֽוֹן.
16:12	שַֽׁאֲל֗וּ לִשְׁלוֹם֨ טְרֽוּפַיְנָ֣ה וּטְרֽוּפֹסָ֔ה הַמִּתְעַמְּל֖וֹת בָּֽאָד֑וֹן שַֽׁאֲל֗וּ לִשְׁלוֹם֨ פֶּרְסִ֣יס הָֽאֲהוּבָ֔ה אֲשֶׁ֛ר עָֽמְלָ֥ה רַבּ֖וֹת בָּֽאָדֽוֹן.
16:13	שַֽׁאֲלוּ֙ לִשְׁל֣וֹם רוּפ֔וֹס הַנִּבְחָ֖ר בָּֽאָד֑וֹן וְלִשְׁל֛וֹם אִמּ֥וֹ שֶׁלּ֖וֹ גַּ֥ם שֶׁלִּֽי.
16:14	שַֽׁאֲל֗וּ לִשְׁל֨וֹם אָֽסֻנְקְרִט֧וֹס פְּלֶג֛וֹן הֶרְמַ֥ס פַּטְרֹבַּ֖ס הֶרְמֶ֑ס וְלִשְׁל֥וֹם הָֽאַחִ֖ים אֲשֶׁ֥ר עִמָּהֶֽם.
16:15	שַֽׁאֲל֗וּ לִשְׁל֨וֹם פִּילֽוֹלוֹג֧וֹס וְיֽוּלִיָּ֛ה נִירִ֥יס וַֽאֲחוֹת֖וֹ וְאֽוֹלֻמְפַּ֑ס וְלִשְׁל֥וֹם כָּל־הַקְּדוֹשִׁ֖ים אֲשֶׁ֥ר עִמָּהֶֽם.
16:16	שַֽׁאֲל֗וּ אִ֛ישׁ לִשְׁל֥וֹם רֵעֵה֖וּ בִּנְשִׁיקָ֣ה קְדוֹשָׁ֑ה כָּל־קְהִלּ֣וֹת הַמָּשִׁ֔יחַ שֹֽׁאֲל֖וֹת לִשְׁלֹֽמְכֶֽם.
16:17	אַ֣ךְ אֲנִי֩ מַזְהִ֨יר אֶתְכֶ֜ם אַחַ֗י לִצְפּ֞וֹת אֶת־הָֽעֹשִׂ֤ים הַמַּֽחֲלֹקוֹת֙ וְהַמִּכְשֹׁלִ֔ים מִח֣וּץ לַלֶּ֔קַח אֲשֶׁ֥ר אַתֶּ֖ם לְמַדְתֶּ֑ם וּנְט֖וּ מֵֽעֲלֵיהֶֽם.
16:18	כִּ֣י ׀ הָֽאֲנָשִׁ֣ים כָּאֵ֗לֶּה אֵינָ֨ם עֹֽבְדִ֜ים אֶת־אֲדֹנֵ֛ינוּ יֵשׁ֥וּעַ הַמָּשִׁ֖יחַ כִּ֣י אִם־אֶת־בִּטְנָ֑ם וּבְשִׂ֤יחַת נֹ֨עַם וּבְאִמְרֵ֣י בְרָכָ֔ה הֵ֥ם מַתְעִ֖ים אֶת־לִבּ֥וֹת הַפְּתָאִֽים.
16:19	כִּ֥י מִשְׁמַעְתְּכֶ֖ם הִגִּ֣יעָה אֶל־הַכֹּ֑ל לָכֵ֞ן שָׂמֵ֤חַ אֲנִי֨ בַּֽעֲבֽוּרְכֶ֔ם אַ֣ךְ ׀ חָפֵ֣ץ אֲנִ֗י לִהְיֽוֹתְכֶ֛ם חֲכָמִ֥ים לַטּ֖וֹב וּפְתָאִ֥ים לָרָֽע.
16:20	וֵֽאלֹהֵ֣י הַשָּׁל֗וֹם יְדַכֶּ֧ה אֶת־הַשָּׂטָ֛ן תַּ֥חַת רַגְלֵיכֶ֖ם בִּמְהֵרָ֑ה וְחֶסֶד֩ אֲדֹנֵ֨ינוּ יֵשׁ֧וּעַ הַמָּשִׁ֛יחַ עִמָּכֶ֖ם אָמֵֽן.
16:21	שֹֽׁאֲלִ֨ים לִשְׁלֹֽמְכֶ֜ם טִמֽוֹתִיּ֣וֹס ׀ חֲבֵ֣ר עֲבֽוֹדָתִ֗י וְלֽוּקִיּ֧וֹס וְיָס֛וֹן וְסֽוֹסִפַּטָ֖ר בְּנֵ֥י מִשְׁפַּחְתִּֽי.
16:22	אֲנִ֤י טֶרְטִיּוֹס֨ הַכֹּתֵ֣ב אֶת־הָֽאִגֶּ֔רֶת שֹׁאֵ֥ל לִשְׁלֹֽמְכֶ֖ם בָּֽאָדֽוֹן.
16:23	שֹׁאֵ֤ל לִשְׁלֹֽמְכֶם֨ גַּיּ֔וֹס מְאָֽרְחִ֖י וּמְאָרֵ֣חַ כָּל־הַקְּהִלָּ֑ה שֹׁאֵ֣ל לִשְׁלֹֽמְכֶ֗ם אֶרַסְט֛וֹס סֹכֵ֥ן הָעִ֖יר וּקְוַרְט֥וֹס הָאָֽח.
16:24	חֶסֶד֩ אֲדֹנֵ֨ינוּ יֵשׁ֧וּעַ הַמָּשִׁ֛יחַ עִם־כֻּלְּכֶ֖ם אָמֵֽן.
16:25	וְל֗וֹ הַיָּכוֹל֨ לְחַזֵּ֣ק אֶתְכֶ֔ם כְּפִ֣י בְשֽׂוֹרָתִ֔י וּכְפִ֥י קְרִיאַ֖ת יֵשׁ֣וּעַ הַמָּשִׁ֑יחַ כְּפִי֨ הִתְגַּלּ֣וּת הַסּ֔וֹד הַנִּסְתָּ֖ר מֵֽעִתּ֥וֹת עוֹלָֽם.
16:26	אַ֚ךְ הַמְּגֻלֶּ֣ה עַתָּ֔ה עַל־יְדֵ֖י כִּתְבֵ֣י הַנְּבִיאִ֑ים וּכְפִ֤י פְקֻדַּת֨ אֱלֹהֵ֣י הָֽעוֹלָ֔ם מוּדָ֥ע לְכָל־הַגּוֹיִ֖ם לְמִשְׁמַ֥עַת הָֽאֱמוּנָֽה.
16:27	לָֽאֱלֹהִים֨ הֶֽחָכָ֣ם לְבַדּ֔וֹ ל֣וֹ הַכָּב֗וֹד בְּיֵשׁ֧וּעַ הַמָּשִׁ֛יחַ לְעֽוֹלָמִ֖ים אָמֵֽן.
