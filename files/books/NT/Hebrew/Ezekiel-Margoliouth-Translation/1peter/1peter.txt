‏I Peter 
1:1	פֶּטְר֕וֹס שְׁלִ֖יחַ יֵשׁ֣וּעַ הַמָּשִׁ֑יחַ אֶל־הַבְּחִירִ֡ים תּֽוֹשְׁבֵי֩ תְפוּצַ֨ת פָּנְט֧וֹס ׀ גָּלָֽטִיָּ֛ה קַפָּדֽוֹקִיָּ֥א ׀ אַסְיָ֖א וּבִיתֽוּנִיָּֽא.
1:2	כְּפִי֩ הַיְדִיעָ֨ה מִקֶּ֜דֶם אֲשֶׂ֨ר לָֽאֱלֹהִ֤ים הָאָב֨ בְּקִדּ֣וּשׁ הָר֔וּחַ לְמִשְׁמַ֕עַת וְלִזְרִיקַ֥ת דַּ֖ם יֵשׁ֣וּעַ הַמָּשִׁ֑יחַ חֵ֥ן לָכֶ֛ם וְשָׁל֖וֹם לְמַרְבֶּֽה.
1:3	בָּר֣וּךְ הָֽאֱלֹהִ֔ים אֲבִ֥י אֲדֹנֵ֖ינוּ יֵשׁ֣וּעַ הַמָּשִׁ֑יחַ אֲשֶׁ֡ר כְּפִי֩ רַֽחֲמָ֨יו הָֽרַבִּ֜ים הוֹלִ֨יד אֹתָ֤נוּ מֵֽחָדָשׁ֨ לְתִקְוָ֣ה חַיָּ֔ה בִּתְקוּמַ֛ת יֵשׁ֥וּעַ הַמָּשִׁ֖יחַ מֵעִ֥ם הַמֵּתִֽים.
1:4	לְנַֽחֲלָ֗ה בִּלְתִּ֧י נִשְׁחָתָ֛ה וּבִלְתִּ֥י נִכְתָּמָ֖ה וּבִלְתִּ֣י נֹבָ֑לֶת הַשְּׁמוּרָ֥ה בַשָּׁמַ֖יִם לָכֶֽם.
1:5	הַנְּצוּרִ֛ים בִּגְבוּרַ֥ת אֱלֹהִ֖ים עַל־יְדֵ֣י הָֽאֱמוּנָ֑ה אֶל־הַתְּשׁוּעָ֕ה הַנְכוֹנָ֥ה לְהִגָּל֖וֹת בָּעֵ֥ת הָאַֽחֲרֽוֹן.
1:6	אֲשֶׁ֥ר בּ֖וֹ אַתֶּ֣ם גָּלִ֑ים בִּהְיֽוֹתְכֶ֣ם עַתָּ֗ה אִ֚ם צָרִ֣יךְ ה֔וּא נֶֽעֱצָבִ֥ים לְמִצְעָ֖ר בְּמַסּ֥וֹת שֹׁנֽוֹת.
1:7	לְמַ֡עַן אֲשֶׁר֩ בֹּ֨חַן אֱמֽוּנַתְכֶ֜ם הַיְקָרָ֤ה יוֹתֵר֨ מִזָּהָ֣ב הַנִּשְׁחָ֔ת גַּ֥ם אִם־נִבְחַ֖ן בָּאֵ֑שׁ תִּמָּצֵ֗א לִתְהִלָּ֤ה וּלְכָבוֹד֨ וּלְתִפְאֶ֔רֶת בְּהִתְגַּלּ֖וּת יֵשׁ֥וּעַ הַמָּשִֽׁיחַ.
1:8	אֲ֠שֶׁר בִּבְלִ֧י רְאֽוֹתְכֶ֛ם אֹת֖וֹ אַתֶּ֣ם אֹֽהֲבִ֑ים אֲשֶׁ֣ר בּ֡וֹ בִּבְלִי֩ חֲזֽוֹתְכֶ֨ם אֹת֤וֹ עַתָּה֨ רַ֣ק בְּהַֽאֲמִֽינְכֶ֔ם אַתֶּ֣ם גָּלִ֗ים בְּשִׂמְחָ֛ה עֲצוּמָ֥ה מִדַּבֵּ֖ר וַֽהֲדוּרָֽה.
1:9	בְּשֵֽׂאתְכֶ֖ם אֶת־אַֽחֲרִ֣ית אֱמֽוּנַתְכֶ֑ם אֵ֖ת תְּשׁוּעַ֥ת נַפְשֽׁוֹתֵיכֶֽם.
1:10	אֲשֶׁ֞ר עַל־אֹד֣וֹת ׀ הַתְּשׁוּעַ֣ה הַזֹּ֗את דָּֽרְשׁ֧וּ וְחָֽקְר֛וּ הֵיטֵ֖ב הַנְּבִיאִ֑ים אֲשֶׁ֥ר נִבְּא֖וּ עַל־הַחֵ֥ן אֲלֵיכֶֽם.
1:11	בְּחָקְרָ֗ם מָ֚ה וְאֵי־ז֣וּ עֵ֔ת חִוָּ֛ה ר֥וּחַ הַמָּשִׁ֖יחַ אֲשֶׁ֣ר בָּהֶ֑ם בְּהָעִ֣יד מִקֶּ֗דֶם אֶת־הָֽעִנּוּיִים֨ אֲשֶׁ֣ר אֶל־הַמָּשִׁ֔יחַ וְעַ֥ל הַתִּפְאָר֖וֹת אֲשֶׁ֥ר אַֽחֲרֵיהֶֽם.
1:12	אֲשֶׁר֘ נִגְלָ֣ה לָהֶם֒ כִּ֣י ׀ לֹ֣א לְעַצְמָ֗ם כִּ֣י אִם־לָ֡נוּ שֵֽׁרְת֣וּ אֶת־הַדְּבָרִ֣ם הָ֠הֵם אֲשֶׁ֤ר עַתָּה֨ הֻגְּד֣וּ לָכֶ֔ם עַל־יְדֵי֨ הַמְּבַשְּׂרִ֣ים אֶתְכֶ֔ם בְּר֥וּחַ הַקֹּ֖דֶשׁ הַשָּׁל֣וּחַ מִן־הַשָּׁמָ֑יִם אֲשֶׁ֣ר אֲלֵיהֶ֔ם מַלְאָכִ֖ים נִכְסָפִ֥ים לְהַשְׁקִֽיף.
1:13	לָכֵ֕ן חִגְר֖וּ מָתְנֵ֣י רֽוּחֲכֶ֑ם וּבְשָׁקְדְכֶ֕ם קַוּ֣וּ בִשְׁלֵמ֗וּת עַל־הַחֵן֨ הֶֽעָתִיד֨ לְהוּבָ֣א אֲלֵיכֶ֔ם בְּהִתְגַּלּ֖וּת יֵשׁ֥וּעַ הַמָּשִֽׁיחַ.
1:14	כְּמ֖וֹ בְּנֵ֣י מִשְׁמָ֑עַת בִּלְתִּ֨י מִתְדַּמִּ֧ים בְּתַבְנִ֛ית לַתַּֽאֲוֹ֥ת הַקּֽוֹדְמ֖וֹת בְּבַֽעֲרֽוּתְכֶֽם.
1:15	כִּ֧י אִם־כְּמ֛וֹ הַקֹּרֵ֥א אֶתכֶ֖ם קָד֣וֹשׁ ה֑וּא כֵּ֗ן הֱי֨וּ גַ֥ם אַתֶּ֛ם קְדוֹשִׁ֖ים בְּכָל־הִתְהַלְּכֽוּת.
1:16	יַ֖עַן כָּת֑וּב וִהְיִיתֶם֨ קְדֹשִׁ֔ים כִּ֥י קָד֖וֹשׁ אָֽנִי.
1:17	וְאִ֣ם ׀ תִּקְרְא֣וּ לְאָ֗ב אֵ֤ת הַשֹּׁפֵט֨ בִּבְלִי֨ מַשֹּׂ֣א פָנִ֔ים כְּפִ֖י פֹּ֣עַל כָּל־אִ֑ישׁ הִתְהַלְּכ֥וּ בְיִרְאָ֖ה כָּל־עֵ֥ת מְגֽוּרֵיכֶֽם.
1:18	בְּדַעְתְּכֶ֗ם כִּ֠י לֹ֣א בִדְבָרִ֤ים נִשְׁחָתִים֨ בְּכֶ֣סֶף וּבְזָהָ֔ב נִפְדֵּיתֶ֖ם מֵֽהִתְהַלְכֽוּתְכֶ֣ם הָֽרֵקָ֑ה הַמְּסוּרָ֥ה לָכֶ֖ם מִן־הָֽאָבֽוֹת.
1:19	כִּ֛י אִם־בְּדַ֥ם הַמָּשִׁ֖יחַ הַיָּקָ֑ר כְּדַ֥ם שֶׂ֛ה בְּלִ֥י מ֖וּם וּבְלִ֥י כָֽתֶם.
1:20	אֲשֶׁ֤ר אָמְנָם֨ נוֹדַ֣ע מִקֶּ֔דֶם לִפְנֵ֖י הִוָּסֵ֣ד הַתֵּבֵ֑ל אַ֥ךְ נִגְלָ֛ה בְּאַֽחֲרִ֥ית הָֽעִתּ֖וֹת בַּֽעֲבֽוּרְכֶֽם.
1:21	אֲשֶׁ֣ר עַל־יָדוֹ֩ מַֽאֲמִינִ֨ים אַתֶּ֜ם בֵּֽאלֹהִ֗ים הַמֵּקִ֤ים אֹתוֹ֨ מֵעִ֣ם הַמֵּתִ֔ים וְהַנֹּתֵ֥ן ל֖וֹ כָּב֑וֹד לְמַ֧עַן תִּהְיֶ֛ה אֱמֽוּנַתְכֶ֥ם וְתִקְוַתְכֶ֖ם בֵּֽאלֹהִֽים.
1:22	וְיַעַן֩ קִדַּשְׁתֶּ֨ם אֶת־נַפְשֽׁוֹתֵיכֶ֜ם בְּמִשְׁמַ֤עַת הָֽאֱמֶת֨ עַל־יְדֵ֣י הָר֔וּחַ לְאַֽהֲבַ֥ת אַחִ֖ים בִּלְתִּ֣י מְחֻנָּ֑פֶת אֶֽהֱב֧וּ בְחָזְקָ֛ה אִ֥ישׁ אֶת־רֵעֵ֖הוּ מִלֵּ֥ב טָהֽוֹר.
1:23	בִּהְיֽוֹתְכֶ֞ם נֽוֹלָדִ֣ים מֵחָדָ֗שׁ לֹ֚א מִזֶּ֣רַע נִשְׁחָ֔ת כִּ֖י אִם־מִבִּלְתִּ֣י נִשְׁחָ֑ת עַל־יְדֵי֨ אִמְרַ֣ת אֱלֹהִ֗ים אֲשֶׁ֥ר הִ֛יא חַיָּ֥ה וְקַיֶּ֖מֶת לְעוֹלָֽם.
1:24	כִּ֤י כָל־בָּשָׂר֨ כְּחָצִ֔יר וְכָל־כְּב֥וֹד אָדָ֖ם כְּצִ֣יץ חָצִ֑יר יָבֵ֥שׁ חָצִ֖יר וְצִיצ֥וֹ נָבֵֽל.
1:25	וּדְבַ֥ר יְהֹוָ֖ה יָק֣וּם לְעוֹלָ֑ם וְזֶ֥ה ה֛וּא הַדָּבָ֖ר הַמְּבֻשָּׂ֥ר לָכֶֽם.
2:1	לָכֵ֕ן בְּהָסִ֣יר מֵֽעֲלֵיכֶ֗ם כָּל־רָעָ֧ה וְכָל־מִרְמָ֛ה וַֽחֲנֻפ֥וֹת וּקְנָא֖וֹת וְכָל־רְכִילֽוֹת.
2:2	כְּעֽוֹלְלִים֨ הַנּֽוֹלָדִ֣ים עַתָּ֔ה תַּֽאֲב֛וּ אֶל־הֶֽחָלָ֥ב הַשִּׂכְלִ֖י וּבִלְתִּ֣י מָה֑וּל לְמַ֛עַן בּ֥וֹ תִגְדְּל֖וּ לִֽישׁוּעָֽה.
2:3	אִם־אָמְנָ֣ם טְעַמְתֶּ֔ם כִּ֥י ט֖וֹב יְהֹוָֽה.
2:4	אֲשֶׁ֥ר אֵלָ֖יו אַתֶּ֣ם נִגָּשִׁ֑ים אֶ֣בֶן חַיָּ֗ה אָמְנָם֨ מְאוּסָ֣ה מֵֽאֲנָשִׁ֔ים אַ֧ךְ נִבְחָרָ֛ה מֵאֵ֥ת הָֽאֱלֹהִ֖ים וִֽיקָרָֽה.
2:5	גַּ֣ם אַתֶּ֗ם כַּֽאֲבָנִ֤ים חַיּוֹת֨ נִבְנִ֣ים אַתֶּ֔ם בַּ֥יִת רֽוּחֲנִ֖י כְּהֻנָּ֣ה קְדוֹשָׁ֑ה לְהַקְרִיב֨ זְבָחִ֣ים רֽוּחֲנִיִּ֔ים רְצוּיִ֣ים לֵֽאלֹהִ֔ים עַל־יְדֵ֖י יֵשׁ֥וּעַ הַמָּשִֽׁיחַ.
2:6	לָכֵ֕ן גַּ֥ם נִמְצָ֖א בַּכְּתָ֑ב הִֽנְנִ֣י ׀ שָׂ֣ם בְּצִיּ֗וֹן אֶ֣בֶן רֹ֤אשׁ פִּנָּה֨ נִבְחָרָ֣ה יְקָרָ֔ה וְהַמַּֽאֲמִ֥ין בָּ֖הּ לֹ֥א יֵבֽוֹשׁ.
2:7	לָכֵ֕ן לָכֶ֥ם הַמַּֽאֲמִינִ֖ים הַיְקָ֑ר אַ֣ךְ לַמֹּרִ֗ים הָאֶ֡בֶן אֲשֶׁר֩ מָֽאֲס֨וּ הַבּוֹנִ֜ים זֹ֤את הָֽיְתָה֨ לְרֹ֣אשׁ פִּנָּ֔ה וְאֶ֥בֶן נֶ֖גֶף וְצ֥וּר מִכְשֽׁוֹל.
2:8	הַמִּתְנַגְּפִ֥ים בַּדָּבָ֖ר בִּמְרוֹתָ֑ם אֲשֶׁ֥ר לָזֹ֖את גַּ֥ם הוּשָֽׂמוּ.
2:9	אַ֣ךְ אַתֶּ֗ם מִשְׁפָּחָ֤ה בְחִירָה֨ מַמְלֶ֣כֶת כֹּֽהֲנִ֔ים גּ֥וֹי קָד֖וֹשׁ עַ֣ם סְגֻלָּ֑ה לְמַ֣עַן תַּגִּ֔ידוּ אֶת־תְּהִלּ֞וֹת הַקּוֹרֵ֥א אֶתְכֶ֛ם מֵחֹ֖שֶׁךְ אֶל־אוֹר֥וֹ הַנִּפְלָֽא.
2:10	אֲשֶׁ֤ר לְפָנִים֨ לֹ֣א עַ֔ם אַ֣ךְ עַתָּ֔ה עַ֥ם אֱלֹהִ֖ים אַתֶּ֑ם אֲשֶׁ֞ר לֹ֤א הֱיִיתֶם֨ מְרֻֽחָמִ֔ים אַ֥ךְ עַתָּ֖ה מְרֻֽחָמִ֥ים אַתֶּֽם.
2:11	הָֽאֲהוּבִ֕ים אֲנִ֛י מַזְהִ֥יר אֶתְכֶ֖ם כְּגֵרִ֣ים וְתֽוֹשָׁבִ֑ים לְהִנָּזֵר֨ מִתַּֽאֲו֣וֹת הַבָּשָׂ֔ר הַנִּלְחָמ֖וֹת עַל־הַנָּֽפֶשׁ.
2:12	בִּהְי֨וֹת לָכֶ֧ם הִתְהַלְּכֽוּתְכֶ֛ם יָפָ֖ה בֵּ֣ין הַגּוֹיִ֑ם לְמַ֡עַן בַּֽאֲשֶׁר֩ הֵ֨ם מְלַשְּׁנִ֤ים אֶתְכֶם֨ כְּפֹ֣עֲלֵי אָ֔וֶן מִן־הַמַּֽעֲשִׂ֤ים הַטּוֹבִים֨ כִּרְאוֹתָ֣ם אֹתָ֔ם יְכַבְּד֥וּ אֶת־הָֽאֱלֹהִ֖ים בְּי֥וֹם הַפְּקֻדָּֽה.
2:13	לָכֵ֕ן הִכָּֽנְע֗וּ לְכֹ֛ל מַֽעֲרֶ֥כֶת הָֽאָדָ֖ם לְמַ֣עַן אֲדֹנָ֑י אִ֚ם לַמֶּ֔לֶךְ כְּמ֖וֹ אֶל־הַשַּׁלִּֽיט.
2:14	אִ֖ם לַמֹּֽשְׁלִ֑ים כְּמ֣וֹ ׀ אֶל־הַשְּׁלוּחִ֣ים עַל־יָדָ֗יו לְנִקְמַת֨ פֹּ֣עֲלֵי אָ֔וֶן וְלִתְהִלַּ֖ת עֹ֥שֵׂי טֽוֹב.
2:15	כִּ֥י כֵ֖ן רְצ֣וֹן הָֽאֱלֹהִ֑ים כִּ֚י בַּֽעֲשׂ֣וֹת ט֔וֹב תַּחְסְמ֕וּ אֶת־בַּֽעֲר֖וּת אֲנָשִׁ֥ים כְּסִילִֽים.
2:16	כְּמ֣וֹ חָפְשִׁ֔ים וְלֹ֣א כָאֵ֗לֶּה אֲשֶׁ֨ר יֵ֥שׁ לָהֶ֛ם הַֽחָפְשׁ֖וּת לְמִכְסֵ֣ה הָֽרָעָ֑ה כִּ֖י אִם־כְּעַבְדֵ֥י הָֽאֱלֹהִֽים.
2:17	אֵ֤ת כָּל־הָֽאָדָם֨ כַּבְּד֔וּ אֶת־הָאַֽחֲוָ֖ה אֱהָ֑בוּ אֶת־הָֽאֱלֹהִ֣ים יְרא֔וּ אֶת־הַמֶּ֖לֶךְ כַּבֵּֽדוּ.
2:18	הָֽעֲבָדִ֕ים הֱי֧וּ נִכְנָעִ֛ים בְּכָל־יִרְאָ֖ה לַֽאֲדֹֽנֵיכֶ֑ם לֹ֤א לְבַד֨ לַטּוֹבִ֣ים וְלָֽרַכִּ֔ים כִּ֖י אִם־גַּ֥ם לָֽעִקְּשִֽׁים.
2:19	כִּ֛י זֹ֥את הִ֖יא חֵ֑ן אִם֩ בַּֽעֲב֨וּר הָֽרַעְי֜וֹן אֲשֶׁ֣ר אֶל־הָֽאֱלֹהִ֗ים יִסְבֹּ֥ל אִישׁ֨ עֲצָבִ֔ים בְּעֻנּוֹת֖וֹ בִּבְלִ֥י יֹֽשֶׁר.
2:20	כִּ֚י מַה־תְּהִלָּ֣ה הִ֔יא אִ֛ם בְּחָטְאֲכֶ֥ם וּבְהֻכּֽוֹתְכֶ֖ם אַתֶּ֣ם סֹֽבְלִ֑ים אַ֡ךְ אִם֩ בַּֽעֲשֽׂוֹתְכֶ֨ם ט֤וֹב ׀ וּבְעֻנּֽוֹתְכֶם֨ אַתֶּ֣ם סֹֽבְלִ֔ים זֹ֥את הִ֛יא חֵ֖ן לִפְנֵ֥י הָֽאֱלֹהִֽים.
2:21	כִּ֖י לְזֹ֣את קֹֽרָאתֶ֑ם יַ֗עַן גַּ֤ם הַמָּשִׁ֨יחַ֨ עֻנָּ֣ה בַֽעֲדֵ֔נוּ וְהִשְׁאִ֥יר לָ֨נוּ֨ תַּבְנִ֔ית לְמַ֥עַן נֵלֵ֖ךְ אַֽחֲרֵ֥י עִקְּבוֹתָֽיו.
2:22	אֲשֶׁ֕ר חֵ֖טְא לֹ֣א עָשָׂ֑ה וְגַ֛ם לֹֽא־נִמְצְאָ֥ה מִרְמָ֖ה בְּפִֽיו.
2:23	אֲשֶׁ֤ר בְּחֹֽרְפוֹ֨ לֹ֣א חֵרֵ֔ף בְּעֻנּוֹת֖וֹ לֹ֣א גָעָ֑ר אַ֛ךְ הִפְקִ֥יד רִיב֖וֹ אֶל־הַשֹּׁפֵ֥ט בְּצֶֽדֶק.
2:24	אֲשֶׁ֣ר ה֗וּא נָשָׂ֧א אֶת־חַטּאוֹתֵ֛ינוּ בִּגְוִיָּת֖וֹ עַל־הָעֵ֑ץ לְמַ֗עַן בְּמוּתֵ֤נוּ לַֽחֲטָאִים֨ נִהְיֶ֣ה לַצְּדָקָ֔ה אֲשֶׁ֥ר בַּֽחֲבֻֽרָת֖וֹ נִרְפָּ֥א לָכֶֽם.
2:25	כִּ֥י הֱיִיתֶ֖ם כְּצֹ֣אן תֹּע֑וֹת אַ֣ךְ שַׁבְתֶּ֥ם עַתָּ֛ה אֶל־הָֽרֹעֶ֖ה וְאֶל־פְּקִ֥יד נַפְשֽׁוֹתֵיכֶֽם.
3:1	כְּמוֹ־כֵן֨ הַנָּשִׁ֔ים הֱיֶ֥ינָה נִכְנָע֖וֹת לְבַֽעֲלֵיכֶ֑ן לְמַ֗עַן גַּ֣ם אִם־אֲחָדִ֞ים אֵינָ֤ם נִשְׁמָעִים֨ אֶל־הַדָּבָ֔ר יִקָּנוּ֨ בִּבְלִ֣י דָבָ֔ר עַל־יְדֵ֖י הִתְהַלְּכ֥וּת הַנָּשִֽׁים.
3:2	בִּרְאוֹתָ֕ם אֶת־הִתְהַלְּכֽוּתְכֶ֥ן הַנְּקִיָּ֖ה אֲשֶׁ֥ר בְּיִרְאָֽה.
3:3	אֲשֶׁר֩ עֶדְיָ֨ן יְהִ֜י לֹ֤א הַֽחִיצוֹן֨ בַּֽאֲרִיגַ֣ת שֵׂעָ֔ר וּבַֽעֲדִיַּ֖ת כְּלֵ֣י זָהָ֑ב א֖וֹ בִּלְבִישַׁ֥ת בְּגָדִֽים.
3:4	כִּי֩ אִם־הָֽאָדָ֨ם הַנִּסְתָּ֜ר אֲשֶׁ֣ר לַלֵּ֗ב בְּאַל־שַׁ֛חַת הָר֥וּחַ הֶֽעָנָ֖ו וְהַשָּׁקֵ֑ט אֲשֶׁ֥ר ה֛וּא יָקָ֥ר מְאֹ֖ד לִפְנֵ֥י הָֽאֱלֹהִֽים.
3:5	כִּ֣י כָ֗כָה לְפָנִ֞ים גַּ֣ם הַנָּשִׁ֧ים הַקְּדוֹשׁ֛וֹת הַמְּקַוּ֥וֹת אֶל־הָֽאֱלֹהִ֖ים הִתְעַדּ֑וּ וְנִכְנְע֖וּ לְבַֽעֲלֵיהֶֽן.
3:6	כְּמ֣וֹ שָׂרָ֗ה נִשְׁמְעָה֨ אֶל־אַבְרָהָ֔ם וְקָֽרְאָ֥ה אֹת֖וֹ אָד֑וֹן אֲשֶׁ֣ר בְּנוֹתֶ֨יהָ֨ נִהְיֵיתֶ֔ן אִם־תַּֽעֲשֶׂ֣ינָה טּ֔וֹב וְלֹ֥א תִירֶ֖אנָה מִכָּל־פַּ֥חַד פִּתְאֹֽם.
3:7	כְּמוֹ־כֵ֣ן הַבְּעָלִ֗ים שִׁכְנ֣וּ עִמָּהֶן֘ כְּפִ֣י דַעַת֒ בְּחַלֵּ֨ק כָּב֜וֹד לָֽאִשָׁ֗ה כְּמ֤וֹ לַכְּלִי֨ הָרַ֣ךְ יֶ֔תֶר וּכְמ֛וֹ חַבְר֥וֹת יְרֻשַּׁתְכֶ֖ם בְּחֵ֣ן הַֽחַיִּ֑ים לְבִלְתִּ֥י הִמָּנַ֖ע תְּפִלּֽוֹתֵיכֶֽם.
3:8	בְּאַֽחֲרִ֕ית הֱי֣וּ כֻלְּכֶ֗ם בַּֽעֲלֵי֨ ר֣וּחַ שָׁוֶ֔ה חַבְרֵ֥י מְעֻנִּ֖ים אֹֽהֲבֵ֣י אַחִ֑ים רַֽחֲמָנִ֖ים שִׁפְלֵ֥י רֽוּחַ.
3:9	בִּלְתִּ֣י מְשַׁלְּמִ֗ים רָעָה֨ תַּ֣חַת רָעָ֔ה א֥וֹ חֵר֖וּף תַּ֣חַת חֵר֑וּף כִּ֤י אִם־לְהֵ֨פֶךְ֨ מְבָ֣רְכִ֔ים בְּדַעְתְּכֶ֗ם כִּ֤י לְזֹאת֨ נִקְרֵאתֶ֔ם לְמַ֥עַן תִּֽירְשׁ֖וּ בְּרָכָֽה.
3:10	כִּ֗י הֶֽחָפֵץ֨ לֶֽאֱהַ֣ב חַיִּ֔ים וְלִרְא֖וֹת יָמִ֣ים טוֹבִ֑ים יִנְצֹ֤ר לְשׁוֹנוֹ֨ מֵרָ֔ע וּשְׂפָתָ֖יו מִדַּבֵּ֥ר מִרְמָֽה.
3:11	יָס֥וּר מֵרָ֖ע וְיַֽעֲשֶׂ֣ה ט֑וֹב יְבַקֵּ֥שׁ שָׁל֖וֹם וְיִרְדְּפֵֽהוּ.
3:12	כִּ֗י עֵינֵ֤י יְהֹוָה֨ עַל־צַדִּיקִ֔ים וְאָזְנָ֖יו אֶל־שַׁוְעָתָ֑ם וּפְנֵ֥י יְהֹוָ֖ה בְּעֹ֥שֵׂי רָֽע.
3:13	וּמִ֣י זֶ֔ה אֲשֶׁ֖ר יָרֵ֣עַ לָכֶ֑ם אִ֣ם נִהְיֵיתֶ֔ם מִתְהַלְּכִ֖ים בְּתַבְנִ֥ית הַטּֽוֹב.
3:14	אַ֗ךְ גַּ֧ם אִם־תְּעֻנּ֛וּ בִּגְלַ֥ל הַצְּדָקָ֖ה אַשְׁרֵיכֶ֑ם וְאֶת־מֽוֹרָאָ֛ם לֹ֥א תִֽירְא֖וּ וְלֹ֥א תַֽעֲרִֽיצוּ.
3:15	אַ֛ךְ אֶת־יְהֹוָ֥ה אֱלֹהִ֖ים הַקְדִּ֣ישׁוּ בְלִבּֽוֹתֵיכֶ֑ם וִהְיוּ֩ נְכוֹנִ֨ים תָּמִ֜יד לְהִתְנַצְּל֗וּת לְכָל־הַשֹּׁאֵ֨ל מִכֶּ֤ם רְאָיָה֨ עַל־הַתִּקְוָה֨ אֲשֶׁ֣ר בָּכֶ֔ם עִ֥ם עֲנָוָ֖ה וְיִרְאָֽה.
3:16	בִּהְי֥וֹת לָכֶ֖ם רַעְי֣וֹן ט֑וֹב לְמַ֡עַן בַּֽאֲשֶׁר֩ הֵ֨ם מְלַשְּׁנִ֤ים אֶתְכֶם֨ כְּפֹֽעֲלֵ֣י רָ֔ע יֵב֨וֹשׁוּ֨ הַמְּחָֽרְפִ֔ים אֶת־הִתְהַלְּכֽוּתְכֶ֥ם הַטּוֹבָ֖ה בַּמָּשִֽׁיחַ.
3:17	כִּי֩ ט֨וֹב לָכֶ֜ם לְעֻנּ֗וֹת אִם־יִרְצֶה֨ רְצ֣וֹן הָֽאֱלֹהִ֔ים בַּֽעֲשֽׂוֹתְכֶ֣ם ט֔וֹב מִבַּֽעֲשֽׂוֹתְכֶ֖ם רָֽע.
3:18	כִּ֣י גַ֣ם הַמָּשִׁ֡יחַ עֻנָּה֩ פַ֨עַם אַחַ֜ת עַל־חֲטָאִ֗ים צַדִּיק֨ בְּעַ֣ד רְשָׁעִ֔ים לְמַ֥עַן יְקָֽרְבֵ֖נוּ אֶל־הָֽאֱלֹהִ֑ים וְהוּמַ֣ת בַּבָּשָׂ֔ר וְהָחְיָ֖ה בָּרֽוּחַ.
3:19	אֲשֶׁ֣ר בּ֗וֹ גַּ֤ם הָלַךְ֨ וְקָרָ֔א אֶל־הָֽרוּח֖וֹת אֲשֶׁ֥ר בְּמִשְׁמָֽר.
3:20	אֲשֶׁ֣ר לְפָנִים֘ לֹ֣א הֶֽאֱמִינוּ֒ כַּֽאֲשֶׁ֨ר חִכָּ֜ה אֹ֣רֶךְ ר֧וּחַ הָֽאֱלֹהִ֛ים בִּימֵ֥י נֹ֖חַ בְּהִתְכּוֹנֵ֣ן הַתֵּבָ֑ה אֲשֶׁ֨ר בָּ֜הּ מְעַטִּ֗ים הֵ֚ם שְׁמֹנֶ֣ה נְפָשׁ֔וֹת נֽוֹשְׁע֖וּ עַל־יְדֵ֥י הַמָּֽיִם.
3:21	אֲשֶׁ֡ר הַדָּבָר֩ הַמְּצֻיָּ֨ר בְּתַבְנִיתָ֜ם מוֹשִׁ֤יעַ עַתָּה֨ גַּ֣ם אֶתְכֶ֔ם הִ֖יא הַטְּבִילָ֑ה לֹ֞א חֲסָרַ֣ת ׀ טִנֹּ֣פֶת הַבָּשָׂ֗ר כִּ֠י אִם־מַֽעֲנֵ֨ה רַעְי֥וֹן טוֹב֨ לִפְנֵ֣י הָֽאֱלֹהִ֔ים עַל־יְדֵ֥י תְקוּמַ֖ת יֵשׁ֥וּעַ הַמָּשִֽׁיחַ.
3:22	אֲשֶׁר֨ עָלָ֣ה אֶל־הַשָּׁמַ֔יִם וְיֶשְׁנ֖וֹ לִימִ֣ין הָֽאֱלֹהִ֑ים וְהֽוּשְׁת֣וּ תַחְתָּ֔יו מַלְאָכִ֥ים וּמֶמְשָׁל֖וֹת וּגְבוּרֽוֹת.
4:1	לָ֠כֵן בַּֽאֲשֶׁ֨ר הַמָּשִׁ֜יחַ עֻנָּ֤ה בַֽעַדְכֶם֨ בַּבָּשָׂ֔ר גַּם־אַתֶּ֗ם אֶת־עֶ֛צֶם הַתְּבוּנָ֥ה הַהִ֖יא הֵֽחָלֵ֑צוּ כִּ֛י הַמְּעֻנֶּ֥ה בַבָּשָׂ֖ר שָׁקַ֥ט מֵחֵֽטְא.
4:2	לְמַעַן֩ יִחְיֶ֨ה מֵֽעַתָּ֜ה אֶת־שְׁאָ֤ר עִתּוֹ֨ בַּבָּשָׂ֔ר לֹ֚א לְתַֽאֲוֹ֣ת אֲנָשִׁ֔ים כִּ֖י אִם־לִרְצ֥וֹן אֱלֹהִֽים.
4:3	כִּ֣י ׀ דַּ֣י לָכֶ֗ם עֵ֤ת הַֽחַיִּים֨ אֲשֶׁ֣ר עָבַ֔ר לִפְעֹ֖ל אֶת־רְצ֣וֹן הַגּוֹיִ֑ם כַּֽאֲשֶׁ֨ר הִתְהַלַּכְתֶּ֜ם בְּזִמּ֣וֹת ׀ בְּתַֽאֲו֗וֹת בִּדְלִיק֥וֹת יַ֨יִן֨ בְּזֽוֹלְל֣וֹת בִּשְׁתִיּ֔וֹת וּבַֽעֲבוֹד֥וֹת אֱלִילִ֖ים הַמְּכֹֽעָרֽוֹת.
4:4	אֲשֶׁ֣ר בָּזֹ֔את הֵ֥ם חֹֽשְׁבִ֖ים לְזָרָ֑ה כִּ֞י אֵֽינְכֶ֣ם ׀ רָצִ֣ים עִמָּהֶ֗ם אֶל־עֶ֠צֶם אֶ֤שֶׁד הַזּֽוֹלְלוּת֨ הַה֔וּא וּמְגַדְּפִ֖ים הֵֽם.
4:5	אֲשֶׁר֨ יִתְּנ֣וּ חֶשְׁבּ֔וֹן אֶל־הֶֽעָתִ֛יד לִשְׁפּ֥וֹט אֶת־הַֽחַיִּ֖ים וְאֶת־הַמֵּתִֽים.
4:6	כִּ֚י־אֶל־זֹ֔את הִתְבַּשְּׂר֖וּ גַּ֣ם הַמֵּתִ֑ים לְמַ֨עַן יִשָּֽׁפְט֜וּ כְּפִ֤י אֲנָשִׁים֙ בַּבָּשָׂ֔ר וְיִחְי֛וּ כְּפִ֥י הָֽאֱלֹהִ֖ים בָּרֽוּחַ.
4:7	אַ֕ךְ קֵ֥ץ כָּל־דְּבָרִ֖ים קָרֵ֑ב לָכֵן֨ הִנָּֽזְר֔וּ וְשִׁקְד֖וּ אֶל־הַתְּפִלּֽוֹת.
4:8	וְלִפְנֵ֣י כָל־דְּבָרִ֔ים אֶֽחֱז֛וּ אַֽהֲבָ֥ה עֲצוּמָ֖ה אִ֣ישׁ לְרֵעֵ֑הוּ כִּ֚י הָאַֽהֲבָ֔ה תְּכַסֶּ֖ה הֲמ֥וֹן חֲטָאִֽים.
4:9	אֶֽהֱב֧וּ לְאָרֵ֛חַ אִ֥ישׁ אֶת־רֵעֵ֖הוּ בִּבְלִ֥י תְלוּנֽוֹת.
4:10	כָּל־אִ֗ישׁ כַּֽאֲשֶׁ֤ר לָקַח֨ מַתְּנַ֣ת חֵ֔ן יְשָֽׁרְת֥וּ אֹתָ֖הּ אִ֣ישׁ לְרֵעֵ֑הוּ כְּסֽוֹכְנִ֣ים טוֹבִ֔ים עַל־חֵ֥ן הָֽאֱלֹהִ֖ים הַמְּרֻבָּֽב.
4:11	אִם־אִ֣ישׁ מְדַבֵּ֗ר כְּמוֹ֘ אִמְר֣וֹת אֱלֹהִים֒ אִם־אִ֣ישׁ מְשָׁרֵ֗ת כְּמוֹ֨ מִן־הַכֹּ֔חַ אֲשֶׁ֥ר הֵפִ֖יק הָֽאֱלֹהִ֑ים לְמַעַן֩ יְכֻבַּ֨ד הָֽאֱלֹהִ֜ים בַּכֹּ֗ל עַל־יְדֵי֨ יֵשׁ֣וּעַ הַמָּשִׁ֔יחַ אֲשֶׁ֨ר ל֜וֹ הַכָּב֧וֹד וְהַגְּבוּרָ֛ה לְעֽוֹלְמֵ֥י עֽוֹלָמִ֖ים אָמֵֽן.
4:12	הָֽאֲהוּבִ֕ים אַל־תַּחְשְׁב֣וּ לְזָרָ֗ה עַל־אֹד֤וֹת הַצְּרִיפָה֨ אֲשֶׁ֣ר בָּכֶ֔ם הַנִּהְיָ֥ה לָכֶ֖ם לְמַסָּ֑ה כְּמ֛וֹ אִם־דָּבָ֥ר זָ֖ר קָרָ֥ה אֶתְכֶֽם.
4:13	כִּ֣י אִם־כְּ֠פִי אֲשֶׁ֨ר אַתֶּ֧ם מִתְחַבְּרֵ֛י עִנּוּיֵ֥י הַמָּשִׁ֖יחַ שְׂמָ֑חוּ לְמַ֗עַן גַּ֛ם בְּהִתְגַּלּ֥וּת כְּבוֹד֖וֹ תִּשְׂמְח֥וּ וְתָגִֽילוּ.
4:14	אִם֩ מְחֹֽרָפִ֨ים אַתֶּ֜ם מִפְּנֵ֨י שֵׁ֤ם הַמָּשִׁ֨יחַ֨ אַשְׁרֵיכֶ֔ם כִּ֥י ר֧וּחַ הַכָּב֛וֹד וְהָֽאֱלֹהִ֖ים נָ֣ח עֲלֵיכֶ֑ם אָמְנָ֗ם עַל־יְדֵיהֶם֨ ה֣וּא מְגֻדָּ֔ף אַ֥ךְ עַל־יְדֵיכֶ֖ם ה֥וּא מְכֻבָּֽד.
4:15	אוּלָ֞ם אַל־יְעֻנֶּ֣ה ׀ אִ֣ישׁ מִכֶּ֗ם כְּרוֹצֵ֨חַ֨ א֣וֹ כְגַנָּ֔ב א֖וֹ כְּפֹעֵ֣ל אָ֑וֶן א֕וֹ כְּחֹתֵ֖ר בִּפְקֻדַּ֥ת אַחֵֽר.
4:16	אַ֛ךְ אִ֥ם כִּמְשִׁיחִ֖י אַל־יֵב֑וֹשׁ כִּ֛י אִם־יְכַבֵּ֥ד אֶת־הָֽאֱלֹהִ֖ים בַּחֵ֥לֶק הַזֶּֽה.
4:17	כִּ֣י הָעֵ֔ת לְהָחֵ֥ל אֶת־הַמִּשְׁפָּ֖ט מִבֵּ֣ית הָֽאֱלֹהִ֑ים וְאִ֤ם בָּרִֽאשֹׁנָה֨ מִמֶּ֔נּוּ מָ֚ה אַֽחֲרִ֣ית אֵ֔לֶּה אֲשֶׁ֛ר אֵינָ֥ם מַֽאֲמִינִ֖ים לִבְשׂוֹרַ֥ת הָֽאֱלֹהִֽים.
4:18	וְאִ֤ם הַצַּדִּיק֨ בְּקֹ֣שִׁי יִוָּשֵׁ֔עַ הָֽרָשָׁ֥ע וְהַֽחַטָּ֖א אַיֵּ֥ה יֵֽרָאֶֽה.
4:19	לָכֵ֗ן גַּ֤ם הַמְעֻנִּים֨ כְּפִי֨ רְצ֣וֹן הָֽאֱלֹהִ֔ים כְּמוֹ֙ אֶל־בּוֹרֵ֣א נֶֽאֱמָ֔ן יַפְקִ֧ידוּ אֵלָ֛יו אֶת־נַפְשֽׁוֹתֵיהֶ֖ם בַּֽעֲשִׂיַּ֥ת טֽוֹב.
5:1	אֶת־הַזְּקֵנִ֨ים אֲשֶׁ֤ר בָּכֶם֨ אֲנִ֣י מַזְהִ֔יר חֲבֵֽרְכֶ֣ם בַּזִּקְנָ֔ה וְעֵ֖ד עִנּוּיֵ֣י הַמָּשִׁ֑יחַ וְגַ֛ם מִתְחַבֵּ֥ר הַכָּב֖וֹד הֶֽעָתִ֥יד לְהִתְגַּלּֽוֹת.
5:2	רְעוּ֩ אֶת־עֵ֨דֶר הָֽאֱלֹהִ֜ים אֲשֶׁ֤ר בָּכֶם֨ בְּפָקְדְכֶ֣ם אֹתָ֔ם לֹ֥א בְאֹ֖נֶס כִּ֣י אִם־בְּרָצ֑וֹן לֹ֗א בְּחֶמְדַּת֨ בֶּ֣צַע רָ֔ע כִּ֖י אִם־בִּנְדִיב֥וּת רֽוּחַ.
5:3	גַּם־לֹ֖א כְּרֹדִ֣ים אֶת־הַנְּחָל֑וֹת כִּ֛י אִם־בִּהְיֽוֹתְכֶ֥ם לְתַבְנִיּ֖וֹת אֶל־הָעֵֽדֵר.
5:4	וְכַֽאֲשֶׁ֥ר יִגָּלֶ֖ה רֹ֣אשׁ הָֽרֹעִ֑ים תִּקְּח֛וּ עֲטֶ֥רֶת תִּפְאֶ֖רֶת בִּלְתִּ֥י נֹבָֽלֶת.
5:5	כְּמוֹ־כֵ֗ן הַצְּעִירִים֘ הִכָּֽנְע֣וּ לַזְּקֵנִים֒ אַ֣ךְ הַכֹּ֗ל הִכָּֽנְעוּ֨ אִ֣ישׁ לְרֵעֵ֔הוּ וְהִתְעַטְּפ֖וּ בְּשִׁפְל֣וּת הָר֑וּחַ כִּ֤י הָֽאֱלֹהִים֨ יִתְנַגֵּ֣ד לַזֵּדִ֔ים וְלַֽעֲנָוִ֖ים יִתֵּ֥ן חֵֽן.
5:6	לָכֵן֨ הַשְׁפִּ֣ילוּ נַפְשְׁכֶ֔ם תַּ֛חַת יַ֥ד הָֽאֱלֹהִ֖ים הַֽחֲזָקָ֑ה לְמַ֛עַן יְרוֹמֵ֥ם אֶתְכֶ֖ם בְּעִתּֽוֹ.
5:7	כָּל־יְהָבְכֶ֖ם הַשְׁלִ֣יכוּ עָלָ֑יו כִּ֛י ה֥וּא דֹאֶ֖ג לָכֶֽם.
5:8	ע֖וּרוּ שְׁקֹ֑דוּ כִּ֣י ׀ מְרִֽיבְכֶ֣ם הַשָּׂטָ֗ן כְּאַרְיֵ֤ה שֹׁאֵג֨ מִתְהַלֵּ֔ךְ בְּבַקְּשׁ֖וֹ אֶת־מִ֥י יְבַלֵּֽעַ.
5:9	אֲשֶׁ֤ר אֵלָיו֨ הִתְנַגְּד֔וּ קַיָּמִ֖ים בָּֽאֱמוּנָ֑ה בְּדַעְתְּכֶ֕ם כִּ֞י עֶ֣צֶם ׀ הָֽעִנּוּיִ֣ים הָאֵ֗לֶּה נִשְׁלָמִ֛ים בְּאַנְשֵׁ֥י אַֽחֲוַתְכֶ֖ם אֲשֶׁ֥ר בַּתֵּבֵֽל.
5:10	וֵֽאלֹהֵ֣י כָל־חֶ֗סֶד הַקֹּרֵ֥א אֹתָ֛נוּ לִכְבוֹד֥וֹ הַנִּצְחִ֖י בְּיֵשׁ֣וּעַ הַמָּשִׁ֑יחַ כְּעֻנּֽוֹתְכֶ֣ם לְמִצְעָ֔ר יְתַקֵּ֧ן יְקַיֵּ֛ם יְגַבֵּ֥ר וִֽייַסֵּ֖ד אֶתְכֶֽם.
5:11	ל֗וֹ הַכָּב֧וֹד וְהַגְּבוּרָ֛ה לְעֽוֹלְמֵ֥י עֽוֹלָמִ֖ים אָמֵֽן.
5:12	עַל־יְדֵ֣י סִלְוָנוֹס֩ אָ֨ח נֶֽאֱמָ֜ן לָכֶ֗ם כַּֽאֲשֶׁר֨ אֲנִ֣י מְחַשֵּׁ֔ב כָּתַ֖בְתִּי בִּמְעַ֣ט מִלִּ֑ים בְּהַזְהִ֣יר וּבְהָעִ֔יד כִּ֣י ׀ זֶ֣ה ה֗וּא חֵ֤ן הָֽאֱלֹהִים֨ הָֽאֲמִתִּ֔י אֲשֶׁ֛ר אַתֶּ֥ם עֹֽמְדִ֖ים בּֽוֹ.
5:13	שֹׁאֶ֣לֶת לִשְׁלֽוֹמְכֶ֔ם הַקְּהִלָּ֥ה הַנִּבְחָרָ֖ה אֲשֶׁ֣ר בְּבָבֶ֑ל וְגַ֖ם מַרְק֥וֹס בְּנִֽי.
5:14	שַֽׁאֲל֧וּ לְשָׁל֛וֹם אִ֥ישׁ לְרֵעֵ֖הוּ בִּנְשִׁיקַ֣ת אַֽהֲבָ֑ה שָׁל֣וֹם לָכֶ֔ם לַכֹּ֗ל אֲשֶׁ֥ר הֵ֛ם בְּיֵשׁ֥וּעַ הַמָּשִׁ֖יחַ אָמֵֽן.
