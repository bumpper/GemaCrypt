‏Philippians 
1:1	פָּא֣וּל וְטִמֽוֹתִיּ֔וֹס עַבְדֵ֖י יֵשׁ֣וּעַ הַמָּשִׁ֑יחַ לְכָל־הַקְּדוֹשִׁ֣ים ׀ בְּיֵשׁ֣וּעַ הַמָּשִׁ֗יחַ אֲשֶׁ֥ר הֵם֨ בְּפִֽילִפִּ֔י עִ֛ם רָאשֵׁ֥י הַמּוֹרִ֖ים וְהַמְּשָֽׁרְתִֽים.
1:2	חֵ֥ן לָכֶ֖ם וְשָׁל֑וֹם מֵאֵת֨ אֱלֹהִ֣ים אָבִ֔ינוּ וְהָֽאָד֖וֹן יֵשׁ֥וּעַ הַמָּשִֽׁיחַ.
1:3	אֲנִ֤י מוֹדֶה֨ לֵֽאלֹהַ֔י בְּכָל־זֵ֖כֶר אֲשֶׁ֥ר לָכֶֽם.
1:4	בְּהַפִּילִ֨י אֶת־הַתְּחִנָּ֤ה בְשִׂמְחָה֨ בְּכָל־עֵ֔ת בְּכָל־תְּפִלָּתִ֖י בְּעַ֥ד כֻּלְּכֶֽם.
1:5	עַ֚ל הִתְחַבְּרֽוּתְכֶ֣ם בַּבְּשׂוֹרָ֔ה מִן־הַיּ֥וֹם הָֽרִאשׁ֖וֹן עַד־עָֽתָּה.
1:6	יַ֚עַן אֲנִ֣י בָט֔וּחַ עַל־עֶ֖צֶם הַדָּבָ֣ר הַזֶּ֑ה כִּ֣י הַמֵּחֵ֤ל בָּכֶם֨ פֹּ֣עַל ט֔וֹב יְכַלֶּ֕ה עַד־י֖וֹם יֵשׁ֥וּעַ הַמָּשִֽׁיחַ.
1:7	כַּֽאֲשֶׁר֩ נָכ֨וֹן לִ֜י לְהַשְׂכִּ֥יל זֹאת֨ עַל־כֻּלְּכֶ֔ם בַּֽעֲב֛וּר הֱיֽוֹתְכֶ֥ם לִ֖י בְּלִבִּ֑י בַּֽאֲשֶׁ֣ר ׀ גַּ֣ם בְּמֽוֹסְרוֹתַ֗י וְגַ֨ם בְּהִתְנַצְּל֤וּת הַבְּשׂוֹרָה֨ וְקִיּוּמָ֔הּ חַבְרֵ֥י חַסְדִּ֖י יֶשְׁכֶ֥ם כֻּלְּכֶֽם.
1:8	כִּ֛י עֵדִ֖י ה֣וּא הָֽאֱלֹהִ֑ים אֵ֗יךְ אֲנִ֤י חֹשֵׁק֨ אֶת־כֻּלְּכֶ֔ם בְּרַֽחֲמֵ֖י יֵשׁ֥וּעַ הַמָּשִֽׁיחַ.
1:9	וְזֹ֖את אֲנִ֣י מִתְפַּלֵּ֑ל כִּ֣י אַֽהֲבַתְכֶ֗ם תַּעְדִּ֨יף ע֜וֹד יוֹתֵ֧ר וְיוֹתֵ֛ר בְּהַכָּרָ֖ה וּבְכָל־דָּֽעַת.
1:10	בַּֽעֲב֥וּר בָּחְנְכֶ֖ם אֵ֣ת הַמְּעֻלִּ֑ים לְמַ֗עַן תִּהְ֤יוּ נְקִיִּים֨ וּבְלִי־מִכְשֹׁ֔ל עַ֖ד י֥וֹם הַמָּשִֽׁיחַ.
1:11	מְלֵאִים֨ פְּרִ֣י צְדָקָ֔ה אֲשֶׁ֥ר הֵ֛ם עַל־יְדֵ֖י יֵשׁ֣וּעַ הַמָּשִׁ֑יחַ לִכְב֥וֹד אֱלֹהִ֖ים וּתְהִלָּתֽוֹ.
1:12	אַ֣ךְ רֹצֶ֥ה אֲנִ֛י כִּ֥י תֵֽדְע֖וּ אֶחָ֑י כִּ֚י הַדְּבָרִ֣ים אֹֽדוֹתַ֔י נָֽפְל֥וּ בְיוֹתֵ֖ר לְהַצְלָחַ֥ת הַבְּשׂוֹרָֽה.
1:13	עַ֠ד אֲשֶׁ֨ר מֽוֹסְרוֹתַ֜י נִהְי֤וּ גְלוּיִם֨ בַּמָשִׁ֔יחַ בְּכֹ֛ל חֲצַ֥ר הַפֶּחָ֖ה וּבְכֹ֥ל הַנִּשְׁאָרִֽים.
1:14	וְרֹ֨ב הָֽאַחִ֤ים בָּֽאָדוֹן֨ בְּהִתְחַזְּקָ֣ם בְּמֽוֹסְרוֹתַ֔י ע֣וֹד יוֹתֵ֤ר הַרְבֵּה֙ הִתְעָֽרְב֔וּ לְדַבֵּ֥ר אֶת־הַדָּבָ֖ר בִּבְלִ֥י פָֽחַד.
1:15	אֲחָדִ֣ים אָמְנָ֗ם קֹֽרְאִים֨ אֶת־הַמָּשִׁ֔יחַ גַּ֛ם בַּֽעֲב֥וּר קִנְאָ֖ה וּמְרִיבָ֑ה וַ֣אֲחָדִ֔ים גַּ֛ם בַּֽעֲב֖וּר רָצ֥וֹן טֽוֹב.
1:16	אֵ֣לֶּה מַגִּידִ֧ים אֶת־הַמָּשִׁ֛יחַ מִמְּרִיבָ֖ה בִּבְלִי־תֹ֑ם בְּדַמּוֹתָ֕ם לְהוֹסִ֥יף צָרָ֖ה עַל־מֽוֹסְרוֹתָֽי.
1:17	אַ֥ךְ אֵ֖לֶּה מֵאַֽהֲבָ֑ה בְּדַעְתָּ֕ם כִּ֛י לְהִתְנַצְּל֥וּת הַבְּשׂוֹרָ֖ה אֲנִ֥י מוּשָֽׂם.
1:18	מָה֘ אֵפוֹא֒ אֶ֣פֶס ׀ בְּכָל־דֶּ֗רֶךְ אִם־לְמַרְאֶה֨ אִם־בֶּֽאֱמֶ֔ת הַמָּשִׁ֖יחַ ה֣וּא מֻגָּ֑ד וּבְזֹאת֨ אֲנִ֣י שָׂמֵ֔חַ אָכֵ֖ן גַּ֥ם אֶשְׂמָֽח.
1:19	כִּ֣י יָדַ֔עְתִּי זֹ֗את תַּגִּ֤יעַ לִי֨ לִֽישׁוּעָ֔ה עַל־יְדֵ֖י תְּפִלַּתְכֶ֑ם וַֽהֲפָ֥קַת ר֖וּחַ יֵשׁ֥וּעַ הַמָּשִֽׁיחַ.
1:20	כְּפִ֤י תֽוֹחַלְתִּי֨ וְתִקְוָתִ֔י כִּ֛י בְּכָל־מְא֖וּמָה לֹ֣א אֵב֑וֹשׁ כִּ֣י אִם־בְּכָל־בִּטְחָ֡ה כַּֽאֲשֶׁ֣ר בְּכָל־עֵת֩ גַּ֨ם עַתָּ֜ה יְגֻדַּ֤ל הַמָּשִׁ֨יחַ֨ בִּגְוִיָּתִ֔י אִ֚ם עַל־יְדֵ֣י חַיִּ֔ים אִ֖ם עַל־יְדֵ֥י מָֽוֶת.
1:21	כִּ֥י לִ֛י חֲי֖וֹת הַמָּשִׁ֑יחַ וּמ֖וֹת בָּֽצַע.
1:22	אַ֗ךְ אִ֚ם חֲי֣וֹת בַּבָּשָׂ֔ר זֹ֥את לִ֖י פְּרִ֣י יְגִיעָ֑ה וְאֵ֛ת אֲשֶׁ֥ר אֶבְחַ֖ר לֹ֥א יָדָֽעְתִּי.
1:23	כִּ֛י מְאֻלָּ֥ץ אֲנִ֖י מִן־הַשְׁתָּ֑יִם בִּהְי֨וֹת לִ֜י הַתְּשׁוּקָ֗ה לִפְטַר֨ וְלִהְי֣וֹת עִם־הַמָּשִׁ֔יחַ הִ֛יא טוֹבָ֖ה יֶ֥תֶר מְאֹֽד.
1:24	אַ֚ךְ שָׁכֹ֣ן בַּבָּשָׂ֔ר נִצְרָ֥ךְ יוֹתֵ֖ר לְמַֽעַנְכֶֽם.
1:25	וּבִהְיוֹתִ֥י בָט֖וּחַ עַל־זֹ֑את אֲנִ֨י יֹדֵ֜עַ כִּ֣י אֶשְׁכֹּ֔ן וְאֶשְׁכֹּ֥ן יַ֨חַד֨ עִם־כֻּלְּכֶ֔ם לְהַצְלָֽחַתְכֶ֖ם וּלְשִׂמְחַ֥ת אֱמֽוּנַתְכֶֽם.
1:26	לְמַ֨עַן תַּעְדִּ֥יף בִּ֛י תְּהִלַּתְכֶ֖ם בַּמָּשִׁ֣יחַ יֵשׁ֑וּעַ עַל־יְדֵ֛י בִּֽיאָתִ֥י ע֖וֹד אֲלֵיכֶֽם.
1:27	רַ֗ק כְּשֹׁוֶ֛ה לִבְשׂוֹרַ֥ת הַמָּשִׁ֖יחַ הִתְהַלֵּ֑כוּ לְמַ֡עַן אִ֣ם אָבוֹא֩ וְאֶרְאֶ֨ה אֶתְכֶ֜ם א֣וֹ ׀ אִם־אֶהְיֶ֣ה בִלְתִּֽי־נוֹכָ֗ח אֶשְׁמַע֨ אֶת־הַדְּבָרִ֣ים אֹדֽוֹתֵיכֶ֔ם כִּ֣י אַתֶּ֤ם עֹֽמְדִים֨ בְּר֣וּחַ אֶחָ֔ד וּבְנֶ֧פֶשׁ אַחַ֛ת נֶֽאֱבָקִ֥ים יַ֖חַד בֶּֽאֱמוּנַ֥ת הַבְּשׂוֹרָֽה.
1:28	וּבִלְתִּ֧י נִבְעָתִ֛ים בְּכָל־דָּבָ֖ר מִפְּנֵ֣י הַשֹּֽׂטְנִ֑ים אֲשֶׁר֩ הִ֨יא לָהֶ֜ם מוֹפֵ֣ת אֲבַדּ֗וֹן אַ֤ךְ לָכֶם֨ לִתְשׁוּעָ֔ה וְזֹ֖את מֵאֵ֥ת אֱלֹהִֽים.
1:29	כִּ֚י לָכֶ֣ם נִתַּ֔ן לְמַ֖עַן הַמָּשִׁ֑יחַ לֹ֤א לְבַד֨ לְהַֽאֲמִ֣ין בּ֔וֹ כִּ֛י אִם־גַּ֥ם לְעֻנּ֖וֹת לְמַֽעֲנֽוֹ.
1:30	בִּהְי֣וֹת לָכֶ֗ם עֶ֚צֶם הַמִּלְחָמָ֣ה הַהִ֔יא אֲשֶׁ֥ר כָּמ֖וֹהָ רְאִ֣יתֶם בִּ֑י וְעַתָּ֕ה אַתֶּ֥ם שֹֽׁמְעִ֖ים בִּֽי.
2:1	לָכֵ֗ן אִם־יֵ֞שׁ אֵי־ז֤וּ הַזְהָרָה֨ בַּמָּשִׁ֔יחַ אִ֛ם אֵי־ז֖וּ נֶֽחָמַ֣ת אַֽהֲבָ֑ה אִ֗ם אֵי־זוּ֨ הִתְחַבְּר֣וּת הָר֔וּחַ אִ֛ם אֵי־ז֥וּ רַֽחַמִ֖ים וַֽחֲסָדִֽים.
2:2	מַלְּאוּ֨ אֶת־שִׂמְחָתִ֔י לְמַ֣עַן תַּשְׂכִּ֔ילוּ אֶת־עֶ֖צֶם הַדָּבָ֣ר הַה֑וּא בִּהְי֨וֹת לָכֶ֜ם עֶ֣צֶם ׀ הָאַֽהֲבָ֣ה הַהִ֗יא בַּֽעֲלֵי֨ נֶ֣פֶשׁ אַחַ֔ת מַשְׂכִּילִ֖ים אֶת־הָֽאֶחָֽת.
2:3	אַל־תַּֽעֲשׂ֤וּ מְא֨וּמָה֨ עַל־פִּ֣י מָד֔וֹן א֖וֹ כְּב֣וֹד הָ֑בֶל כִּ֣י ׀ אִם־בְּעַנְוַ֣ת ר֗וּחַ תַּחְשְׁב֛וּ אִ֥ישׁ אֶת־רֵעֵ֖הוּ נַֽעֲלִ֥ים עַל־עַצְמָֽם.
2:4	אַל־תַּבִּ֥יטוּ כָל־אִ֖ישׁ אֶת־הַדְּבָרִ֣ים שֶׁלָּכֶ֑ם כִּ֚י אִם־כָּל־אִ֔ישׁ גַּ֕ם אֵ֖ת אֲשֶׁ֥ר לַֽאֲחֵרִֽים.
2:5	כִּ֥י זֹאת֨ הַשְׂכִּ֣ילוּ בָכֶ֔ם אֵ֥ת אֲשֶׁ֛ר גַּ֖ם בַּמָּשִׁ֥יחַ יֵשֽׁוּעַ.
2:6	אֲשֶׁ֗ר בִּהְיוֹתוֹ֨ בְּתַבְנִ֣ית אֱלֹהִ֔ים לֹֽא־חָשָׁ֣ב לְגָזֵ֔ל לִהְי֥וֹת שָׁוֶ֖ה לֵֽאלֹהִֽים.
2:7	כִּ֚י אִם־הֶֽעֱרָ֣ה אֶת־נַפְשׁ֔וֹ וְלָקַ֖ח תַּבְנִ֣ית עָ֑בֶד וְנִהְיָ֖ה בִּדְמ֥וּת בְּנֵֽי־אָדָֽם.
2:8	וּבְהִמָּֽצְא֨וֹ בְתֹ֤אַר כְּאָדָם֨ הִשְׁפִּ֣יל אֶת־עַצְמ֔וֹ בִּהְיוֹת֥וֹ נִשְׁמָ֖ע עַד־מָ֑וֶת אָכֵ֖ן מ֥וֹת הַצְּלָֽב.
2:9	עַל־כֵּ֗ן גַּ֧ם הָֽאֱלֹהִ֛ים רוֹמֵ֥ם אֹת֖וֹ מְאֹ֑ד וְנָ֤תַן לוֹ֙ שֵׁ֔ם אֲשֶׁ֖ר עַ֥ל כָּל־שֵֽׁם.
2:10	לְמַ֛עַן בְּשֵׁ֥ם יֵשׁ֖וּעַ תִּכְרַ֣ע כָּל־בֶּ֑רֶךְ מֵאֵ֜לֶּה אֲשֶׁ֤ר בַּשָּׁמַ֨יִם֨ וּבָאָ֔רֶץ וּמֵאֵ֛לֶּה אֲשֶׁ֖ר מִתַּ֥חַת לָאָֽרֶץ.
2:11	וְכָל־לָשׁ֣וֹן תּוֹדֶ֔ה כִּ֛י יֵשׁ֥וּעַ הַמָּשִׁ֖יחַ ה֣וּא הָֽאָד֑וֹן לִכְב֖וֹד אֱלֹהִ֥ים הָאָֽב.
2:12	לָכֵ֣ן אֲהוּבַ֗י כַּֽאֲשֶׁר֨ תָּמִ֣יד נִשְׁמַעְתֶּ֔ם לֹ֣א לְבַ֞ד כְּמ֣וֹ בְנֹֽכְחוּתִ֗י כִּ֧י אִם־עַתָּ֛ה הַרְבֵּ֥ה יוֹתֵ֖ר בְּאֵֽין־נֹֽכְחוּתִ֑י בְּיִרְאָ֧ה וּבִרְעָדָ֛ה פַּֽעֲל֖וּ אֶת־פְּעֻלַּ֥ת תְּשֽׁוּעַתְכֶֽם.
2:13	כִּ֣י הָֽאֱלֹהִ֗ים ה֣וּא הַפֹּעֵ֥ל בָּכֶ֛ם גַּ֥ם חֲפֹ֖ץ גַּ֣ם פְּעֹ֑ל כְּפִ֖י הָֽרָצ֥וֹן הַטּֽוֹב.
2:14	אֶת־הַכֹּ֣ל עֲשׂ֔וּ בִּבְלִ֥י תְלוּנ֖וֹת וְהִתְוַכְּחֽוֹת.
2:15	לְמַ֨עַן תִּהְי֜וּ תְּמִימִ֣ים וּנְקִיִּ֗ים בְּנֵ֤י אֱלֹהִים֨ בְּלִי־שִׁמְצָ֔ה בְּת֛וֹךְ דּ֥וֹר עִקֵּ֖שׁ וּפְתַלְתֹּ֑ל אֲשֶׁ֣ר בֵּֽינֵיהֶ֗ם אַתֶּ֧ם מְאִירִ֛ים כִּמְאוֹר֖וֹת בַּתֵּבֵֽל.
2:16	בְּהַֽחֲזִיק֨ אֶת־דְּבַ֣ר הַֽחַיִּ֔ים לִתְהִלָּ֥ה לִ֖י לְי֣וֹם הַמָּשִׁ֑יחַ כִּ֗י לֹ֤א לָרִיק֨ רַ֔צְתִּי גַּ֛ם לֹ֥א לָרִ֖יק עָמָֽלְתִּי.
2:17	אַ֠ךְ אִם־גַּ֨ם אֲנִ֥י נִסָּ֛ךְ עַל־זֶ֥בַח אֱמֽוּנַתְכֶ֖ם וְשֵֽׁרוּתָ֑הּ אֲנִ֣י שָׂמֵ֔חַ וְאֶשְׂמַ֖ח עִם־כֻּלְּכֶֽם.
2:18	וְעַל֨ הַדָּבָ֣ר הַזֶּ֔ה גַּ֥ם אַתֶּ֖ם שְׂמָ֑חוּ וְשִׂמְח֖וּ עִמָּדִֽי.
2:19	וַֽאֲקַוֶּה֨ בָּֽאָד֣וֹן יֵשׁ֔וּעַ לִשְׁל֧וֹחַ לָכֶ֛ם מְהֵרָ֖ה אֶת־טִמֽוֹתִיּ֑וֹס לְמַ֤עַן גַּם־אֲנִי֨ אֶגָּפֶ֔שׁ כַּֽאֲשֶׁ֥ר אֵדַ֛ע אֶת־הַדְּבָרִ֖ים אֹדֽוֹתֵיכֶֽם.
2:20	כִּ֚י אֵ֣ין לִ֔י בַּ֖עַל נֶ֣פֶשׁ שָׁוָ֑ה אֲשֶׁ֛ר יִדְאַ֥ג בְּזַכּ֖וּת עַל־אֹדֽוֹתֵיכֶֽם.
2:21	כִּ֚י הַכֹּ֣ל מְבַקְשִׁ֔ים אֵ֥ת הַדְּבָרִ֖ים אֲשֶׁ֣ר לָהֶ֑ם לֹ֕א אֵ֛ת אֲשֶׁ֖ר לְיֵשׁ֥וּעַ הַמָּשִֽׁיחַ.
2:22	אַ֥ךְ אֶת־בָּחְנ֖וֹ אַתֶּ֣ם יֽוֹדְעִ֑ים כִּ֗י כְּמ֥וֹ בֵן֨ לָאָ֔ב עָבַ֥ד עִמִּ֖י בַּבְּשׂוֹרָֽה.
2:23	לָכֵ֕ן אֶת־זֶ֛ה אֲנִ֥י מְקַוֶּ֖ה לִשְׁל֣וֹחַ בִּמְהֵרָ֑ה כַּֽאֲשֶׁ֥ר אֶרְאֶ֖ה אֶת־הַדְּבָרִ֥ים אֹֽדוֹתָֽי.
2:24	אַ֛ךְ בָּט֥וּחַ אֲנִ֖י בָּֽאָד֑וֹן כִּ֛י גַּ֥ם אָֽנֹכִ֖י אָב֥וֹא מְהֵרָֽה.
2:25	אַ֞ךְ חָשַׁ֣בְתִּי לְצֹ֗רֶךְ לִשְׁל֤וֹחַ אֲלֵיכֶם֨ אֶת־אֶפַּפְרֹֽדִיט֣וֹס אָחִ֔י וַֽחֲבֵ֥ר עֲבֹֽדָתִ֖י וַֽחֲבֵ֣ר מִלְחַמְתִּ֑י אַ֧ךְ שֶׁלָּכֶ֛ם שָׁלִ֖יחַ וּמְשָׁרֵ֥ת צָרְכִּֽי.
2:26	בַּֽאֲשֶׁ֥ר הָיָ֛ה חֹשֵׁ֥ק אֶת־כֻּלְּכֶ֖ם וּמִתְעַטֵּ֑ף יַ֥עַן שְׁמַעְתֶּ֖ם כִּ֥י חָלָֽה.
2:27	כִּ֣י אָמְנָ֗ם חָלָה֨ קָר֣וֹב לַמָּ֔וֶת אַ֥ךְ הָֽאֱלֹהִ֖ים חָנַ֣ן אֹת֑וֹ וְלֹא֩ אֹת֨וֹ לְבַ֜ד כִּ֣י ׀ אִם־גַּ֣ם אֹתִ֗י לְ֠מַעַן לֹֽא־יִהְיֶ֥ה לִ֛י יָג֖וֹן עַל־יָגֽוֹן.
2:28	לָכֵ֕ן בְּיֶ֥תֶר זְרִיז֖וּת שָׁלַ֣חְתִּי אֹת֑וֹ לְמַ֗עַן בִּרְאֽוֹתְכֶ֨ם אֹת֥וֹ עוֹד֨ תִּשְׂמָ֔חוּ וַֽאֲנִ֕י עִצְּבוֹנִ֖י יִגָּרֵֽעַ.
2:29	לָכֵ֕ן קַבְּל֥וּהוּ בָֽאָד֖וֹן בְּכָל־שִׂמְחָ֑ה וְהָֽאֲנָשִׁ֣ים כָּאֵ֔לֶּה יִהְי֥וּ לָכֶ֖ם יְקָרִֽים.
2:30	כִּ֗י בַּֽעֲב֨וּר פֹּ֤עַל הַמָּשִׁ֨יחַ֨ קָרַ֣ב עַד־מָ֔וֶת בִּבְלִ֥י הַשְׂכִּ֖יל אֶת־נַפְשׁ֑וֹ לְמַ֣עַן יְמַלֵּ֔א אֶת־מַחְס֖וֹר שֵׁרֽוּתְכֶ֥ם אֵלָֽי.
3:1	בָּאַֽחֲרִית֨ אַחַ֔י שִׂמְח֖וּ בַּֽיהֹוָ֑ה לִכְתּ֨וֹב אֲלֵיכֶ֜ם אֶת־עֶ֣צֶם ׀ הַדְּבָרִ֣ים הָאֵ֗לֶּה אָמְנָ֞ם לֹ֤א קָשֶׁה֨ לִ֔י אַ֥ךְ לָכֶ֖ם בֶּֽטַח.
3:2	רְאוּ֨ אֶת־הַכְּלָבִ֔ים רְא֖וּ אֶת־הַפֹּֽעֲלִ֣ים הָֽרָעִ֑ים רְא֖וּ אֶת־הַֽחִתּֽוּךְ.
3:3	כִּ֤י אֲנַ֨חְנוּ֨ הַמּוּלָ֔ה הָעֹֽבְדִ֤ים אֶת־הָֽאֱלֹהִים֨ בָּר֔וּחַ וּמִתְהַלְלִ֖ים בַּמָּשִׁ֣יחַ יֵשׁ֑וּעַ וְלֹ֥א בַבָּשָׂ֖ר בֹּֽטְחִֽים.
3:4	אַ֣ף ׀ כִּ֣י גַם־אֲנִ֗י יֵ֥שׁ לִ֛י דְּבַ֥ר בִּטָּח֖וֹן גַּ֣ם בַּבָּשָׂ֑ר אִם־אַחֵ֤ר מְדַמֶּה֨ לִבְט֣וֹחַ בַּבָּשָׂ֔ר אֲנִ֖י יוֹתֵֽר.
3:5	נָמ֞וֹל בֶּן־שְׁמוֹנַ֣ת יָמִ֗ים מִמּוֹלֶ֤דֶת יִשְׂרָאֵל֨ מִשֵּׁ֣בֶט בִּנְיָמִ֔ין עִבְרִ֖י מֵֽעִבְרִ֑ים לְפִ֥י הַתּוֹרָ֖ה פָּרֽוּשׁ.
3:6	לְפִ֣י קִנְאָ֔ה רֹדֵ֖ף אֶת־הַקְּהִלָּ֑ה לְפִ֧י הַצְּדָקָ֛ה הַנִּהְיָ֥ה בַתּוֹרָ֖ה בְּלִ֥י שִׁמְצָֽה.
3:7	אוּלָ֕ם הַדְּבָרִ֗ים אֲשֶׁ֛ר הָ֥יוּ לִ֖י בָּ֑צַע אֹתָ֥ם חָשַׁ֛בְתִּי לְמַ֥עַן הַמָּשִׁ֖יחַ לְנֵֽזֶק.
3:8	וְאָמְנָ֗ם אֲנִ֨י חֹשֵׁ֤ב אֶת־הַכֹּל֨ לִהְי֣וֹת נֵ֔זֶק בַּֽעֲב֨וּר יִתְר֥וֹן דַּ֛עַת יֵשׁ֥וּעַ הַמָּשִׁ֖יחַ אֲדֹנִ֑י אֲשֶׁ֨ר לְמַֽעֲנ֜וֹ סָבַ֣לְתִּי ׀ אֶת־נֵ֣זֶק הַכֹּ֗ל וְחֹשֵׁ֨ב אֲנִ֤י אֹתָם֨ לִהְי֣וֹת אַשְׁפּ֔וֹת לְמַ֥עַן אֶקְנֶ֖ה אֶת־הַמָּשִֽׁיחַ.
3:9	וְאֶמָּצֵא֘ בּוֹ֒ בִּבְלִי֩ הֱי֨וֹת לִ֜י צִדְקָתִ֣י שֶׁלִּ֗י אֲשֶׁ֥ר הִיא֨ מִן־הַתּוֹרָ֔ה כִּ֥י אִם־זֹ֔את אֲשֶׁ֛ר עַל־יְדֵ֖י אֱמוּנַ֣ת הַמָּשִׁ֑יחַ הַצְּדָקָ֕ה אֲשֶׁ֥ר מֵֽאֱלֹהִ֖ים בָּֽאֱמוּנָֽה.
3:10	לְמַ֣עַן ׀ דַּ֣עַת אֹת֗וֹ וְאֵת֨ גְּבוּרַ֣ת תְּקֽוּמָת֔וֹ וְאֵ֖ת הִתְחַבְּר֣וּת עִנּוּיָ֑יו בְּהִשָּֽׁווֹתִ֖י לְתַבְנִ֥ית מוֹתֽוֹ.
3:11	אִ֤ם אֵפוֹא֨ אַגִּ֔יעַ אֶל־הַתְּקוּמָ֖ה מֵעִ֥ם הַמֵּתִֽים.
3:12	לֹ֗א כִּ֚י כְּבָ֣ר לָקַ֔חְתִּי א֖וֹ כְּבָ֣ר הָשְׁלָ֑מְתִּי אַ֣ךְ ׀ אֲנִ֣י רֹדֵ֗ף אִם֩ גַּ֨ם אֶתְפֹּ֜שׂ אֵ֠ת אֲשֶׁ֤ר בַּֽעֲבוּרוֹ֨ גַּ֣ם נִתְפַּ֔שְׂתִּי מֵאֵ֥ת הַמָּשִׁ֖יחַ יֵשֽׁוּעַ.
3:13	אַחַ֕י אֵ֥ין אֲנִ֛י מְחַשֵּׁ֥ב אֶת־עַצְמִ֖י כִּ֣י תָפָ֑שׂתִּי אַ֣ךְ אַחַ֔ת שָׁכֹ֗חַ אֵ֤ת הַדְּבָרִים֨ אֲשֶׁ֣ר לְאָח֔וֹר וְנָטֹ֥ה אֶל־אֵ֖לֶּה אֲשֶׁ֥ר לְפָנִֽים.
3:14	אֶל־הַמַּטָּרָ֖ה אֲנִ֣י רֹדֵ֑ף לְכֶ֨תֶר הַנִּצָּח֜וֹן אֲשֶׁ֨ר לִקְרִיאַ֧ת הָֽאֱלֹהִ֛ים מִמַּ֖עְלָה בַּמָּשִׁ֥יחַ יֵשֽׁוּעַ.
3:15	לָכֵ֕ן כֻּלָּ֗נוּ אֲשֶׁ֛ר אֲנַ֥חְנוּ שְׁלֵמִ֖ים אֶת־זֹ֣את נַשְׂכִּ֑ילָה וְאִ֨ם מְא֤וּמָה אַחֶ֨רֶת֨ אַתֶּ֣ם מַשְׂכִּילִ֔ים גַּם־זֹ֕את הָֽאֱלֹהִ֖ים יְגַלֶּ֥ה לָכֶֽם.
3:16	אֶ֚פֶס אֶל־אֲשֶׁ֣ר הִגַּ֔עְנוּ בְּעֶ֛צֶם הַקַּ֥ו הַה֖וּא נִתְהַלֵּ֑כָה וְעֶ֛צֶם הַדָּבָ֥ר הַה֖וּא נַשְׂכִּֽילָה.
3:17	הֱי֨וּ מִתְהַלְּכִ֥ים יַחְדָּ֛יו בְּתַבְנִיתִ֖י אֶחָ֑י וְהַבִּ֨יטוּ֨ אֶל־הַמִּתְהַלְּכִ֣ים כָּ֔כָה כַּֽאֲשֶׁ֥ר אֲנַ֛חְנוּ מֽוּשָׂמִ֥ים לָכֶ֖ם לְתַבְנִֽית.
3:18	כִּ֣י ׀ רַבִּ֣ים מִתְהַלְּכִ֗ים אֲשֶׁ֤ר לִפְעָמִים֨ אָמַ֣רְתִּי לָכֶ֔ם וְעַתָּ֕ה גַּ֥ם בִּבְכ֖וֹת אֲנִ֣י אֹמֵ֑ר כִּ֣י הֵ֔ם אֹֽיְבֵ֖י צְלַ֥ב הַמָּשִֽׁיחַ.
3:19	אֲשֶׁ֨ר אַֽחֲרִיתָ֜ם אֲבַדּ֗וֹן אֲשֶׁ֤ר אֱלֹֽהֵיהֶ֣ם הַבֶּ֔טֶן וּכְבוֹדָ֖ם בְּבָשְׁתָּ֑ם הַמַּשְׂכִּילִ֖ים אֵ֥ת הָֽאַרְצִֽיּוֹת.
3:20	כִּ֥י אֶזְרָֽחוּתֵ֖נוּ בַּשָּׁמַ֣יִם יֵ֑שׁ אֲשֶׁ֣ר מִשָּׁ֗ם אֲנַ֜חְנוּ גַּ֤ם מְצַפִּים֨ לַמוֹשִׁ֔יעַ לָֽאָד֖וֹן יֵשׁ֥וּעַ הַמָּשִֽׁיחַ.
3:21	אֲשֶׁ֤ר יְחַלֵּף֨ אֶת־גְּוִיַּ֣ת שִׁפְלוּתֵ֔נוּ לִהְיוֹתָ֛הּ שָׁוַ֥ת הַתַּבְנִ֖ית לִגְוִיַּ֣ת כְּבוֹ֑דוֹ כְּפִ֣י הַפְּעֻלָּ֔ה אֲשֶׁ֨ר בָּ֜הּ ה֣וּא יָכ֗וֹל גַּ֧ם לְהַכְנִ֛יעַ אֵלָ֖יו אֶת־הַכֹּֽל.
4:1	עַל־כֵּ֗ן אַחַי֨ הַנֶּֽאֱהָבִ֣ים וְהַֽחֲשׁוּקִ֔ים שִׂמְחָתִ֖י וַֽעֲטַרְתִּ֑י כָּ֛כָה עִמְד֥וּ בַֽיהֹוָ֖ה אֲהוּבִֽים.
4:2	אֶת־אֶוְהֹֽדִיָּה֨ אֲנִ֣י מַזְהִ֔יר וְאֶת־סֽוּנִטוּכָ֖ה אֲנִ֣י מַזְהִ֑יר לְהַשְׂכִּ֗יל אֶת־עֶ֛צֶם הַדָּבָ֥ר הַה֖וּא בָּֽאָדֽוֹן.
4:3	וּמְבַקֵּ֤שׁ אֲנִי֨ גַּ֣ם מִמְּךָ֔ חֲבֵר֥ עֻלִּ֖י הַזָּ֑ךְ עֲזֹ֣ר עִמָּהֶ֗ן אֲשֶׁ֨ר בַּבְּשׂוֹרָ֜ה נִלְחֲמ֤וּ עִמִּי֨ וְעִ֣ם קְלִימֶ֔נְסְ וּשְׁאָר֨ חַבְרֵ֣י עֲבֹֽדָתִ֔י אֲשֶׁ֥ר שְׁמוֹתָ֖ם בְּסֵ֥פֶר הַֽחַיִּֽים.
4:4	שִׂמְח֥וּ בַֽיהֹוָ֖ה תָּמִ֑יד וְע֛וֹד אֲנִ֥י אֹמֵ֖ר שְׂמָֽחוּ.
4:5	נַ֧חַת רְצֽוֹנְכֶ֛ם יִוָּדַ֖ע לְכֹ֣ל בְּנֵֽי־אָדָ֑ם הָֽאָד֖וֹן קָרֽוֹב.
4:6	מְא֖וּמָה אַל־תִּדְאָ֑גוּ כִּ֣י אִם־בַּכֹּ֗ל בַּתְּפִלָּ֤ה וּבַתְּחִנָּה֨ עִם־תּוֹדָ֔ה תִּוָּדַ֥עְנָה מִשְׁאֲלֽוֹתֵיכֶ֖ם אֶל־הָֽאֱלֹהִֽים.
4:7	וּשְׁל֣וֹם הָֽאֱלֹהִ֗ים הַנַּֽעֲלֶה֨ עַ֣ל כָּל־תְּבוּנָ֔ה יִשְׁמֹ֧ר אֶת־לִבּֽוֹתֵיכֶ֛ם וְעֶשְׁתֹּנֽוֹתֵיכֶ֖ם בַּמָּשִׁ֥יחַ יֵשֽׁוּעַ.
4:8	בָּאַֽחֲרִית֘ אַחַי֒ כָּל־הַדְּבָרִ֡ים אֲשֶׁר֩ הֵ֨ם אֱמֶ֜ת כָּל־אֲשֶׁ֣ר ׀ הֵ֣ם כֵּנִ֗ים כָּל־אֲשֶׁר֩ הֵ֨ם יְשָׁרִ֜ים כָּל־אֲשֶׁ֣ר ׀ הֵ֣ם טְהוֹרִ֗ים כָּל־אֲשֶׁ֥ר הֵם֨ נְעִימִ֔ים כָּל־אֲשֶׁ֥ר הֵ֖ם מִשֵּׁ֣מַע ט֑וֹב אִם־יֵ֞שׁ אֵי־זֶ֣ה חַ֗יִל אִ֚ם אֵי־ז֣וּ תְהִלָּ֔ה אֶת־אֵ֖לֶּה חֲשֹֽׁבוּ.
4:9	אֵ֣ת הַדְּבָרִ֡ים אֲשֶׁ֣ר גַּם֩ לְמַדְתֶּ֨ם וְקִבַּלְתֶּ֤ם וּשְׁמַעְתֶּם֨ וּרְאִיתֶ֣ם בִּ֔י אֶת־אֵ֖לֶּה עֲשׂ֑וּ וֵֽאלֹהֵ֥י הַשָּׁל֖וֹם יְהִ֥י עִמָּכֶֽם.
4:10	וְשָׂמַ֨חְתִּי בַֽיהֹוָ֜ה שִׂמְחָ֣ה גְדוֹלָ֗ה כִּ֤י עַתָּה֨ בְּאַֽחֲרִ֔ית הִפְרַחְתֶּ֥ם ע֖וֹד לְהַשְׂכִּ֣יל אֵלָ֑י כַּֽאֲשֶׁר֨ גַּ֥ם הִשְׂכַּלְתֶּ֔ם אַ֥ךְ חֲסַרְתֶּ֖ם תֹּֽאֲנָֽה.
4:11	לֹ֕א כִּ֛י עַל־דְּבַ֥ר מַחְס֖וֹר אֲנִ֣י אֹמֵ֑ר כִּ֣י ׀ אֲנִ֣י לָמַ֗דְתִּי בְּכֹל֨ אֲשֶׁ֣ר אֲנִ֣י בָהֶ֔ם לִהְי֖וֹת מִסְתַּפֵּֽק.
4:12	יֹדֵ֤עַ אֲנִי֨ לְהִשָּׁפֵ֔ל וְיֹדֵ֥עַ אֲנִ֖י גַּ֣ם לְהַעְדִּ֑יף בַּכֹּ֨ל וּבְכָל־דָּבָ֜ר לֻמַּ֗דְתִּי גַּ֤ם לִשְׂבּ֨וֹעַ֨ וְלִרְעַ֔ב גַּ֥ם לְהַעְדִּ֖יף וְלַחְסֽוֹר.
4:13	אֶת־הַכֹּל֨ אֲנִ֣י יָכֹ֔ל בַּמָּשִׁ֖יחַ הַמְּחַזֵּ֥ק אֹתִֽי.
4:14	אֶ֚פֶס הֵֽיטַבְתֶּ֣ם לַֽעֲשׂ֔וֹת בְּחָבְרְכֶ֥ם עִמִּ֖י בְּצָֽרָתִֽי.
4:15	אַ֠ךְ יְדַעְתֶּ֣ם גַּם־אַתֶּם֘ הַפִּֽילִפִּים֒ כִּי֩ בִתְחִלַּ֨ת הַבְּשׂוֹרָ֜ה כַּֽאֲשֶׁ֤ר יָצָ֨אתִי֨ מִמַּקֵּדֹֽנִיָּ֔א אֵ֤ין קְהִלָּה֨ חִלְּקָ֣ה לִ֔י בִּדְבַ֥ר נְתִינָ֖ה וּלְקִיחָ֑ה זֽוּלָתִ֖י אַתֶּ֥ם לְבַדְכֶֽם.
4:16	כִּ֗י גַּ֚ם בְּתֶסָּלֽוֹנִיקִ֔י גַּ֣ם פַּ֤עַם אַחַת֨ וּשְׁתַּ֔ים שְׁלַחְתֶּ֥ם לִ֖י לְצָרְכִּֽי.
4:17	לֹ֛א כִּֽי־אֲנִ֥י מְבַקֵּ֖שׁ אֶת־הַמַּתָּ֑ן כִּ֣י אִם־אֲנִ֧י מְבַקֵּ֛שׁ אֶת־הַפְּרִ֖י הַמַּעְדִּ֥יף לְחִשְׁבּֽוֹנְכֶֽם.
4:18	אַ֥ךְ אֲנִ֛י יֵֽשׁ־לִ֥י הַכֹּ֖ל וַֽאֲנִ֣י מַעְדִּ֑יף נִמְלֵ֗אתִי בְּקַחְתִּ֨י מֵֽאֶפַּפְרֹֽדִיט֜וֹס אֶת־הַדְּבָרִ֣ים ׀ אֲשֶׁ֣ר מֵֽאִתְּכֶ֗ם רֵ֧יחַ נִיחֹ֛חַ זֶ֥בַח מְקֻבָּ֖ל נִרְצֶ֥ה לָֽאֱלֹהִֽים.
4:19	אַ֣ךְ אֱלֹהַ֔י יְמַלֵּ֖א אֶת־כָּל־צָרְכְּכֶ֑ם כְּפִ֤י עָשְׁרוֹ֨ בְּכָב֔וֹד בַּמָּשִׁ֖יחַ יֵשֽׁוּעַ.
4:20	וְלָֽאֱלֹהִ֣ים אָבִ֔ינוּ הַכָּב֛וֹד לְעֽוֹלְמֵ֥י עֽוֹלָמִ֖ים אָמֵֽן.
4:21	שַֽׁאֲל֛וּ לִשְׁל֥וֹם כָּל־קָד֖וֹשׁ בַּמָּשִׁ֣יחַ יֵשׁ֑וּעַ הָֽאַחִים֨ אֲשֶׁ֣ר אִתִּ֔י שֹֽׁאֲלִ֖ים לִשְׁלֽוֹמְכֶֽם.
4:22	כָּל־הַקְּדוֹשִׁ֖ים שֹֽׁאֲלִ֣ים לִשְׁלֽוֹמְכֶ֑ם אַ֤ךְ בְּיוֹתֵר֨ אֵ֔לֶּה אֲשֶׁ֖ר מִבֵּ֥ית הַקֵּיסָֽר.
4:23	חֵ֣ן אֲדֹנֵ֜ינוּ יֵשׁ֧וּעַ הַמָּשִׁ֛יחַ עִם־כֻּלְכֶ֖ם אָמֵֽן.
